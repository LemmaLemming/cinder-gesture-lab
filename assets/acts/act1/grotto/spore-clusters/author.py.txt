#!/usr/bin/env python3
"""Original hand-set low gill tufts. No source image pixels are read to draw.

Static decoration only: available, lifted/slack active and fallen spent art.
This inert source is preserved for review; no native runtime wiring is authored.
"""
from pathlib import Path
import hashlib
import json
import sys
from PIL import Image, ImageDraw

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[4]
STATES = ("available", "active", "spent")
PALETTE = {
    "ink": (20, 20, 20, 255), "coal": (39, 39, 39, 255),
    "shade": (67, 67, 67, 255), "mid": (105, 105, 105, 255),
    "silver": (147, 147, 147, 255), "light": (193, 193, 193, 255),
    "white": (231, 231, 231, 255),
}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def poly(d, points, material):
    d.polygon(points, fill=PALETTE[material])
    d.line(points + [points[0]], fill=PALETTE["ink"], width=1)


def line(d, points, material, width=1):
    d.line(points, fill=PALETTE[material], width=width)


def draw(state):
    im = Image.new("RGBA", (24, 20), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    # Actual drawn substrate, not a shadow, cue ring or field boundary.
    poly(d, [(3, 18), (7, 16), (16, 16), (21, 18), (20, 19),
             (4, 19)], "coal")
    line(d, [(5, 18), (8, 17), (16, 17), (19, 18)], "shade")
    if state == "available":
        # A short pale stalk supports a dense low tuft of irregular gill fans.
        poly(d, [(10, 12), (13, 12), (14, 18), (10, 18)], "silver")
        line(d, [(11, 13), (11, 17)], "light")
        poly(d, [(8, 17), (3, 11), (4, 8), (7, 6), (10, 9),
                 (11, 12)], "light")
        line(d, [(5, 9), (7, 7), (9, 10)], "white")
        line(d, [(5, 11), (8, 15)], "mid")
        line(d, [(8, 10), (9, 14)], "silver")
        poly(d, [(11, 16), (8, 8), (10, 4), (13, 4), (16, 7),
                 (15, 11)], "light")
        line(d, [(10, 6), (12, 5), (14, 7)], "white")
        line(d, [(10, 9), (12, 14)], "mid")
        line(d, [(13, 8), (13, 12)], "silver")
        poly(d, [(15, 17), (13, 11), (15, 7), (18, 7), (21, 10),
                 (21, 12), (18, 15)], "light")
        line(d, [(15, 9), (17, 8), (19, 10)], "white")
        line(d, [(15, 11), (16, 15)], "mid")
        line(d, [(19, 12), (17, 15)], "silver")
        line(d, [(8, 17), (11, 18), (16, 17)], "silver")
    elif state == "active":
        # Lifted, slack petals stay connected to the same base. No particles.
        poly(d, [(10, 12), (13, 12), (14, 18), (10, 18)], "silver")
        line(d, [(11, 13), (11, 17)], "light")
        poly(d, [(10, 17), (6, 13), (2, 8), (3, 6), (5, 6),
                 (8, 11), (11, 13)], "light")
        line(d, [(3, 8), (5, 8), (7, 11)], "white")
        line(d, [(5, 11), (9, 15)], "mid")
        poly(d, [(12, 16), (8, 6), (11, 2), (14, 3), (14, 6),
                 (12, 10), (14, 15)], "light")
        line(d, [(10, 5), (11, 3), (13, 4)], "white")
        line(d, [(10, 7), (12, 13)], "mid")
        line(d, [(13, 6), (12, 9)], "silver")
        poly(d, [(15, 17), (15, 13), (16, 9), (19, 5), (21, 6),
                 (21, 9), (19, 12), (17, 16)], "light")
        line(d, [(18, 8), (19, 6), (20, 7)], "white")
        line(d, [(18, 10), (16, 15)], "mid")
        line(d, [(8, 17), (11, 18), (16, 17)], "silver")
    else:
        assert state == "spent"
        # Fallen gill folds and short stalk remain as a quiet physical remnant.
        poly(d, [(10, 14), (12, 12), (14, 14), (13, 18),
                 (10, 18)], "mid")
        line(d, [(11, 14), (12, 13), (13, 14)], "silver")
        poly(d, [(3, 16), (4, 13), (7, 12), (10, 13), (12, 16),
                 (13, 18), (6, 18)], "silver")
        line(d, [(5, 14), (7, 13), (9, 14)], "light")
        line(d, [(5, 16), (8, 17), (10, 16)], "shade")
        line(d, [(7, 14), (9, 16)], "white")
        poly(d, [(14, 15), (17, 12), (20, 13), (22, 16),
                 (20, 18), (15, 17)], "silver")
        line(d, [(17, 14), (19, 14), (20, 15)], "light")
        line(d, [(16, 16), (19, 17)], "shade")
        line(d, [(18, 15), (20, 16)], "white")
    return im

        })
        expanded = im.resize((192, 160), Image.Resampling.NEAREST)
        sd.text((i * 240 + 12, 9), f"cluster / {state}", fill=(231, 231, 231))
        sheet.paste(expanded, (i * 240 + 24, 35), expanded)
        sd.text((i * 240 + 12, 201), "24x20 / same feet pivot", fill=(193, 193, 193))
    assert len({a["sha256"] for a in assets}) == 3

        "sprite_offset_centered_pixels": [0, 10], "pixel_size_provisional": 0.016,

        "existing_cluster_framing_box": {"size_world": [0.4, 0.5, 0.4],

        "preview": {"file": sheet_path.name, "sha256": sha(sheet_path), "bytes": sheet_path.stat().st_size,
                    "native_nearest_scale": 8, "role": "static review sheet, not a runtime atlas or screenshot"},
        "assets": assets,
        "validation": {"native_dimensions": 3, "hard_alpha": 3, "same_palette": 3,
                       "aligned_canvas_feet_pivot": 3, "bottom_substrate_y19": 3,
