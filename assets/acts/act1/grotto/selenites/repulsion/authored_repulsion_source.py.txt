#!/usr/bin/env python3
"""Hand-authored environmental poses; no source image pixels are read to draw.

Reuse the original costume's integer drawing primitives, but construct new
limb/head/weapon arrangements. Static art has no animation or physics clock.
"""
from pathlib import Path
import hashlib
import importlib.util
import json
import shutil
import sys
from PIL import Image, ImageDraw, ImageFont

sys.dont_write_bytecode = True
ROOT = Path(__file__).resolve().parents[1]
OLD = ROOT / "assets/acts/act1/grotto/selenites"
OUT = OLD / "repulsion"
PRIMITIVES = OLD / "build_selenite_cutouts.py"
EXPECTED_PRIMITIVES = "676c2a4ae78173e2be63beeea27bdd89c4056a893ce4b122f5d6a347363505af"
STATES = ("recoil", "retreat", "regroup")
FACINGS = ("front", "side", "back")
ROLES = {
    "swarm": {"entity_id": "C31", "design_id": "A1-E2", "basis_id": "C20", "width": 48,
              "family": "act1_grotto_selenite_C31_swarm"},
    "guard": {"entity_id": "C32", "design_id": "A1-E3", "basis_id": "C22", "width": 80,
              "family": "act1_grotto_selenite_C32_spear_guard"},
}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


assert sha(PRIMITIVES) == EXPECTED_PRIMITIVES, "Original primitive source changed"
spec = importlib.util.spec_from_file_location("original_selenite_primitives", PRIMITIVES)
old = importlib.util.module_from_spec(spec)
spec.loader.exec_module(old)  # Defines primitives only; NEVER call old.main().


def shoulder(d, point, small):
    x, y = point
    if small:
        old.rect(d, (x - 2, y - 1, x + 2, y + 2), "shade")
    else:
        old.poly(d, [(x - 3, y - 2), (x + 2, y - 2), (x + 3, y + 3),
                     (x - 3, y + 3)], "shade", "ink")
    old.rect(d, (x - 1, y - 1, x, y + 1), "silver")


def carried_spear(d, state):
    # The same broad triangular theatrical prop, newly carried/tucked/lowered.
    # These endpoints are picture coordinates, never an attack or reach model.
    ends = {
        "recoil": ((52, 72), (68, 36), (73, 20)),
        "retreat": ((20, 71), (65, 53), (76, 45)),
        "regroup": ((28, 55), (61, 69), (75, 76)),
    }
    tail, neck, tip = ends[state]
    d.line([tail, neck, tip], fill=old.PALETTE["ink"], width=5)
    d.line([tail, neck], fill=old.PALETTE["silver"], width=3)
    d.line([(tail[0], tail[1] - 1), (neck[0], neck[1] - 1)],
           fill=old.PALETTE["white"], width=1)
    dx, dy = tip[0] - neck[0], tip[1] - neck[1]
    ax, ay = -dy, dx
    denom = max(abs(ax), abs(ay))
    ax, ay = round(ax * 8 / denom), round(ay * 8 / denom)
    old.poly(d, [(neck[0] + ax, neck[1] + ay), tip,
                 (neck[0] - ax, neck[1] - ay), (neck[0] - 2, neck[1] + 2)],
             "light", "ink")
    old.poly(d, [neck, tip, (neck[0] - ax + 1, neck[1] - ay + 1)], "white")
    d.line([neck, tip], fill=old.PALETTE["silver"], width=1)
    d.line([(neck[0] - 2, neck[1] + 1), (neck[0] + 2, neck[1] - 1)],
           fill=old.PALETTE["ink"], width=2)


def draw_swarm(facing, state):
    im = Image.new("RGBA", (48, 80), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    base = 24 if facing != "side" else 25
    # A lean changes pixels inside the retained quad, never the actor origin.
    cx = base - 2 if state == "recoil" else base
    cy = {"recoil": 34, "retreat": 31, "regroup": 32}[state]
    top, bottom = cy + 10, 59 if state == "recoil" else 58
    lean = {"recoil": 2, "retreat": -2, "regroup": 0}[state]
    legs = {
        "recoil": [((cx - 8, 67), (base - 11, 79, -1)),
                   ((cx + 8, 69), (base + 10, 79, 1))],
        "retreat": [((cx - 9, 65), (base - 12, 75, -1)),
                    ((cx + 8, 67), (base + 10, 79, 1))],
        "regroup": [((cx - 6, 67), (base - 8, 79, -1)),
                    ((cx + 5, 67), (base + 8, 79, 1))],
    }[state]
    for hip, (knee, (fx, fy, direction)) in zip(
            [(cx - 3 + lean, bottom), (cx + 3 + lean, bottom)], legs):
        old.limb(d, [hip, knee, (fx, fy - 4)], 4)
        old.band(d, knee); old.band(d, (fx, fy - 6))
        old.boot(d, fx, fy, direction)
    old.torso(d, cx, top, bottom, facing, True, lean)
    elbows, wrists = {
        "recoil": ([(cx - 10, 49), (cx + 10, 46)],
                   [(cx - 7, 40), (cx + 8, 39)]),
        "retreat": ([(cx - 11, 49), (cx + 10, 45)],
                    [(cx - 16, 56), (cx + 13, 39)]),
        "regroup": ([(cx - 10, 52), (cx + 9, 51)],
                    [(cx - 8, 60), (cx + 10, 57)]),
    }[state]
    shoulders = [(cx - 6, top + 1), (cx + 5, top + 1)]
    for origin, elbow, wrist in zip(shoulders, elbows, wrists):
        old.limb(d, [origin, elbow, wrist], 4)
        old.band(d, elbow); shoulder(d, origin, True)
    old.mask(d, cx + (2 if state == "retreat" else 0), cy, facing, True)
    for wrist in wrists:
        old.hand(d, wrist)
    return im


def draw_guard(facing, state):
    im = Image.new("RGBA", (80, 80), (0, 0, 0, 0))
    d = ImageDraw.Draw(im)
    base = 40 if facing != "side" else 37
    cx = base - 2 if state == "recoil" else base
    cy = {"recoil": 31, "retreat": 29, "regroup": 28}[state]
    top, bottom = cy + 11, 60
    lean = {"recoil": 3, "retreat": -2, "regroup": 0}[state]
    # Carried spear is behind cloth and hands, with an intact visible arrowhead.
    carried_spear(d, state)
    legs = {
        "recoil": [((cx - 10, 68), (base - 13, 79, -1)),
                   ((cx + 11, 69), (base + 13, 79, 1))],
        "retreat": [((cx - 10, 65), (base - 13, 75, -1)),
                    ((cx + 9, 68), (base + 12, 79, 1))],
        "regroup": [((cx - 7, 68), (base - 10, 79, -1)),
                    ((cx + 6, 68), (base + 10, 79, 1))],
    }[state]
    for hip, (knee, (fx, fy, direction)) in zip(
            [(cx - 4 + lean, bottom), (cx + 4 + lean, bottom)], legs):
        old.limb(d, [hip, knee, (fx, fy - 4)], 5)
        old.band(d, knee); old.band(d, (fx, fy - 6))
        old.boot(d, fx, fy, direction)
    old.torso(d, cx, top, bottom, facing, False, lean)
    elbows, wrists = {
        "recoil": ([(cx - 13, 47), (55, 55)], [(cx - 9, 38), (58, 56)]),
        "retreat": ([(cx - 14, 57), (50, 50)], [(31, 66), (53, 58)]),
        "regroup": ([(cx - 12, 53), (cx + 11, 56)], [(33, 58), (54, 66)]),
    }[state]
    shoulders = [(cx - 8, top), (cx + 7, top)]
    for origin, elbow, wrist in zip(shoulders, elbows, wrists):
        old.limb(d, [origin, elbow, wrist], 5)
        old.band(d, elbow); shoulder(d, origin, False)
    old.mask(d, cx + (3 if facing == "side" else 0) +
             (2 if state == "retreat" else 0), cy, facing)
    for wrist in wrists:
        old.hand(d, wrist)
    return im

    originals = []

        originals.append({"path": str(path.relative_to(ROOT)),

    assert len(originals) == 30
    # Confirm the personally inspected old sheet is an exact display of all30.
    existing_sheet = Image.open(OLD / "selenite_contact_sheet.png").convert("RGB")

        for facing_index, facing in enumerate(FACINGS):
            for pose_index, pose in enumerate(old.POSES):
                native = Image.open(OLD / f"{role}_{facing}_{pose}.png").convert("RGBA")
                expanded = native.resize((native.width * 3, native.height * 3), Image.Resampling.NEAREST)
                expected = Image.new("RGB", expanded.size, (65, 65, 65))
                expected.paste(expanded, (0, 0), expanded)
                left = pose_index * 250 + (250 - expanded.width) // 2
                upper = (role_index * 3 + facing_index) * 270 + 25
                assert existing_sheet.crop((left, upper, left + expanded.width,
                                            upper + expanded.height)).tobytes() == expected.tobytes()
    assets = []
    sheet = Image.new("RGB", (3 * 260, 6 * 280), (65, 65, 65))
    sd = ImageDraw.Draw(sheet)
    for role_index, (role, meta) in enumerate(ROLES.items()):
        for facing_index, facing in enumerate(FACINGS):
            for state_index, state in enumerate(STATES):
                im = (draw_swarm if role == "swarm" else draw_guard)(facing, state)
                path = OUT / f"{role}_{facing}_{state}.png"
                im.save(path)
                colors = set(im.get_flattened_data()); alpha = {v[3] for v in colors}
                assert alpha == {0, 255}
                assert colors <= set(old.PALETTE.values()) | {(0, 0, 0, 0)}
                bounds = im.getchannel("A").getbbox()
                assert bounds[0] > 0 and bounds[1] > 0 and bounds[2] < im.width
                assert bounds[3] == 80
                assert sha(path) not in {v["sha256"] for v in originals}
                width = meta["width"]
                item = {
                    "id": f"ACT1-{meta['entity_id']}-{facing.upper()}-ENV-{state.upper()}",
                    "file": path.name, "sha256": sha(path), "bytes": path.stat().st_size,

                    "basis_id": meta["basis_id"], "reuse_family": meta["family"],

                    "environmental_pose": state, "native_grid": [width, 80],
                    "feet_pivot_pixels": [width // 2, 80],
                    "sprite_offset_centered_pixels": [0, 40], "pixel_size": 0.022,
                    "full_quad_world": [width * 0.022, 1.76],
                    "full_quad_from_feet_world": [-width * 0.011, 0.0, width * 0.011, 1.76],
                    "opaque_bounds_exclusive_pixels": list(bounds),
                    "opaque_bounds_from_feet_world": [(bounds[0] - width / 2) * 0.022,
                        (80 - bounds[3]) * 0.022, (bounds[2] - width / 2) * 0.022,
                        (80 - bounds[1]) * 0.022],
                    "opaque_pixels": sum(1 for v in im.get_flattened_data() if v[3] == 255),
                    "collision": "none; existing actor capsule remains authoritative",
                    "occlusion": "ordinary-depth actor art; full quad must join public camera/art custody before runtime",
                    "timing": "none; static pose, no duration/clock/phase threshold/root motion",
                    "readiness": "original_pixels_validated_static_only; not_imported_or_bound_or_spore_portrait_tested",
                    "source_ids": ["G10", "G25", "F10"],
                }
                assets.append(item)
                x, y = state_index * 260, (role_index * 3 + facing_index) * 280
                sd.text((x + 8, y + 5), f"{meta['entity_id']} {facing} / {state}", fill=(231, 231, 231))
                expanded = im.resize((width * 3, 240), Image.Resampling.NEAREST)
                sheet.paste(expanded, (x + (260 - expanded.width) // 2, y + 29), expanded)

    assert len(assets) == 18 and len({a["sha256"] for a in assets}) == 18

    refs = []

        "palette": old.PALETTE,
        "palette_role": "same seven grayscale costume materials; game interpretation, not authentic film color",

            "recoil": "living intact actor shields body; guard tucks carried spear",
