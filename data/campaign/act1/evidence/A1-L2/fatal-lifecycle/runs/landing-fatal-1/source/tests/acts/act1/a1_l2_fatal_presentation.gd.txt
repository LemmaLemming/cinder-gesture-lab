extends SceneTree
## Narrow production Shell landing-fatal regression. Registry acceptance and the
## generic L1 handoff are TEST ONLY. HP2/zero shells are explicit predecessor
## initial conditions, carried by the real transition into a legitimate living
## L2 entry checkpoint. Native landing damage causes death; no forced phase,
## clock, motion, authored checkpoint or private pause repair is used.
const Shell = preload("res://scripts/campaign/shell.gd")
const Registry = preload("res://scripts/campaign/registry.gd")
const Exact = preload("res://scripts/campaign/exact_json.gd")
const Codec = preload("res://scripts/campaign/snapshot_codec.gd")
const LEVEL_PATH: String = "res://scenes/acts/act1/a1_l2.tscn"
const HANDOFF_PATH: String = "res://tests/fixtures/campaign/live_a1_l1.tscn"
var game: CinderCampaignShell
var raw: Dictionary
var test_root: String
var registry_text: String
var checks: int = 0
var failures: int = 0
var finishing: bool = false
var death_count: int = 0
var death_clock_s: float = 0.0
var death_shells: int = 0
var hits: Array[Dictionary] = []
var progress: Array[String] = []
var old_fps: int
var old_audio: AudioBusLayout


func _initialize() -> void:
	test_root = "user://test-act1-crater-fatal-%d/" % OS.get_process_id()
	old_fps = Engine.max_fps
	old_audio = AudioServer.generate_bus_layout()
	_run.call_deferred()


func _run() -> void:
	create_timer(40.0, true).timeout.connect(func() -> void:
		if not finishing:
			_diagnose("watchdog")
			_expect(false, "bounded actual landing-fatal fixture completes within40 seconds")
			_finish())
	root.size = Vector2i(540, 1170)
	_cleanup_files()
	registry_text = FileAccess.get_file_as_string(Registry.DATA_PATH)
	raw = JSON.parse_string(registry_text)
	for info: Dictionary in raw.levels:
		if info.id in ["A1-L1", "A1-L2"]:
			info.scene_path = HANDOFF_PATH if info.id == "A1-L1" else LEVEL_PATH
			info.readiness = "accepted"
			info.accepted_commit = "f".repeat(40)
			info.api_revision = Registry.API_REVISION
	if not _new_shell():
		_finish(); return
	await _settle()
	_expect(paused and game.active_level == null and game.menu.page_name() == "title", "isolated actual production Title starts paused without an active scene")
	if not _press("BeginStoryButton"):
		_finish(); return
	await _settle()
	if not _expect(game.campaign_error.is_empty() and is_instance_valid(game.active_level) and game.active_level.scene_file_path == HANDOFF_PATH, "public Begin starts only the existing generic test L1 handoff: " + game.campaign_error):
		_diagnose("Begin"); _finish(); return
	# Staged test INITIAL resources, not a heal or a forged authored boundary.
	# No native physics runs during the real completion/contact handoff.
	game.player.hp = 2.0
	game.player.shells = 0
	var accepted: bool = game.active_level.request_completion("test-fatal-handoff")
	if not _expect(accepted, "generic test predecessor accepts its actual public completion request"):
		_finish(); return
	await _settle()
	accepted = game.active_level.request_contact_exit("test-fatal-contact", game.player)
	if not _expect(accepted, "generic test predecessor accepts its same-Hero contact request"):
		_finish(); return
	await _settle()
	if not _expect(_actual_ready(), "real transition prepares and installs the actual L2 in its own candidate world: " + game.campaign_error):
		_diagnose("L2 transition"); _finish(); return
	var checkpoint: Dictionary = game.attempts.state().story.checkpoint.duplicate(true)
	var checkpoint_wire: String = Exact.stringify(checkpoint)
	var captured: Dictionary = game.capture_campaign_snapshot()
	if not _expect(not checkpoint.is_empty() and not captured.is_empty() and Exact.stringify(captured) == checkpoint_wire, "real L2 entry checkpoint is the complete exact installed living unit"):
		_diagnose("entry checkpoint"); _finish(); return
	_expect(checkpoint.player.resources.hp == 2.0 and not checkpoint.player.resources.dead and checkpoint.player.resources.shells == 0 and checkpoint.level.local.beat_index == 0 and checkpoint.level.progress.checkpoint_ids.is_empty(), "actual entry retains HP2/zero ammo without fabricating an authored L2 checkpoint")
	_expect(game.attempts.state().completed_main == ["A1-L1"] and not game.active_level.is_completed() and _presentation_ready(), "actual L2 opens unfinished with all future sources pristine and required presentation bound")
	var entry_envelope: Dictionary = _envelope()
	var entry_generation: int = int(entry_envelope.get("generation", -1))
	var identities: Dictionary = _identities()
	game.player.died.connect(_notice_death)
	game.active_level.checkpoint_requested.connect(func(_id: String, id: String, _kind: String) -> void: progress.append(id))
	game.active_level.completion_requested.connect(func(_id: String, id: String) -> void: progress.append(id))
	game.active_level.contact_exit_requested.connect(func(_id: String, id: String) -> void: progress.append(id))
	var landing: CinderLaneMechanism = (game.active_level.get("circles") as Dictionary)["landing-impact"]
	landing.hit_resolved.connect(func(id: String, cycle: int, result: Dictionary) -> void:
		hits.append({"hero_id": id, "cycle": cycle, "result": result.duplicate(true)}))
	if not _press("ResumeButton"):
		_finish(); return
	var died_naturally: bool = await _wait_for(func() -> bool: return death_count > 0, 800)
	await _settle()
	_diagnose("completed fatal boundary")
	if not _expect(died_naturally and death_count == 1 and game.player.dead and game.player.hp == 0.0, "real native landing damage delivers one actual Hero death"):
		_finish(); return
	_expect(hits.size() == 1 and hits[0].hero_id == "hero" and hits[0].cycle == 1 and hits[0].result.opportunity_consumed and hits[0].result.accepted and hits[0].result.hp_damage == 2.0, "one native landing opportunity supplies the genuine fatal HP loss")
	_expect(paused and game.menu.page_name() == "pause" and game.campaign_error.is_empty(), "production death drain reaches the normal paused save menu: " + game.campaign_error)
	var fatal: Dictionary = game.capture_campaign_snapshot()
	if not _expect(not fatal.is_empty(), "whole fatal Player/Level/Shell captures at the deferred completed-tick barrier: " + game.campaign_error):
		_finish(); return
	var fatal_wire: String = Exact.stringify(fatal)
	_expect(fatal.player.resources.dead and fatal.player.resources.hp == 0.0 and fatal.player.resources.shells == death_shells and fatal.level.local.scheduler.clock_s == death_clock_s, "fatal unit keeps actual resources and the exact lethal native tick without an added simulation step")
	_expect(_identities() == identities and _presentation_ready(), "fatal cancellation retains actual source/puff/cue identities and their truthful required presentation")
	_expect(fatal.level.local.beat_index == 0 and fatal.level.local.completed_beats.is_empty() and fatal.level.progress.checkpoint_ids.is_empty() and not fatal.level.progress.completed and fatal.level.progress.contact_exit_id.is_empty() and progress.is_empty(), "death grants no landing clear, checkpoint, future activation, completion or exit")
	_expect(Exact.stringify(fatal.level.local.sources) == Exact.stringify(checkpoint.level.local.sources) and fatal.level.local.scheduler.reservations.is_empty() and fatal.level.local.framing.is_empty() and fatal.level.local.custody.is_empty() and fatal.level.local.sequence.is_empty(), "fatal cancellation preserves all four pristine future actors and retires actual leases/witnesses")
	var impact: Dictionary = fatal.level.local.circles["landing-impact"]
	_expect(impact.status == "cancelled" and impact.phase == "clear" and impact.cycle == 1 and impact.last_cancel_reason == "actual_player_death" and impact.hit_ids == ["hero"], "landing retains its genuine fatal cycle and consumed opportunity with a clear cancelled cue")
	var decoded: Dictionary = Exact.parse(fatal_wire)
	_expect(not fatal_wire.is_empty() and decoded.get("accepted", false) and decoded.get("value") is Dictionary and Exact.stringify(decoded.value) == fatal_wire, "whole fatal aggregate retains exact native scalar bits through public transport")
	var attempts_wire: String = Exact.stringify(game.attempts.state())
	var fatal_disk: String = FileAccess.get_file_as_string(test_root + "campaign.json")
	var fatal_envelope: Dictionary = _envelope()
	_expect(Exact.stringify(game.attempts.active_snapshot()) == fatal_wire and Exact.stringify(game.attempts.state().story.checkpoint) == checkpoint_wire and _durable_matches(), "actual fatal active save preserves the original living entry checkpoint and complete exact disk payload")
	_expect(int(fatal_envelope.get("generation", -1)) == entry_generation + 1, "one actual fatal pause publishes one new durable generation")
	await create_timer(0.08, true).timeout
	captured = game.capture_campaign_snapshot()
	_expect(Exact.stringify(captured) == fatal_wire and Exact.stringify(game.attempts.state()) == attempts_wire and FileAccess.get_file_as_string(test_root + "campaign.json") == fatal_disk, "held fatal pause changes no resource, presentation, clock, attempt or generation")
	if not _press("ResumeButton"):
		_finish(); return
	await _settle()
	captured = game.capture_campaign_snapshot()
	_expect(paused and game.player.dead and game.menu.page_name() == "pause" and Exact.stringify(captured) == fatal_wire and _identities() == identities and Exact.stringify(game.attempts.state()) == attempts_wire and FileAccess.get_file_as_string(test_root + "campaign.json") == fatal_disk, "actual public Resume rejects the dead Hero without native state or durable mutation")
	game.free()
	await process_frame
	if not _new_shell():
		_finish(); return
	await _settle()
	_diagnose("fresh fatal Title")
	if not _expect(paused and game.active_level == null and game.menu.page_name() == "title" and game.campaign_error.is_empty(), "fresh production Title validates the exact fatal saved unit in a candidate world: " + game.campaign_error):
		_finish(); return
	_expect(Exact.stringify(game.attempts.active_snapshot()) == fatal_wire and Exact.stringify(game.attempts.state().story.checkpoint) == checkpoint_wire and FileAccess.get_file_as_string(test_root + "campaign.json") == fatal_disk, "fresh load retains fatal active/living checkpoint and writes no replacement generation")
	if not _press("ContinueStoryButton"):
		_finish(); return
	await _settle()
	_diagnose("fresh fatal Continue")
	captured = game.capture_campaign_snapshot()
	if not _expect(_actual_ready() and not captured.is_empty() and Exact.stringify(captured) == fatal_wire and game.player.dead and _presentation_ready(), "actual fresh Continue restores the complete dead state and required presentation exactly: " + game.campaign_error):
		_finish(); return
	var continued: Dictionary = _identities()
	_expect(_all_new(identities, continued) and FileAccess.get_file_as_string(test_root + "campaign.json") == fatal_disk, "Continue installs new real world/Hero/source identities without changing the durable generation")
	if not _press("ResumeButton"):
		_finish(); return
	await _settle()
	captured = game.capture_campaign_snapshot()
	_expect(paused and game.player.dead and game.menu.page_name() == "pause" and Exact.stringify(captured) == fatal_wire and FileAccess.get_file_as_string(test_root + "campaign.json") == fatal_disk, "fresh dead Continue also requires Retry and rejects public Resume unchanged")
	if not _press("RetryButton"):
		_finish(); return
	await _settle()
	_diagnose("living checkpoint Retry")
	captured = game.capture_campaign_snapshot()
	if not _expect(_actual_ready() and not captured.is_empty() and Exact.stringify(captured) == checkpoint_wire, "actual public Retry restores the exact earlier living entry unit: " + game.campaign_error):
		_finish(); return
	_expect(_all_new(continued, _identities()) and paused and game.menu.page_name() == "resume" and not game.player.dead and game.player.hp == 2.0 and game.player.shells == 0 and _presentation_ready(), "Retry installs new native identities at original living HP2/zero ammo without extra healing/refill")
	var retry_envelope: Dictionary = _envelope()
	_expect(int(retry_envelope.get("generation", -1)) == int(fatal_envelope.get("generation", -1)) + 1 and _durable_matches() and Exact.stringify(game.attempts.state().story.checkpoint) == checkpoint_wire and game.attempts.state().completed_main == ["A1-L1"], "Retry publishes only its restored active generation and preserves original checkpoint/progress")
	_finish()


func _notice_death() -> void:
	# Read-only scalars from the real callback. No snapshot is attempted inside
	# the Player/native source transaction and no pause policy is overridden.
	death_count += 1
	death_shells = game.player.shells
	death_clock_s = (game.active_level.get("scheduler") as CinderThreatScheduler).get_clock()


func _new_shell() -> bool:
	game = Shell.new()
	var configured: bool = game.configure_runtime(raw, test_root + "campaign.json", test_root + "settings.json", test_root + "preferences.json")
	if not _expect(configured, "test-only metadata configures actual production Shell with isolated process-specific user paths"):
		game.free(); return false
	root.add_child(game)
	return true


func _actual_ready() -> bool:
	return is_instance_valid(game.active_level) and game.active_level.scene_file_path == LEVEL_PATH and game.active_level.level_id == "A1-L2" and game.active_level.hero == game.player and game.active_level.shared_shell == game and game.active_level.contract_error().is_empty() and game.campaign_error.is_empty()


func _presentation_ready() -> bool:
	if not _actual_ready(): return false
	for actor: Act1RushSelenite in (game.active_level.get("sources") as Dictionary).values():
		if not actor.dormant or actor.dead or actor.hp != 20.0 or actor.visible or actor.is_in_group("enemies") or not actor.art_binding_error().is_empty() or actor.get_cue().state().phase != "clear" or actor.get_art().call("pose_name") != "standing": return false
	for impact: CinderLaneMechanism in (game.active_level.get("circles") as Dictionary).values():
		var puff := impact.get_node_or_null("SquareImpactFragments") as MeshInstance3D
		if not is_instance_valid(puff) or puff.visible or impact.get_cue().state().phase != "clear" or impact.state().status == "running": return false
	return true


func _identities() -> Dictionary:
	var result: Dictionary = {"world": game.world.get_instance_id(), "native_world": game.world.get_world_3d().get_instance_id(), "hero": game.player.get_instance_id(), "level": game.active_level.get_instance_id(), "scheduler": (game.active_level.get("scheduler") as Node).get_instance_id()}
	for collection: String in ["sources", "circles"]:
		var actors: Dictionary = game.active_level.get(collection)
		for id: String in actors:
			result[collection + "/" + id] = actors[id].get_instance_id()
			result[collection + "/" + id + "/cue"] = actors[id].get_cue().get_instance_id()
	return result


func _all_new(before: Dictionary, after: Dictionary) -> bool:
	if before.size() != after.size(): return false
	for id: String in before:
		if not after.has(id) or before[id] == after[id]: return false
	return true


func _envelope() -> Dictionary:
	if not FileAccess.file_exists(test_root + "campaign.json"): return {}
	var value: Variant = JSON.parse_string(FileAccess.get_file_as_string(test_root + "campaign.json"))
	return value if value is Dictionary else {}


func _durable_matches() -> bool:
	var envelope: Dictionary = _envelope()
	var decoded: Dictionary = Exact.parse(String(envelope.get("payload_json", "")))
	return envelope.get("format_version") == 2 and decoded.get("accepted", false) and decoded.get("value") is Dictionary and Exact.stringify(decoded.value) == Exact.stringify(game.attempts.state())


func _diagnose(stage: String) -> void:
	if not is_instance_valid(game): return
	var diagnostic: Dictionary = {"stage": stage, "paused": paused, "death_count": death_count, "death_clock_s": death_clock_s, "death_shells": death_shells, "native_hits": hits.duplicate(true), "requested_progress": progress.duplicate(), "campaign_error": game.campaign_error, "menu_page": game.menu.page_name() if is_instance_valid(game.menu) else "", "attempts": game.attempts.state() if game.attempts != null else {}, "durable_envelope": _envelope()}
	if is_instance_valid(game.active_level) and game.active_level.scene_file_path == LEVEL_PATH:
		var level: CinderLevel = game.active_level
		var scheduler: CinderThreatScheduler = level.get("scheduler") as CinderThreatScheduler
		var paired: Dictionary = scheduler.snapshot_state(level.call("scheduler_bindings")) if paused else {}
		var actors: Dictionary = {}
		var circles: Dictionary = {}
		for id: String in level.get("sources"):
			var actor: Act1RushSelenite = (level.get("sources") as Dictionary)[id]
			actors[id] = {"state": actor.snapshot_state(paired) if paused and not paired.is_empty() else {}, "snapshot_error": actor.last_snapshot_error, "visible": actor.visible, "pose": actor.get_art().call("pose_name"), "art_error": actor.art_binding_error()}
		for id: String in level.get("circles"):
			var impact: CinderLaneMechanism = (level.get("circles") as Dictionary)[id]
			var puff := impact.get_node("SquareImpactFragments") as MeshInstance3D
			circles[id] = {"state": impact.snapshot_state(level.call("scheduler_bindings")) if paused else {}, "snapshot_error": impact.last_snapshot_error, "cue_phase": impact.get_cue().state().phase, "puff_visible": puff.visible, "puff_basis": [Codec.vector3(puff.basis.x), Codec.vector3(puff.basis.y), Codec.vector3(puff.basis.z)]}
		diagnostic["player"] = game.player.snapshot_state() if paused else {}
		diagnostic["player_error"] = game.player.last_snapshot_error
		diagnostic["scheduler"] = paired
		diagnostic["scheduler_error"] = scheduler.last_snapshot_error
		diagnostic["sources"] = actors
		diagnostic["circles"] = circles
		diagnostic["level"] = level.snapshot_state() if paused else {}
		diagnostic["level_error"] = level.last_snapshot_error
		diagnostic["level_capture_error"] = String(level.get("last_capture_error")) if level.scene_file_path == LEVEL_PATH else ""
	elif is_instance_valid(game.active_level):
		diagnostic["level"] = game.active_level.snapshot_state() if paused else {}
		diagnostic["player"] = game.player.snapshot_state() if paused else {}
	print("A1-L2 FATAL PRESENTATION DIAGNOSTIC ", Exact.stringify(diagnostic))


func _press(name: String) -> bool:
	var button := game.menu.find_child(name, true, false) as Button
	if not _expect(is_instance_valid(button) and not button.disabled and button.is_visible_in_tree(), "actual menu exposes enabled " + name): return false
	button.pressed.emit()
	return true


func _settle() -> void:
	for index: int in range(6):
		if finishing: return
		await process_frame


func _wait_for(predicate: Callable, attempts: int) -> bool:
	for index: int in range(attempts):
		if finishing: return false
		if predicate.call() == true: return true
		await create_timer(0.01, true).timeout
	return false


func _finish() -> void:
	if finishing: return
	finishing = true
	paused = true
	if is_instance_valid(game):
		if is_instance_valid(game.active_level):
			var native: CinderThreatScheduler = game.active_level.get("scheduler") as CinderThreatScheduler if game.active_level.scene_file_path == LEVEL_PATH else null
			game.active_level.exit_level()
			if is_instance_valid(native):
				_expect(not native.is_physics_processing() and native.reservations().is_empty() and get_nodes_in_group("enemies").is_empty(), "actual L2 exit retires native processing, leases and live target groups")
		if is_instance_valid(game.fx): game.fx.clear()
	await create_timer(0.5, true).timeout
	if is_instance_valid(game): game.free()
	paused = false
	await process_frame
	_expect(FileAccess.get_file_as_string(Registry.DATA_PATH) == registry_text, "test-only acceptance metadata leaves the canonical campaign registry unchanged")
	_cleanup_files()
	Engine.max_fps = old_fps
	AudioServer.set_bus_layout(old_audio)
	print("A1-L2 LANDING FATAL PRESENTATION: %d checks, %d failures (TEST predecessor/metadata; no compound or full-route acceptance)" % [checks, failures])
	quit(1 if failures else 0)


func _cleanup_files() -> void:
	var directory := DirAccess.open(test_root)
	if directory != null:
		for filename: String in directory.get_files():
			DirAccess.remove_absolute(ProjectSettings.globalize_path(test_root + filename))
	DirAccess.remove_absolute(ProjectSettings.globalize_path(test_root))


func _expect(ok: bool, message: String) -> bool:
	checks += 1
	if not ok:
		failures += 1
		push_error("FAIL: " + message)
	return ok
