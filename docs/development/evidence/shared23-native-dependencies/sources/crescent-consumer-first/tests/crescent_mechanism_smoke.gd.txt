	mechanism.state_changed.connect(func(_value: Dictionary) -> void: events.append("phase"))
	mechanism.get_cue().state_changed.connect(func(_value: Dictionary) -> void: events.append("cue"))
	mechanism.hit_resolved.connect(func(_id: String, _cycle: int, _result: Dictionary) -> void: events.append("hit"))

	var before: Dictionary = _pair(arena)
	mechanism.last_error = "prior-mechanism"
	mechanism.last_snapshot_error = "prior-mechanism-snapshot"
	scheduler.last_error = "prior-scheduler"
	scheduler.last_snapshot_error = "prior-scheduler-snapshot"
	var cue_before: Dictionary = mechanism.get_cue().state()
	var preview: Dictionary = mechanism.preview_start("hero", context, opening, bearing)
	_expect(preview.get("accepted", false), "actual nondefault continuous crescent bearing has pure ordinary-primary witness: " + String(preview.get("reason", "")))
	_expect(mechanism.last_error == "prior-mechanism" and mechanism.last_snapshot_error == "prior-mechanism-snapshot" and scheduler.last_error == "prior-scheduler" and scheduler.last_snapshot_error == "prior-scheduler-snapshot", "pure preview preserves both mechanism and Scheduler diagnostics")
	_expect(events.is_empty() and mechanism.state().cycle == 0 and mechanism.get_cue().state() == cue_before and _same(before, _pair(arena)), "pure preview keeps actor/source/clock/cycle/serial/cooldown/cue and emits no events")
	if not preview.get("accepted", false):
		await _dispose(arena)
		return
	_expect(preview.candidate.geometry == _shape(bearing) and preview.candidate.source_position == mechanism.global_position and preview.candidate.opening_position == opening and not preview.proof.uses_blast and not preview.proof.uses_invulnerability, "prospective candidate retains exact selected bearing, configured radii/sector/source and selected ordinary opening")

	edited_copy.proof.landing += Vector3.RIGHT

			"origin": bad.candidate.geometry.origin += Vector3.RIGHT

			"landing": bad.proof.landing += Vector3.RIGHT
			"array-direction": bad.candidate.geometry.direction = Codec.vector3(bearing)
			"unknown": bad.admission_authority = true
		var unchanged: Dictionary = _pair(arena)
		_expect(not mechanism.start("hero", context, opening, bearing, bad).accepted and _same(unchanged, _pair(arena)) and events.is_empty(), "tampered " + mutation + " preview rejects before cycle/serial/lease/cue allocation")
	var unchanged: Dictionary = _pair(arena)
	_expect(not mechanism.start("hero", context, opening, Vector3.RIGHT, preview).accepted and _same(unchanged, _pair(arena)), "changing selected bearing between pure preview and start rejects exact correspondence")
	for invalid: Variant in [Vector3.ZERO, Vector3(2, 0, 0), Vector3(1, 1, 0), Vector3(INF, 0, 0), [1.0, 0.0, 0.0], "right"]:

func _test_actual_inner_pocket_dash() -> void:
	var arena: Dictionary = await _arena("crescent", Vector3(-1.35, 0.02, 0))
	var mechanism: CinderLaneMechanism = arena.mechanism
	var hero: CinderPlayer = arena.hero
	var events: Array[Dictionary] = []
	mechanism.hit_resolved.connect(func(_id: String, _cycle: int, result: Dictionary) -> void: events.append(result))
	var preview: Dictionary = mechanism.preview_start("hero", _context(arena), null, Vector3.RIGHT)
	var answer: Dictionary = mechanism.start("hero", _context(arena), null, Vector3.RIGHT, preview) if preview.get("accepted", false) else preview
	_expect(answer.get("accepted", false), "hollow pocket fixture obtains actual scheduled crescent cycle: " + String(answer.get("reason", "")))
	if not answer.get("accepted", false):
		await _dispose(arena)
		return
	_expect(await _until_phase(mechanism, "active") and events.is_empty(), "native capsule inside hollow pocket remains safe through warning/lock into active")
	var hp: float = hero.hp
	_expect(hero.request_dash(Vector3.RIGHT), "actual shared public dash crosses hollow inner pocket during active crescent")
	await _ticks(18)
	var records: Array[Dictionary] = hero.get_world_action_records()
	_expect(records.size() == 1 and records[0].kind == "dash" and not records[0].collision_shortened, "actual native collision/floor path publishes completed full dash in static supported world")
	if records.size() == 1:
		var record: Dictionary = records[0]
		var path: Array[Dictionary] = _record_path(record)
		var radius: float = (hero.get_node("BodyCollision").shape as CapsuleShape3D).radius
		_expect(not Geometry.timed_path_hits(_shape(Vector3.RIGHT), path, float(record.started_at_s), float(record.completed_at_s), radius), "full measured actual capsule route remains inside genuinely safe hollow domain")
		var clear_floor: bool = true
		for point: Dictionary in record.path:
			clear_floor = clear_floor and arena.region.safe_rect.has_point(Vector2(point.position.x, point.position.z))
		_expect(clear_floor and hero.hp == hp and events.is_empty() and mechanism.state().hit_ids.is_empty(), "safe actual active path consumes no opportunity or HP; no decorative filled-disc damage proxy")
	await _dispose(arena)

		var before: Dictionary = _pair(arena)
		mechanism.last_error = "legacy-pure-diagnostic"
		_expect(not mechanism.preview_start("hero", _context(arena), null, Vector3.RIGHT).accepted and mechanism.last_error == "legacy-pure-diagnostic", kind + " explicit bearing rejects pure preflight without diagnostic changes")

		_expect(answer.get("accepted", false) and mechanism.state().geometry.kind == kind and mechanism.state().cycle == 1, kind + " legacy two-argument admission remains supported with unchanged geometry")
		await _dispose(arena)

func _shape(direction: Vector3) -> Dictionary:
	return Geometry.crescent(Vector3.ZERO, direction, 2.4, 3.0, 0.5)


func _context(arena: Dictionary) -> Dictionary:
	return {"encounter_id": "crescent-unit", "world_revision": 1, "recognition_s": 0.12, "attack_input_margin_s": 0.02, "escape_directions": [Vector3.LEFT, Vector3.RIGHT, Vector3.FORWARD, Vector3.BACK], "return_directions": [Vector3.LEFT, Vector3.RIGHT, Vector3.FORWARD, Vector3.BACK], "floor_regions": [arena.region]}


func _bindings(arena: Dictionary) -> Dictionary:
	return {"world_root": arena.root, "owners": {"crescent-mechanism": arena.mechanism}, "floors": {"main-floor": arena.region}}


func _pair(arena: Dictionary) -> Dictionary:
	var prior: bool = paused
	paused = true
	var bindings: Dictionary = _bindings(arena)
	var local: Dictionary = arena.mechanism.snapshot_state(bindings)
	var scheduler: Dictionary = arena.scheduler.snapshot_state(bindings)
	var hero: Dictionary = arena.hero.snapshot_state()
	_expect(not local.is_empty() and not scheduler.is_empty() and not hero.is_empty(), "exact comparison uses nonempty native mechanism/Scheduler/Hero envelopes")
	paused = prior
	return {"hero": hero, "scheduler": scheduler, "mechanism": local}


func _exact_pair(value: Dictionary) -> Dictionary:
	var encoded: String = Exact.stringify(value)
	var parsed: Dictionary = Exact.parse(encoded)
	_expect(not encoded.is_empty() and parsed.get("accepted", false) and Exact.stringify(parsed.get("value", {})) == encoded, "whole paused aggregate survives exact tagged JSON bits/types without decimal clock tolerance")
	return parsed.get("value", {})


func _reject_unchanged(arena: Dictionary, bad: Dictionary, bindings: Dictionary) -> bool:
	var before: Dictionary = _pair(arena)
	var cue: Dictionary = arena.mechanism.get_cue().state()
	return not arena.mechanism.restore_state(bad, bindings) and _same(before, _pair(arena)) and _native_same(cue, arena.mechanism.get_cue().state())


func _same(left: Dictionary, right: Dictionary) -> bool:
	var encoded: String = Exact.stringify(left)
	return not encoded.is_empty() and encoded == Exact.stringify(right)


func _native_same(left: Variant, right: Variant) -> bool:
	if typeof(left) != typeof(right): return false
	if left is Dictionary:
		if left.size() != right.size(): return false
		for key: String in left:
			if not right.has(key) or not _native_same(left[key], right[key]): return false
		return true
	if left is Array:
		if left.size() != right.size(): return false
		for index: int in range(left.size()):
			if not _native_same(left[index], right[index]): return false
		return true
	return left == right


func _record_path(record: Dictionary) -> Array[Dictionary]:
	var path: Array[Dictionary] = []
	for index: int in range(1, record.path.size()):
		path.append({"from": record.path[index - 1].position, "to": record.path[index].position, "start_s": record.path[index - 1].time_s, "end_s": record.path[index].time_s})
	return path


func _next_float(value: float) -> float:
	var bytes := PackedByteArray()
	bytes.resize(8)
	bytes.encode_double(0, value)
	bytes.encode_u64(0, bytes.decode_u64(0) + 1)
	return bytes.decode_double(0)


func _until_phase(mechanism: CinderLaneMechanism, phase: String) -> bool:
	for _index: int in range(450):
		if mechanism.state().phase == phase: return true
		await _ticks(1)
	return false


func _ticks(count: int) -> void:
	for _index: int in range(count):
		await physics_frame
		await process_frame


func _dispose(arena: Dictionary) -> void:
	arena.root.queue_free()
	await process_frame


func _expect(condition: bool, description: String) -> void:
	_checks += 1
	if not condition:
		_failures += 1
		push_error(description)
