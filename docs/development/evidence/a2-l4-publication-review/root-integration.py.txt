def require(value, message):
    if not value:
        raise RuntimeError(message)


def record(path):
    require(path.resolve() == path and not path.is_symlink(), 'Literal nonsymlink required: ' + str(path))
    if not path.exists():
        return {'exists': False, 'regular': False, 'symlink': False, 'sha256': None, 'bytes': None, 'mode': None}
    before = path.stat()
    require(stat.S_ISREG(before.st_mode), 'Regular file required')
    with path.open('rb') as handle:
        opened = os.fstat(handle.fileno())
        data = handle.read()
        after_read = os.fstat(handle.fileno())
    after = path.stat()
    key = lambda s: (s.st_dev, s.st_ino, s.st_mode, s.st_size, s.st_mtime_ns)
    require(key(before) == key(opened) == key(after_read) == key(after), 'Read changed inode/metadata')
    return {'exists': True, 'regular': True, 'symlink': False, 'sha256': hashlib.sha256(data).hexdigest(),
            'bytes': len(data), 'mode': format(stat.S_IMODE(before.st_mode), '04o')}


def write_new(path, data, mode=0o644):
    path.parent.mkdir(parents=True, exist_ok=True)
    require(path.parent.resolve() == path.parent, 'Symlink parent refused')
    with path.open('xb') as handle:
        os.fchmod(handle.fileno(), mode)
        handle.write(data)
        handle.flush()
        os.fsync(handle.fileno())
