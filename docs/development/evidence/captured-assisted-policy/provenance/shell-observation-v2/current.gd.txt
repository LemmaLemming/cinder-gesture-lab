func _distinct_ids(old: Dictionary, fresh_shell: bool = false, require_playback: bool = false) -> bool:
	var current: Dictionary = _ids()
	var required: Array[String] = ["shell", "hero", "parent", "phase", "scheduler", "gate", "floor"]
	if require_playback: required.append("playback")
	for key: String in required:
		if not old.has(key) or not current.has(key) or old[key] == "0" or current[key] == "0": return false

	return true
