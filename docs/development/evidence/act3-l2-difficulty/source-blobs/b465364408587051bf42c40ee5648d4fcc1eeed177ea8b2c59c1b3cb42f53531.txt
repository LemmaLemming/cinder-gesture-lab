func _exercise() -> String:
	for _episode: int in range(12):
		var warning: Dictionary = await _next_warning()
		if warning.is_empty(): return "no genuine fresh warning in bounded unchanged interval; " + _diagnostic()
		_source_id = warning.id
		_current = warning.record.duplicate(true)
		var reason: String = _proof_error(warning)
		if not reason.is_empty(): return reason
		for segment: Dictionary in warning.proof.path:
			if segment.kind in ["escape_dash", "positioning_dash"]:
				reason = await _at_time(float(segment.start_s))
				if reason.is_empty(): reason = await _dash(segment)
			elif segment.kind == "ordinary_primary":
				var acted_id: String = _source_id
				reason = await _primary(segment, warning.proof)
				if reason.is_empty() and _targets.has(acted_id): _target_hits[acted_id] = int(_target_hits.get(acted_id, 0)) + 1
			else: reason = await _hold(segment)
			if not reason.is_empty(): break
		if reason == Replan: continue
		if not reason.is_empty(): return reason
		var all_hit: bool = true
		for id: String in _targets: all_hit = all_hit and _target_hits.has(id)
		if all_hit: return ""
	return "changed-court target opportunity not completed within12 real proof/replan episodes"

func _tick_safe() -> String:
	if paused and is_instance_valid(_game) and is_instance_valid(_hero) and not _hero.dead:
		var clock: float = _scheduler.get_clock()
		var hero_clock: float = _hero.get_world_action_clock()
		var prior_input: Dictionary = _game.call("get_input_observation_state")
		_game.call("resume_lab") # Explicit public resume after observed application pause; no clock/deadline adjustment.
		if paused or _scheduler.get_clock() != clock or _hero.get_world_action_clock() != hero_clock or _game.call("get_input_observation_state") != prior_input: return "public resume changed saved clocks/input or failed to resume"
		_public_resumes.append({"clock_s": clock, "hero_clock_s": hero_clock, "observed_background_notifications": _game.get("background_pause_events").duplicate()})
	var reason: String = await super._tick_safe()
	if not reason.is_empty(): return reason
	if _targets.size() == 2 and _level.call("_source_running", _targets[0]) and _level.call("_source_running", _targets[1]): _court_union = true
	if not portrait_dir.is_empty() and not _targets.is_empty():
		for id: String in _targets:
			var actual: Dictionary = _sources[id].call("state")
			if actual.phase in ["warning", "lock", "active", "recovery"]:
				var key: String = selected_court + "-" + id + "-" + String(actual.phase)
				if not captures.has(key):
					reason = await _portrait(key)
					if not reason.is_empty(): return reason
	return ""

func _portrait(key: String) -> String:
	if DisplayServer.get_name() == "headless": return "actual renderer required for requested brief portrait"
	var old_paused: bool = paused
	paused = true # Disclosed complete native boundary freeze for the raster only.
	var before: Dictionary = {"player": _hero.snapshot_state(), "level": _level.snapshot_state()}
	if before.player.is_empty() or before.level.is_empty(): return "brief portrait requires actual complete native pair"
	await process_frame
	await RenderingServer.frame_post_draw
	var image: Image = root.get_texture().get_image()
	var path: String = portrait_dir.path_join(key + ".png")
	var code: int = image.save_png(path)
	var unchanged: bool = ExactJson.stringify(before) == ExactJson.stringify({"player": _hero.snapshot_state(), "level": _level.snapshot_state()})
	var note: Dictionary = {"png": path, "sha256": FileAccess.get_sha256(path), "dimensions": [image.get_width(), image.get_height()], "native_pair_exact": ExactJson.stringify(before), "unchanged_native_pair": unchanged}
	var f: FileAccess = FileAccess.open(path + ".json", FileAccess.WRITE)
	if f != null: f.store_string(JSON.stringify(note, "  ") + "\n"); f.close()
	captures[key] = note
	paused = old_paused
	return "" if code == OK and unchanged else "actual portrait write or paused native pair stability failed"

func _sources_observation() -> Dictionary:
	var result: Dictionary = {}
	for name: String in _source_names: result[name] = FileAccess.get_sha256("res://" + name) if FileAccess.file_exists("res://" + name) else null
	return result

func _expect(ok: bool, label: String, detail: String = "") -> bool:
	_checks += 1
	var row: Dictionary = {"n": _checks, "ok": ok, "label": label, "detail": detail}
	rows.append(row)
	if not ok:
		_failures += 1
		if first_failure.is_empty(): first_failure = row.duplicate(true)
		push_error("FAIL: " + label + "; " + detail)
	else: print("PASS: " + label)
	return ok

func _end() -> void:
	if _finishing: return
	_finishing = true
	await _dispose()
	_finish()

func _finish() -> void:
	if not _finishing:
		_end.call_deferred(); return
	var report: Dictionary = {"checks": _checks, "failures": _failures, "rows": rows, "first_failure": first_failure, "complete": complete, "success": complete and _failures == 0, "setup": _setup, "explicit_public_resumes": _public_resumes, "profile": selected_profile, "court": selected_court, "kit": _kit, "receipts": receipts, "captures": captures, "source_before": source_before, "source_after": _sources_observation(), "scope": "Changed court ordinary actions/native placement only; disclosed stable Hero and initial zero ammo. Earlier sources alive/authoritative. No route/allclear/exit/ShellCampaignAttempts/legacy/FPS/balance claim."}
	if not output.is_empty():
		var f: FileAccess = FileAccess.open(output, FileAccess.WRITE)
		if f != null: f.store_string(JSON.stringify(report, "  ") + "\n"); f.close()
	print("FOREST_CHANGED_COURT ", _checks, "/", _failures, " complete=", complete, " report=", output, " sha256=", FileAccess.get_sha256(output))
	quit(0 if complete and _failures == 0 else 1)
