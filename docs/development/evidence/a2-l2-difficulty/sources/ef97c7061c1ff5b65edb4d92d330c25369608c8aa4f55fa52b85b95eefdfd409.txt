"""Frozen named old-Weybridge producer; one owned child through canonical queue."""
from pathlib import Path
import contextlib
import datetime
import hashlib
import importlib.util
import json
import os
import re
import sys
import uuid

HERE = Path(__file__).resolve().parent
ROOT = HERE.parents[1]
SCRIPT = str((HERE / "producer.gd").relative_to(ROOT))
SHIM = str((HERE / "profile_main.gd").relative_to(ROOT))
RUNNER = str(Path(__file__).resolve().relative_to(ROOT))
FILES = (
    SCRIPT, SHIM, RUNNER, "scripts/dev/dev.py", "project.godot",
    "scenes/main.tscn", "scenes/player.tscn", "scenes/acts/act2/a2_l2.tscn",
    "scripts/game.gd", "scripts/player.gd", "scripts/equipment.gd",
    "scripts/effects.gd", "scripts/hud.gd", "scripts/campaign/level.gd",
    "scripts/acts/act2/horsell_common.gd", "scripts/acts/act2/horsell_sequence.gd",
    "scripts/acts/act2/weybridge.gd", "scripts/acts/act2/weybridge_sequence.gd",
    "scripts/acts/act2/weybridge_kit.gd", "scripts/acts/act2/heath_kit.gd",
    "scripts/acts/act2/weybridge_scout_actor.gd",
    "scripts/acts/act2/ray_scout_actor.gd", "scripts/acts/act2/ray_scout_visual.gd",
    "scripts/acts/act2/ray_scout_exchange.gd",
    "scripts/acts/act2/salvage_handler_actor.gd",
    "scripts/acts/act2/salvage_handler_visual.gd",
    "scripts/acts/act2/giant_foot_visual.gd",
    "scripts/combat/threat_scheduler.gd", "scripts/combat/lane_mechanism.gd",
    "scripts/combat/threat_geometry.gd", "scripts/combat/difficulty.gd",
    "scripts/combat/body_sweep.gd", "scripts/cues/threat_cue.gd",
    "scripts/cues/interaction_cue.gd", "scripts/campaign/snapshot_codec.gd",
    "scripts/campaign/exact_json.gd", "scripts/campaign/save_store.gd",
    "data/design/player_equipment.json", "data/design/difficulty_profiles.json",
)
NAMES = tuple(sorted(set(FILES) | {p + ".uid" for p in FILES if p.endswith(".gd")}))
ORIGINAL_ROOT = "171680483795c75272d0d7e8b64a85c348d2b4b552acb92a2d3638febe694a2b"
ORIGINAL_SEQUENCE = "d28de78224f6985c377a13f9a4d9874ac2766aeb9bc744efe9640ce05f8f7dfc"
TIMEOUT = 180


def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def snapshot():
    return {p: sha(ROOT / p) if (ROOT / p).is_file() else None for p in NAMES}


def exact_sources(seal, observed):
    if seal != observed:
        raise ValueError("Fixed source/UID drift: " + repr([p for p in NAMES if seal.get(p) != observed.get(p)]))


class Tee:
    def __init__(self, original, log):
        self.original, self.log = original, log

    def write(self, value):
        self.log.write(value)
        return self.original.write(value)

    def flush(self):
        self.log.flush()
        self.original.flush()


def main():
    if len(sys.argv) != 1:
        raise ValueError("This frozen producer accepts no user arguments")
    stamp = datetime.datetime.now(datetime.timezone.utc).strftime("%Y%m%dT%H%M%S.%fZ")
    out = HERE / ("native-" + stamp + "-" + uuid.uuid4().hex[:8])
    out.mkdir(exist_ok=False)

    def save(name, value):
        (out / name).write_text(json.dumps(value, indent=2, allow_nan=False) + "\n")

    result = {"kind": "new-executed-original-weybridge-15-key-producer", "started_at": now(), "output": str(out), "closed": False, "engine_started": False, "owned_child_terminal": False, "passed": False, "scope": "Exact original Weybridge root/sequence at the fixed named current mixed shared baseline. New native caller pairs, not historical payloads, CampaignAttempts or route evidence."}
    queue = {"queued": False, "acquired": False, "timeout_seconds_after_acquisition": TIMEOUT}
    save("result.json", result)
    save("queue.json", queue)
    code = 2
    print("Weybridge original producer output:", out, flush=True)
    with (out / "raw.log").open("w") as log, contextlib.redirect_stdout(Tee(sys.stdout, log)), contextlib.redirect_stderr(Tee(sys.stderr, log)):
        try:
            seal_record = json.loads((HERE / "sealed-inputs.json").read_text())
            seal = seal_record["names"]
            if set(seal) != set(NAMES) or str(ROOT) != seal_record["root"]:
                raise ValueError("The exact frozen named seal is required")
            if any(seal[p] is None for p in FILES) or {p for p, h in seal.items() if h is None} != {SCRIPT + ".uid", SHIM + ".uid"}:
                raise ValueError("Only both explicitly absent private caller UID sidecars are allowed")
            if seal["scripts/acts/act2/weybridge.gd"] != ORIGINAL_ROOT or seal["scripts/acts/act2/weybridge_sequence.gd"] != ORIGINAL_SEQUENCE:
                raise ValueError("Refuse migrated or altered original Weybridge root/sequence")
            save("sealed-inputs.json", seal_record)
            before = snapshot()
            save("sources-before-queue.json", before)
            exact_sources(seal, before)
            # Retain only these explicitly named bytes before entering the queue.
            # Hash joins at lock/after child qualify the same bytes; no census.
            blobs = out / "source-blobs"
            blobs.mkdir()
            retained = {}
            for p, h in seal.items():
                if h is None:
                    continue
                b = (ROOT / p).read_bytes()
                if hashlib.sha256(b).hexdigest() != h:
                    raise ValueError("Source changed during named byte capture: " + p)
                target = blobs / (h + ".txt")
                if not target.exists():
                    target.write_bytes(b)
                retained[p] = {"sha256": h, "blob": str(target.relative_to(out)), "origin": "captured_before_queue; exact at-lock/after-child joins required"}
            save("source-retention.json", {"scope": "Explicit names only; no complete resource/global UID/dependency closure.", "files": retained})
            spec = importlib.util.spec_from_file_location("cinder_a2_l2_original_producer_dev", ROOT / "scripts/dev/dev.py")
            dev = importlib.util.module_from_spec(spec)
            sys.modules[spec.name] = dev
            policy = sys.dont_write_bytecode
            try:
                sys.dont_write_bytecode = True
                spec.loader.exec_module(dev)
            finally:
                sys.dont_write_bytecode = policy
            env = dict(os.environ)
            env["CINDER_A2_L2_LEGACY_OUTPUT"] = str(out)
            env["CINDER_A2_L2_LEGACY_SOURCES"] = json.dumps(["res://" + p for p in NAMES])
            context = dev.ProjectContext(root=ROOT, environ=env)
            command = [str(dev.resolve_godot(context)), "--path", str(ROOT), "--headless", "--log-file", str(out / "engine.log"), "--script", "res://" + SCRIPT]
            save("command.json", {"argv": command, "cwd": str(ROOT), "timeout_seconds_after_acquisition": TIMEOUT, "environment_scope": "Only unique output and fixed source-name list added to the owned child; inherited machine configuration is not changed or dumped."})
            queue.update({"queued": True, "queued_at": now(), "lock_path": str(dev.shared_lock_path(ROOT))})
            save("queue.json", queue)
            with dev.godot_lock(context.root) as lock_fd:
                queue.update({"acquired": True, "acquired_at": now()})
                save("queue.json", queue)
                at_lock = snapshot()
                save("sources-at-lock.json", at_lock)
                exact_sources(seal, at_lock)
                result.update({"engine_started": True, "engine_started_at": now()})
                save("result.json", result)
                returned = dev.run_validation(command, context, lock_fd, TIMEOUT, detect_godot_errors=True)
                result.update({"owned_child_terminal": True, "owned_child_terminal_at": now(), "engine_return_code": returned, "return_code_scope": "Canonical dev return: zero confirms its owned direct-child wait returned0. Nonzero may be a synthetic script/resource/timeout failure, not an independently exposed OS exit; no process-group claim."})
                save("result.json", result)
                after = snapshot()
                save("sources-after-child.json", after)
                exact_sources(seal, after)
            queue["lock_context_released_at"] = now()
            save("queue.json", queue)
            log.flush()
            if returned != 0:
                raise ValueError("Canonical owned validation returned " + str(returned))
            report_path = out / "native-report.json"
            report = json.loads(report_path.read_text())
            rows = report["rows"]
            assert report["checks"] == len(rows) > 0
            assert [r["check"] for r in rows] == list(range(1, len(rows) + 1))
            assert all(type(r["ok"]) is bool for r in rows)
            assert report["failures"] == sum(r["ok"] is False for r in rows) == 0
            assert report["first_failure"] == {} and report["complete"] is True and report["success"] is True
            assert report["source_before"] == report["source_after"] == {"res://" + p: h for p, h in seal.items()}
            packets = report["packets"]
            assert len(packets) == 4 and {(r["profile"], r["stage"]) for r in packets} == {(p, s) for p in ["standard", "challenge"] for s in ["entrance", "ray-warning"]}
            for packet in packets:
                ef, sf = out / packet["exact_file"], out / packet["store_file"]
                assert ef.parent == sf.parent == out
                assert sha(ef) == packet["exact_sha256"] and ef.stat().st_size == packet["exact_bytes"]
                assert sha(sf) == packet["store_sha256"] and sf.stat().st_size == packet["store_bytes"]
                envelope, wire = json.loads(sf.read_text()), ef.read_text()
                assert envelope["format_version"] == 2 and envelope["payload_json"] == wire and envelope["sha256"] == hashlib.sha256(wire.encode()).hexdigest()
                assert packet["original_local_key_count"] == len(packet["original_local_keys"]) == 15
                assert packet["outer_local_snapshot_version"] == packet["sequence_schema"] == 1
                assert packet["actual_scheduler_profile"] == packet["profile"]
            raw = (out / "raw.log").read_text()
            assert re.findall(r"(?m)^CINDER_A2_L2_LEGACY_REPORT path=(.+) sha256=([0-9a-f]{64})$", raw) == [(str(report_path), sha(report_path))]
            assert re.findall(r"(?m)^A2-L2 original15-key producer: (\d+) checks/(\d+) failures; complete=(true|false); packets=(\d+)$", raw) == [(str(report["checks"]), "0", "true", "4")]
            errors = re.findall(r"(?m)^(?:ERROR:|SCRIPT ERROR:).*", raw)
            result.update({"passed": True, "native_report": {"path": str(report_path), "sha256": sha(report_path), "checks": len(rows), "failures": 0, "complete": True, "packets": packets, "rows_footer_source_and_store_joins": True}, "raw_error_lines": errors, "raw_log_error_free": not errors})
            code = 0
        except KeyboardInterrupt:
            result.update({"interrupted": True, "error": "Interrupted; canonical dev owns only its direct-child cleanup."})
            code = 130
        except Exception as exc:
            result["error"] = repr(exc)
            print("Original Weybridge producer failed:", repr(exc), flush=True)
        finally:
            result.update({"closed": True, "closed_at": now(), "runner_return_code": code})
            save("result.json", result)
            save("sources-at-runner-close.json", snapshot())
            print("Weybridge original producer runner CLOSED code=" + str(code), flush=True)
    save("evidence-index.json", {"created_after_raw_log_close": True, "files": {str(p.relative_to(out)): {"sha256": sha(p), "bytes": p.stat().st_size} for p in sorted(out.rglob("*")) if p.is_file()}, "scope": result["scope"]})
    return code


if __name__ == "__main__":
    raise SystemExit(main())
