func _new_shell(expected: Dictionary) -> bool:
	receiver_events.clear(); watched.clear()
	production = Shell.new()
	if not _expect(production.configure_runtime({}, isolated + "campaign.json", isolated + "settings.json", isolated + "preferences.json"), "actual canonical Shell uses no substitute readiness registry", production.campaign_error): return false
	root.add_child(production)
	for _frame: int in range(FRAME_LIMIT):
		await process_frame
		if not _valid_shell(): return false
		if paused and not production.is_pause_requested() and production.menu.page_name() == "title" and production.player == null and production.active_level == null and _same(production.attempts.state(), expected):
			return _expect(true, "exact entire expected canonical startup session settles at Title")
	return _expect(false, "bounded canonical exact Title startup session", production.campaign_error)

func _completed_replay_boundary(canonical: CinderCampaignRegistry) -> bool:
	var path: String = ""
	var digest: String = ""
	for argument: String in OS.get_cmdline_user_args():
		if argument.begins_with("--completed-session="): path = argument.trim_prefix("--completed-session=")
		if argument.begins_with("--completed-sha256="): digest = argument.trim_prefix("--completed-sha256=")
	if not _expect(not path.is_empty() and digest.length() == 64 and FileAccess.get_sha256(path) == digest, "actual owner's completed-return session is explicitly hash pinned"): return false
	var decoded: Dictionary = Exact.parse(FileAccess.get_file_as_string(path))
	if not _expect(decoded.get("accepted", false) and decoded.get("value") is Dictionary, "complete finite owner completed-return session parses exactly"): return false
	var submitted: Dictionary = decoded.value
	var model: CinderCampaignAttempts = Attempts.new(canonical)
	if not _expect(model.state_error(submitted).is_empty() and submitted.side_attempt == null and submitted.completed_main == expected_session.completed_main and submitted.completed_optional == ["A3-O1"] and submitted.reward_ids == ["A3-O1-completion-stamp"] and submitted.story.level_id == STORY_ID and submitted.unlocked_equipment == expected_session.unlocked_equipment, "genuine owner return carries one stamp and unchanged main/equipment boundary"): return false
	if not await _dispose_shell(): return false
	var writer: CinderSaveStore = Store.new(isolated + "campaign.json")
	writer.payload_validator = model.saved_payload_error
	if not _expect(writer.write_payload(submitted), "actual owner complete session enters isolated exact Store", writer.last_error): return false
	protected = submitted.story.snapshot.duplicate(true)
	if not await _new_shell(submitted): return false
	production.menu.continue_story_requested.emit()
	if not await _wait_installed(STORY_ID, "resume", protected) or not _installed(protected): return false
	production.menu.replay_requested.emit("A3-O1", ReplayEquipment.STARTER)
	if not await _wait_installed("A3-O1", "resume"): return false
	var replay: Dictionary = production.capture_campaign_snapshot()
	if not _installed(replay): return false
	var state: Dictionary = production.attempts.state()
	var local: Dictionary = production.active_level.call("state")
	if not _expect(state.side_attempt.kind == "replay" and local.collections.is_empty() and not production.active_level.is_completed() and state.completed_optional == submitted.completed_optional and state.reward_ids == submitted.reward_ids and _same(state.story, submitted.story), "fresh canonical optional Replay preserves exactly one genuine existing stamp"): return false
	var refs: Array[WeakRef] = []
	_collect_refs(_world_container(), refs)
	var before: Dictionary = _core_ids()
	production.menu.leave_side_requested.emit()
	if not await _wait_installed(STORY_ID, "journey", protected, before) or not _installed(protected) or not await _released(refs, "canonical Replay Leave frees optional native world"): return false
	state = production.attempts.state()
	return _expect(state.side_attempt == null and state.completed_main == submitted.completed_main and state.completed_optional == submitted.completed_optional and state.reward_ids == submitted.reward_ids and state.unlocked_equipment == submitted.unlocked_equipment and _same(state.story, submitted.story), "canonical replay return retains exact one stamp and protected main/equipment/Story boundaries") and _disk(state)

		if FileAccess.file_exists(path): DirAccess.remove_absolute(ProjectSettings.globalize_path(path))
	DirAccess.remove_absolute(ProjectSettings.globalize_path(isolated))
	AudioServer.set_bus_layout(old_audio); Engine.max_fps = old_fps

TEST prefix11; no inherited native held/marker/completion execution, earned route, GUI, portrait or broad balance credit." % [checks, failures, str(_registration_done)])
