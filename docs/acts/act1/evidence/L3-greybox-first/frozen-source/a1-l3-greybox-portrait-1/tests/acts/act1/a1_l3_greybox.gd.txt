func _initialize() -> void:
	started_ms = Time.get_ticks_msec()
	process_frame.connect(_watchdog)
	_run.call_deferred()


func _run() -> void:
	portrait = "--portrait" in OS.get_cmdline_user_args()
	root.size = PortraitSize
	if portrait:
		if not _require(DisplayServer.get_name() != "headless", "portrait mode requires the actual graphical renderer"):
			await _finish(); return
		capture_dir = "res://.cinder/captures/l3-components-%d" % Time.get_ticks_usec()
		if not _require(DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(capture_dir)) == OK, "portrait evidence directory is writable"):
			await _finish(); return
		print("SCRIPTED L3 COMPONENT PORTRAIT: native_focus_observed=", DisplayServer.window_is_focused(), "; one initial public resume per world; normal FOCUS_OUT unchanged; ", capture_dir)
	if not await _open_world(GrovePath, "grove", Vector3(0, 0.1, 15)):
		await _finish(); return
	if not await _swipe(Vector3.FORWARD, "genuine grove admission approach"):
		await _finish(); return
	var first_id: String = await _first_warning()
	if first_id.is_empty() or not await _fight_cycle(first_id, true):
		await _finish(); return
	var first: CharacterBody3D = sources[first_id]
	_require(first.get("dead") and first.get("hp") == 0.0 and not first.is_in_group("enemies"), "real primary defeats one retained C31 without a scripted damage call")
	_require(not level.get("greybox_clear") and not level.is_completed(), "two living swarmers remain; this component proof claims no full room/level clear")
	if not await _close_world(false):
		await _finish(); return
	if not await _open_world(GuardPath, "guard", Vector3(0, 0.1, -26)):
		await _finish(); return
	if not await _wait(func() -> bool: return _running_warning("lone-guard"), "isolated guard publishes a real warning"):
		await _finish(); return
	var guard: CharacterBody3D = sources["lone-guard"]
	var first_guard_cycle: int = int(guard.call("state").cycle)
	var old_lease: String = String(guard.call("state").reservation_id)
	if not await _fight_cycle("lone-guard", false):
		await _finish(); return
	var hurt: Dictionary = guard.call("state")
	if not _require(not guard.get("dead") and float(guard.get("hp")) > 0.0 and float(guard.get("hp")) < 24.0 and hurt.last_cancel_reason == "actual_player_damage" and hurt.reservation_id.is_empty() and scheduler.reservation_state(old_lease).is_empty(), "ordinary primary interrupts a surviving C32 and cancels its original native lease"):
		await _finish(); return
	if not await _capture("guard-surviving-hit"):
		await _finish(); return
	if not await _wait(func() -> bool: return _running_warning("lone-guard") and int(guard.call("state").cycle) > first_guard_cycle, "surviving guard naturally settles and publishes a fresh complete warning"):
		await _finish(); return
	if not await _fight_cycle("lone-guard", false):
		await _finish(); return
	if not await _wait(func() -> bool: return guard.get("dead") and level.get("greybox_clear"), "actual parent observes the real isolated guard defeat"):
		await _finish(); return
	_require(guard.get("dead") and guard.get("hp") == 0.0 and not guard.is_in_group("enemies") and level.get("greybox_clear") and not level.is_completed(), "second genuine ordinary primary clears the isolated guard without campaign completion")
	if not await _capture("guard-defeated") or not await _close_world(false):
		await _finish(); return
	# A distinct entered world exercises direct deferred tree removal with a
	# running lease. It deliberately does not call exit_level() before removal.
	if not await _open_world(GrovePath, "direct-queue-free", Vector3(0, 0.1, 15)) or not await _swipe(Vector3.FORWARD, "direct teardown genuine admission approach"):
		await _finish(); return
	if (await _first_warning()).is_empty() or not await _close_world(true):
		await _finish(); return
	await _finish()


func _open_world(path: String, scope: String, spawn: Vector3) -> bool:
	# A genuine coroutine, waiting before construction preserves exact spawn.
	await process_frame
	current_scope = scope
	sources.clear(); admissions.clear(); phase_observations.clear()
	world_records.clear(); input_observations.clear(); hit_events.clear()
	events = 0; max_preparing = 0; preparing_violation = false
	game = MainScene.instantiate()
	game.set("level_scene_path", path)
	root.add_child(game)
	level = game.get("active_level") as CinderLevel
	hero = game.get("player") as CinderPlayer
	scheduler = level.get("scheduler") as CinderThreatScheduler if level != null else null
	if not _require(level != null and hero != null and scheduler != null and level.contract_error().is_empty() and level.hero == hero and hero.global_position == spawn, scope + " enters through the one shared Hero at its genuine authored spawn"):
		return false
	sources = (level.get("sources") as Dictionary).duplicate()
	initial_hp = hero.hp
	var current_ids: Array = level.call("current_source_ids")
	if not _require(sources.size() == 4 and current_ids.size() == (1 if scope == "guard" else 3), scope + " retains exactly three C31s and one C32 with only its actual room activated"):
		return false
	for id: String in sources:
		var actor: CharacterBody3D = sources[id]
		var state: Dictionary = actor.call("state")
		var active: bool = id in current_ids
		if not _require(state.source_id == id and state.role_id == ("A1-E3" if id == "lone-guard" else "A1-E2") and state.dormant == (not active) and actor.is_in_group("enemies") == active and state.hp == (24.0 if id == "lone-guard" else 16.0) and not state.dead, scope + " stable source " + id + " starts alive with correct native role, dormancy and HP"):
			return false
		actor.connect("state_changed", Callable(self, "_notice_phase").bind(id))
		actor.connect("hit_resolved", Callable(self, "_notice_hit").bind(id))
		(actor.call("get_cue") as CinderThreatCue).state_changed.connect(func(_state: Dictionary) -> void: events += 1)
	level.connect("native_admission_published", _notice_admission)
	hero.world_action_executed.connect(func(record: Dictionary) -> void: world_records.append(record.duplicate(true)); events += 1)
	game.connect("input_observed", func(observation: Dictionary) -> void: input_observations.append(observation.duplicate(true)))
	game.call("resume_lab") # Initial public resume only; never repeat on focus loss.
	return _require(not paused, scope + " initial public preview resume leaves ordinary focus handling intact")


func _first_warning() -> String:
	if not await _wait(func() -> bool:
		for id: String in level.call("current_source_ids"):
			if _running_warning(id): return true
		return false, "real room admits a native warning within the component window"):
		return ""
	for id: String in level.call("current_source_ids"):
		if _running_warning(id): return id
	return ""


func _running_warning(id: String) -> bool:
	var state: Dictionary = sources[id].call("state")
	return state.status == "running" and state.phase == "warning" and admissions.has(String(state.reservation_id))


func _fight_cycle(id: String, pause_warning: bool) -> bool:
	var actor: CharacterBody3D = sources[id]
	var state: Dictionary = actor.call("state")
	var lease: String = String(state.reservation_id)
	var cycle: int = int(state.cycle)
	var answer: Dictionary = admissions.get(lease, {})
	if not _require(answer.get("accepted", false) and answer.get("armed", false) and answer.reservation.source_instance_id == actor.get_instance_id() and state.source_id == id and not answer.proof.uses_blast and not answer.proof.uses_invulnerability, id + " native accepted proof belongs to this actual retained source and needs neither blast nor invulnerability"):
		return false
	var proof: Dictionary = answer.proof.duplicate(true)
	var reservation: Dictionary = answer.reservation.duplicate(true)
	if not _require(float(proof.primary_time_s) >= float(reservation.active_until_s) and float(proof.response_complete_s) < float(reservation.recovery_until_s), id + " ordinary primary and full response fit the actual stopped-source recovery"):
		return false
	var length: float = (reservation.geometry["to"] as Vector3).distance_to(reservation.geometry["from"])
	if not _require(reservation.geometry.kind == "lane" and absf(length - (2.0 if id == "lone-guard" else 1.0)) <= PositionTolerance and absf(float(reservation.geometry.radius) - 0.38) <= 0.00001, id + " native footprint is the authored short grounded lane"):
		return false
	if not _framing(id + " warning"):
		return false
	if pause_warning and not await _pause_component(id, lease): return false
	if not await _capture(id + "-warning"): return false
	var source_start: Vector3 = actor.global_position
	var escape: Vector3 = proof.landing - hero.global_position
	if not _require(absf(Vector2(escape.x, escape.z).length() - float(hero.equipment.resolved_stats().dash_distance)) <= PositionTolerance, id + " published safe landing is one full legal dash"):
		return false
	if not await _swipe(escape, id + " native safe escape") or not _framing(id + " safe landing"):
		return false
	for phase: String in ["lock", "active", "recovery"]:
		var key: String = _phase_key(id, cycle, lease, phase)
		if not await _wait(func() -> bool: return phase_observations.has(key), id + " accepted-cycle native " + phase): return false
		state = actor.call("state")
		if not _require(state.reservation_id == lease and int(state.cycle) == cycle and state.status == "running", id + " observed " + phase + " remains the same living native cycle"): return false
		if state.phase == phase:
			if not _framing(id + " " + phase): return false
			if phase == "active" and not await _capture(id + "-active"): return false
	state = actor.call("state")
	if id == "lone-guard":
		if not _require(_planar_distance(actor.global_position, source_start) <= PositionTolerance and actor.velocity.is_zero_approx(), "guard native lane keeps its real physical source stationary"):
			return false
	else:
		if not _require(_planar_distance(actor.global_position, source_start) > 0.9 and actor.global_position.distance_to(reservation.opening_position) <= PositionTolerance and state.adapter.finished, "swarm native short lunge genuinely travels to its planted ordinary-primary opening"):
			return false
	if not await _wait(func() -> bool: return scheduler.get_clock() >= float(proof.primary_time_s), id + " actual native primary deadline"):
		return false
	var returning: Vector3 = proof.attack_position - hero.global_position
	returning.y = 0.0
	if returning.length() > PositionTolerance:
		if not _require(absf(returning.length() - float(hero.equipment.resolved_stats().dash_distance)) <= PositionTolerance, id + " published return is one full ordinary dash"):
			return false
		if not await _swipe(returning, id + " native primary return"): return false
	if not await _capture(id + "-primary-opening"): return false
	return await _primary(actor, id)


func _pause_component(id: String, lease: String) -> bool:
	if not _require(game.call("request_pause_deferred"), "public deferred pause reaches a whole native component barrier"): return false
	if not await _wait(func() -> bool: return paused, "actual pause menu becomes visible", 2.0, true): return false
	var frozen: Dictionary = _live_observation()
	if not _require(not frozen.player.is_empty() and frozen.sources[id].reservation_id == lease, "paused component retains the actual Player and warning source"):
		return false
	var envelope: Dictionary = {"api_revision": CinderLevel.API_REVISION, "schema_version": CinderLevel.SNAPSHOT_SCHEMA_VERSION, "level_id": level.level_id, "scene_path": level.scene_file_path, "local_snapshot_version": level.local_snapshot_version, "progress": {"completed": false, "completion_id": "", "contact_exit_id": "", "checkpoint_id": "", "checkpoint_kind": "", "checkpoint_ids": {}}, "local": {}}
	if not _require(level.snapshot_state().is_empty() and not level.last_snapshot_error.is_empty() and not level.snapshot_error(envelope).is_empty() and not level.snapshot_error_with_player(envelope, frozen.player).is_empty() and not level.restore_state(envelope), "component refuses capture, pure aggregate validation and restore of a full campaign envelope"):
		return false
	var until: int = Time.get_ticks_msec() + 120
	while Time.get_ticks_msec() < until and not finishing:
		await process_frame
	if not _require(_live_observation() == frozen, "pause and unsupported envelope checks change no Player/source/cue/sample/clock/action bit or event"):
		return false
	if not await _capture(id + "-paused"): return false
	var before: Dictionary = game.call("get_input_observation_state")
	var records: Array[Dictionary] = hero.get_world_action_records()
	var button: Button = _resume_button(game.get("hud") as Node)
	if not _require(button != null, "actual shared pause GUI exposes Resume"): return false
	var point: Vector2 = button.get_global_rect().get_center()
	var press := InputEventMouseButton.new()
	press.button_index = MOUSE_BUTTON_LEFT; press.pressed = true; press.position = point; press.global_position = point
	Input.parse_input_event(press)
	await process_frame
	var release := InputEventMouseButton.new()
	release.button_index = MOUSE_BUTTON_LEFT; release.position = point; release.global_position = point
	Input.parse_input_event(release)
	await process_frame
	return _require(not paused and game.call("get_input_observation_state") == before and hero.get_world_action_records() == records, "real GUI Resume is consumed without movement, attack or aim changes")


func _live_observation() -> Dictionary:
	var actor_states: Dictionary = {}
	var cues: Dictionary = {}
	for id: String in sources:
		actor_states[id] = sources[id].call("state").duplicate(true)
		cues[id] = (sources[id].call("get_cue") as CinderThreatCue).state().duplicate(true)
	return {"player": hero.snapshot_state(), "sources": actor_states, "cues": cues, "scheduler_clock": scheduler.get_clock(), "reservations": scheduler.reservations(), "events": events, "world_actions": hero.get_world_action_records(), "input": game.call("get_input_observation_state"), "checkpoint": level.current_checkpoint(), "completed": level.is_completed()}


func _swipe(direction: Vector3, label: String) -> bool:
	if not await _ready_input(label): return false
	var horizontal := Vector3(direction.x, 0, direction.z).normalized()
	var finish := Vector2(270, 650)
	var delta: Vector2 = _screen_delta(horizontal) * 180.0
	if not _require(_input_safe(finish - delta) and _input_safe(finish) and (game.call("screen_to_direction", delta) as Vector3).dot(horizontal) > 0.9999, label + " uses the actual camera basis and safe screen coordinates"):
		return false
	var sequence: int = _last_sequence()
	var before: Dictionary = game.call("get_input_observation_state")
	var origin: Vector3 = hero.global_position
	var press := InputEventScreenTouch.new()
	press.index = 7; press.pressed = true; press.position = finish - delta
	Input.parse_input_event(press)
	await process_frame
	if not _guard_input(): return false
	var drag := InputEventScreenDrag.new()
	drag.index = 7; drag.position = finish; drag.relative = delta
	Input.parse_input_event(drag)
	await process_frame
	if not _guard_input(): return false
	var release := InputEventScreenTouch.new()
	release.index = 7; release.position = finish
	Input.parse_input_event(release)
	await process_frame
	if not _guard_input(): return false
	swipes += 1
	if not await _ready_input(label + " stopped landing"): return false
	var records: Array[Dictionary] = hero.get_world_action_records(sequence)
	var observed: Dictionary = game.call("get_input_observation_state")
	var normalized: Vector2 = finish / Vector2(PortraitSize)
	var distance: float = float(hero.equipment.resolved_stats().dash_distance)
	return _require(records.size() == 1 and records[0].kind == "dash" and not records[0].collision_shortened and absf(float(records[0].distance) - distance) <= PositionTolerance and records[0].direction.dot(horizontal) > 0.9999 and _planar_distance(origin, hero.global_position) > distance - PositionTolerance and records[0].landing.distance_to(hero.global_position) <= PositionTolerance and records[0].path.size() >= 2 and observed.sequence == before.sequence + 1 and observed.last_observation.kind == "swipe_release" and observed.anchor_normalized == normalized and game.call("get_aim_anchor_normalized") == normalized, label + " executes one full recorded recognizer dash with the literal release anchor")


func _primary(actor: CharacterBody3D, label: String) -> bool:
	if not await _ready_input(label + " primary readiness"): return false
	while Time.get_ticks_msec() - last_primary_ms <= 300:
		if not _guard_input(): return false
		await process_frame
	var offset: Vector3 = actor.global_position - hero.global_position
	offset.y = 0.0
	var direction: Vector3 = offset.normalized()
	var anchor: Vector2 = game.call("get_aim_anchor")
	var tap: Vector2 = anchor + _screen_delta(direction) * 160.0
	if not _require(actor.is_in_group("enemies") and offset.length() <= float(hero.equipment.resolved_stats().primary_range) and _input_safe(tap) and (game.call("aim_direction", tap) as Vector3).dot(direction) > 0.9999, label + " ordinary reach and exact aiming use the actual stopped body and last release"):
		return false
	var hp_before: float = float(actor.get("hp"))
	var sequence: int = _last_sequence()
	var press := InputEventScreenTouch.new()
	press.index = 7; press.pressed = true; press.position = tap
	Input.parse_input_event(press)
	await process_frame
	if not _guard_input(): return false
	var release := InputEventScreenTouch.new()
	release.index = 7; release.position = tap
	hero.shells = 0 # Immediate release-only ammo stage; passive reload stays live.
	Input.parse_input_event(release)
	await process_frame
	if not _guard_input(): return false
	last_primary_ms = Time.get_ticks_msec()
	primaries += 1
	var records: Array[Dictionary] = hero.get_world_action_records(sequence)
	var observed: Dictionary = game.call("get_input_observation_state")
	return _require(records.size() == 1 and records[0].kind == "primary" and records[0].hits == 1 and float(actor.get("hp")) < hp_before and records[0].direction.dot(direction) > 0.9999 and observed.last_observation.kind == "primary_tap" and observed.last_observation.accepted and observed.last_observation.world_action_sequence == records[0].sequence and (game.call("get_aim_anchor") as Vector2).is_equal_approx(anchor), label + " immediate native first tap hits with zero starting shells and preserves aim")


func _ready_input(label: String) -> bool:
	return await _wait(func() -> bool:
		var state: Dictionary = hero.get_threat_response_state()
		return state.stable and float(state.dash_cooldown_left_s) == 0.0 and float(state.primary_cooldown_left_s) == 0.0 and float(state.commitment_remaining_s) == 0.0, label + " reaches actual native input readiness", 4.0)


func _wait(predicate: Callable, label: String, seconds: float = 8.0, allow_pause: bool = false) -> bool:
	var deadline: int = Time.get_ticks_msec() + int(seconds * 1000.0)
	while not aborted and not finishing and Time.get_ticks_msec() < deadline:
		if not allow_pause and not _guard_input(): return false
		if predicate.call(): return _require(true, label)
		_sample_preparing()
		await process_frame
	if aborted or finishing: return false
	print("L3 WAIT DIAGNOSTIC ", label, " source_states=", _source_states(), " last_admission=", level.get("last_admission") if is_instance_valid(level) else {}, " encounter_error=", level.get("last_encounter_error") if is_instance_valid(level) else "", " camera=", game.call("get_camera_framing_state") if is_instance_valid(game) else {}, " paused=", paused)
	return _require(false, label + " within the finite component window")


func _guard_input() -> bool:
	if aborted or finishing: return false
	return _require_unexpected(not paused and is_instance_valid(hero) and not hero.dead, "unexpected focus/pause/death stops scripted input; fixture never forces resume")


func _framing(label: String) -> bool:
	var points: Array = level.camera_framing_points()
	return _require(not points.is_empty() and points.size() <= 224 and level.last_camera_framing_error.is_empty() and String(game.call("camera_framing_error", points)).is_empty(), label + " contains actual source/art/full lane and native response corners in the shared portrait camera")


func _notice_admission(id: String, answer: Dictionary) -> void:
	if not answer.get("accepted", false) or not sources.has(id): return
	var actor: CharacterBody3D = sources[id]
	var state: Dictionary = actor.call("state")
	if answer.reservation.source_instance_id == actor.get_instance_id() and state.reservation_id == answer.reservation_id:
		admissions[String(answer.reservation_id)] = answer.duplicate(true)


func _notice_phase(state: Dictionary, id: String) -> void:
	events += 1
	var lease: String = String(state.get("reservation_id", ""))
	if not lease.is_empty() and state.get("status") == "running":
		var key: String = _phase_key(id, int(state.cycle), lease, String(state.phase))
		if not phase_observations.has(key):
			var cue: Dictionary = (sources[id].call("get_cue") as CinderThreatCue).state()
			phase_observations[key] = {"source_id": id, "cycle": state.cycle, "reservation_id": lease, "phase": state.phase, "clock_s": scheduler.get_clock(), "source_position": state.source_position, "cue": cue}
			_require(cue.phase == state.phase and cue.geometry == state.geometry and cue.source_position == state.source_position and cue.source_visible and (cue.footprint_visible == (state.phase != "recovery")) and (cue.active_fill_visible == (state.phase == "active")), id + " native phase publishes the matching required cue before damage")
	_sample_preparing()


func _notice_hit(hero_id: String, cycle: int, result: Dictionary, id: String) -> void:
	events += 1
	if result.get("accepted", false): hit_events.append({"source_id": id, "hero_id": hero_id, "cycle": cycle, "result": result.duplicate(true)})


func _sample_preparing() -> void:
	if not is_instance_valid(level) or finishing: return
	var count: int = 0
	for id: String in level.call("current_source_ids"):
		if not is_instance_valid(sources.get(id)): continue
		if sources[id].call("state").phase in ["warning", "lock"]: count += 1
	max_preparing = maxi(max_preparing, count)
	if count > 1: preparing_violation = true


func _close_world(direct: bool) -> bool:
	paused = true
	await process_frame
	_sample_preparing()
	if not _require(hero.hp == initial_hp and hit_events.is_empty() and not preparing_violation and max_preparing == 1, current_scope + " actual safe responses preserve HP with exactly one maximum preparing swarm"):
		return false
	var encoded: Array[Dictionary] = []
	for record: Dictionary in world_records:
		var value: Dictionary = CinderPlayer.encode_world_action_record(record, hero.get_world_action_clock())
		if not _require(not value.is_empty() and record.kind in ["dash", "primary"], current_scope + " actual world action has supported exact native transport"): return false
		encoded.append(value)
	if not _require(input_observations.size() == world_records.size(), current_scope + " every actual accepted gesture has its executed world record"):
		return false
	worlds.append({"scope": current_scope, "initial_hp": initial_hp, "final_hp": hero.hp, "max_preparing": max_preparing, "hit_events": hit_events.duplicate(true), "phases": _portable(phase_observations), "admissions": _portable(admissions), "world_actions": encoded, "input_observations": _portable(input_observations), "source_states": _portable(_source_states()), "direct_queue_free": direct})
	var old_game: Node = game
	var old_level: Node = level
	var old_hero: Node = hero
	var old_scheduler: Node = scheduler
	var old_sources: Array = sources.values().duplicate()
	if direct:
		if not _require(not scheduler.reservations().is_empty(), "direct entered queue_free starts with an actual running native lease"): return false
	else:
		level.exit_level()
		if not _require(not scheduler.is_physics_processing() and scheduler.reservations().is_empty(), current_scope + " explicit level exit stops Scheduler and removes every native lease"): return false
		for actor: CharacterBody3D in old_sources:
			if not _require(not actor.is_physics_processing() and not actor.is_in_group("enemies"), current_scope + " explicit exit retires source physics and target membership"): return false
	var effects: PixelEffects = game.get("fx") as PixelEffects
	if is_instance_valid(effects): effects.clear() # Public voice cleanup, not game state.
	game.queue_free()
	await process_frame
	var removed: bool = not is_instance_valid(old_game) and not is_instance_valid(old_level) and not is_instance_valid(old_hero) and not is_instance_valid(old_scheduler)
	for actor: Variant in old_sources: removed = removed and not is_instance_valid(actor)
	sources.clear(); game = null; level = null; hero = null; scheduler = null
	paused = false
	await process_frame
	return _require(removed and get_nodes_in_group("enemies").is_empty(), current_scope + " deferred world cleanup frees real source/Hero/Scheduler nodes and leaves no live enemy group")


func _capture(stage: String) -> bool:
	if not portrait: return not aborted and not finishing
	await RenderingServer.frame_post_draw
	if aborted or finishing: return false
	var image: Image = root.get_texture().get_image()
	var path: String = capture_dir.path_join("%02d-%s-%s.png" % [shots.size() + 1, current_scope, stage])
	if not _require(image != null and image.get_size() == PortraitSize and image.save_png(path) == OK, "actual540x1170 renderer saves " + stage): return false
	shots.append({"scope": current_scope, "stage": stage, "path": ProjectSettings.globalize_path(path), "hero_position": _portable(hero.global_position), "hp": hero.hp, "paused": paused, "native_focus": DisplayServer.window_is_focused(), "sources": _portable(_source_states()), "camera": _portable(game.call("get_camera_framing_state"))})
	return true


func _source_states() -> Dictionary:
	var result: Dictionary = {}
	for id: String in sources:
		if is_instance_valid(sources[id]): result[id] = sources[id].call("state").duplicate(true)
	return result


func _screen_delta(direction: Vector3) -> Vector2:
	var camera: Camera3D = game.get("camera") as Camera3D
	var right: Vector3 = camera.global_basis.x
	var down: Vector3 = camera.global_basis.z
	right.y = 0.0; down.y = 0.0
	right = right.normalized(); down = down.normalized()
	var determinant: float = right.x * down.z - right.z * down.x
	return Vector2((direction.x * down.z - direction.z * down.x) / determinant, (right.x * direction.z - right.z * direction.x) / determinant).normalized()


func _resume_button(surface: Node) -> Button:
	if surface == null: return null
	if surface is Button and (surface as Button).is_visible_in_tree() and (surface as Button).text.begins_with("RESUME"): return surface as Button
	for child: Node in surface.get_children():
		var candidate: Button = _resume_button(child)
		if candidate != null: return candidate
	return null


func _phase_key(id: String, cycle: int, lease: String, phase: String) -> String:
	return "%s/%d/%s/%s" % [id, cycle, lease, phase]


func _last_sequence() -> int:
	var records: Array[Dictionary] = hero.get_world_action_records()
	return 0 if records.is_empty() else int(records.back().sequence)


func _input_safe(point: Vector2) -> bool:
	return point.is_finite() and Rect2(20, 300, 500, 670).has_point(point)


func _planar_distance(a: Vector3, b: Vector3) -> float:
	return Vector2(a.x - b.x, a.z - b.z).length()


func _portable(value: Variant) -> Variant:
	if value is Vector3: return [value.x, value.y, value.z]
	if value is Vector2: return [value.x, value.y]
	if value is Array:
		var array: Array = []
		for item: Variant in value: array.append(_portable(item))
		return array
	if value is Dictionary:
		var dictionary: Dictionary = {}
		for key: Variant in value: dictionary[String(key)] = _portable(value[key])
		return dictionary
	if value is Object: return "native object omitted: " + value.get_class()
	return value


func _require(condition: bool, label: String) -> bool:
	checks += 1
	if not condition:
		failures += 1
		aborted = true
		push_error(label)
	return condition


func _require_unexpected(condition: bool, label: String) -> bool:
	if condition: return true
	if aborted: return false
	return _require(false, label)


func _watchdog() -> void:
	if not finishing and Time.get_ticks_msec() - started_ms > WatchdogMs:
		_require_unexpected(false, "whole component fixture completes within70 seconds")
		_finish.call_deferred()


func _finish() -> void:
	if finishing: return
	finishing = true
	if is_instance_valid(level): level.exit_level()
	if is_instance_valid(game):
		var effects: PixelEffects = game.get("fx") as PixelEffects
		if is_instance_valid(effects): effects.clear()
		game.queue_free()
	paused = false
	await process_frame
	if portrait and not capture_dir.is_empty():
		var file := FileAccess.open(capture_dir.path_join("evidence.json"), FileAccess.WRITE)
		if file != null:
			file.store_string(JSON.stringify({"scope": "actual-input neutral/Standard C31/C32 component greyboxes; one of three grove swarmers and isolated surviving guard; no spores/fulllevel/campaign save/finalart/kit-profile matrix/human balance/performance claim", "checks": checks, "failures": failures, "actual_swipes": swipes, "actual_primaries": primaries, "normal_focus_out_preserved": true, "worlds": worlds, "captures": shots}, "\t"))
			file.close()
		else:
			checks += 1; failures += 1
			push_error("component portrait evidence file failed to open")
	if process_frame.is_connected(_watchdog): process_frame.disconnect(_watchdog)
	print("A1-L3 ACTUAL COMPONENT GREYBOX: %d checks, %d failures; %d swipes, %d primaries; NO TELEPORTS / NO SPORES / NO FULL-LEVEL CLAIM" % [checks, failures, swipes, primaries])
	quit(1 if failures else 0)
