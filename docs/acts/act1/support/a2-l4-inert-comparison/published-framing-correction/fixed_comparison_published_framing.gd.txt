	var plan: Dictionary = await _first_plan()
	if plan.is_empty(): return false
	if plan.actor_id != "putney_handler": return _stop("inconclusive", "First actual selected candidate proof is not the proposed fixed-lane role; no wait/preference override")
	_component["native_admission_escape_proof_not_hold_certificate"] = plan.duplicate(true)
	var lease: Dictionary = plan.reservation
	# Independently assess a stationary capsule through the entire committed
	# native cycle. No part of the native escape path certifies this hold.
	var hold: Array[Dictionary] = [{"from": _station, "to": _station, "start_s": _begin_s, "end_s": float(lease.recovery_until_s), "kind": "TEST_stationary_hold_independent"}]
	_component["independent_held_path"] = hold.duplicate(true)
	var attacked: bool = false
	var complete_s: float = INF
	while _clock() <= float(lease.recovery_until_s) + EPS:
		if Vector2(_hero().global_position.x, _hero().global_position.z).distance_to(STATION_XZ) > EPS:
			return _stop("inconclusive", "Candidate left the one fixed stationary XZ; no reposition/search")
		var error: String = _path_union_error(hold)
		if not error.is_empty(): return _stop("refused_or_inconclusive", "Independent complete held-capsule assessment: " + error)
		if not attacked and _window_open(plan.actor_id) and float(_hero().get_threat_response_state().primary_cooldown_left_s) <= EPS:
			complete_s = _clock() + float(_hero().stats.primary_cooldown)
			if complete_s > float(lease.recovery_until_s) - CinderThreatScheduler.TIME_MARGIN: return _stop("inconclusive", "Actual native opening cannot retain complete ordinary cadence")
			if not bool(_scheduler().call("_opening_reachable", _hero().global_position, _actor(plan.actor_id).global_position, float(_hero().stats.primary_range))): return _stop("refused", "Candidate actual ordinary opening fails native range/LOS")
			if not _primary(plan.actor_id): return false
			attacked = true
			_component["actual_primary_complete_s"] = complete_s
		if attacked and _clock() + EPS >= complete_s:
			_component["actual_hold_observed_until_s"] = _clock()
			_component["assessed_held_extent_is_not_an_executed_whole_recovery_claim"] = true
			return _expect(_incoming.is_empty() and _hero().hp == _initial_hp and not _fired.has("blast"), "actual candidate", "real stationary hold and useful ordinary primary through complete cadence; zero incoming native contact/HP loss")
		if not _plan_current(plan) and float(_actor(plan.actor_id).get("hp")) > 0.0:
			return _stop("inconclusive", "Candidate native source cancelled/retired; no recreated cycle")
		if not await _tick(): return false
	return _stop("inconclusive", "Candidate had no executed useful ordinary response within its actual native recovery")


func _first_plan() -> Dictionary:
	for frame: int in range(180):
		if not _live(): _stop("refused", _runtime_error()); return {}
		var latest: Dictionary = {}
		var state: Dictionary = _state()
		# Same latest-actual-lock criterion and authored iteration on ties as
		# the original fixture. Never wait on a chosen ID while another proof
		# is already due, or select an artificial left/right route.
		for id: String in state.active_ids:
			var current: Dictionary = state.exchanges.get(id, {})
			if current.get("status") in ["cancelled", "complete", "defeated"]:
				_stop("inconclusive", "Actual source retired before a supported response: " + id); return {}
			if current.get("status") != "running": continue
			var lease: Dictionary = _find_lease(String(current.get("reservation_id", "")))
			var proof: Dictionary = current.get("proof", {})
			if current.has("mechanism_id"): proof = _level().call("mechanism_proof", String(current.mechanism_id))
			if not lease.get("armed", false) or not proof.get("accepted", false): continue
			var plan: Dictionary = {"actor_id": id, "cycle": current.cycle, "reservation": lease.duplicate(true), "proof": proof.duplicate(true), "resolved_role": current.resolved_role.duplicate(true)}
			if latest.is_empty() or float(lease.lock_from_s) > float(latest.reservation.lock_from_s): latest = plan
		if not latest.is_empty(): return latest
		if not await _tick(): return {}
	_stop("inconclusive", "Bounded actual first native admission/proof unavailable; no source/pose change")
	return {}


func _execute_dash(plan: Dictionary, segment: Dictionary) -> bool:
	var hero: CinderPlayer = _hero()
	var response: Dictionary = hero.get_threat_response_state()
	response.merge(_level().call("_response_context"), true)
	var response_error: String = _scheduler().call("response_error", response)
	if not response_error.is_empty(): return _stop("inconclusive", "Actual native pre-dash response preflight: " + response_error)
	if not response.get("stable", false) or float(response.dash_cooldown_left_s) > EPS or _clock() > float(segment.start_s) + 1.0 / float(Engine.physics_ticks_per_second) + EPS:
		return _stop("inconclusive", "Actual native dash readiness/deadline differs; no late repaired proof")
	if hero.global_position.distance_to(segment.from) > CinderThreatScheduler.FEET_TOLERANCE:
		return _stop("inconclusive", "Actual native dash no longer begins at its selected proof origin")
	var error: String = _path_union_error(plan.proof.path)
	if not error.is_empty(): return _stop("inconclusive", "Full native response unsafe before actual dash: " + error)
	var direction: Vector3 = segment.to - segment.from
	direction.y = 0.0
	var before: int = _actions.size()
	if not _swipe(direction.normalized()): return false
	for frame: int in range(90):
		if not await _tick(): return false
		var pending_error: String = _path_union_error(plan.proof.path)
		if not pending_error.is_empty(): return _stop("inconclusive", "New actual/held pressure invalidated original complete dash/primary path: " + pending_error)
		for action: Dictionary in _actions.slice(before):
			if action.kind != "dash": continue
			if not _expect((action.landing as Vector3).distance_to(segment.to) <= CinderThreatScheduler.FEET_TOLERANCE and not action.collision_shortened and action.path.size() >= 2, "actual original", "native dash completes its selected whole supported segment"): return false
			return true
	return _stop("inconclusive", "Actual dash did not publish a completed world-action record")


func _path_error(proof: Dictionary) -> String:
	if proof.get("accepted") != true or proof.get("uses_blast") != false or proof.get("uses_invulnerability") != false or proof.get("proof_scope") != "static_box_floor_full_dash_stationary_primary" or not proof.get("path") is Array or proof.path.size() not in [4, 6]: return "Complete native supported ordinary-primary proof required"
	if not _geometry.finite_number(proof.get("primary_time_s")) or not _geometry.finite_number(proof.get("response_complete_s")) or float(proof.response_complete_s) != float(proof.primary_time_s) + float(_hero().stats.primary_cooldown): return "Native proof omits exact full selected primary cadence"
	var kinds: Array[String] = ["recognition_and_ready", "escape_dash", "recovery_wait", "ordinary_primary"]
	if proof.path.size() == 6: kinds.assign(["recognition_and_ready", "escape_dash", "recovery_wait", "positioning_dash", "primary_ready", "ordinary_primary"])
	var previous: Dictionary = {}
	for index: int in range(proof.path.size()):
		var segment: Variant = proof.path[index]
		if not segment is Dictionary or segment.size() != 5 or segment.get("kind") != kinds[index] or not _geometry.finite_vector(segment.get("from")) or not _geometry.finite_vector(segment.get("to")) or not _geometry.finite_number(segment.get("start_s")) or not _geometry.finite_number(segment.get("end_s")) or float(segment.end_s) < float(segment.start_s): return "Native full response segment unsupported"
		if not previous.is_empty() and (segment.from != previous.to or float(segment.start_s) != float(previous.end_s)): return "Native complete response is discontinuous"
		previous = segment
	return ""


func _path_union_error(path: Array) -> String:
	if not _live(): return _runtime_error()
	if not _union_error.is_empty(): return _union_error
	for key: String in _pending:
		if not _pending[key].is_empty(): return "Native unresolved pending/deferred paths retained: " + key
	var floors: Array = _level().call("floor_regions")
	if path.size() == 1 and path[0].get("kind") == "TEST_stationary_hold_independent":
		var response: Dictionary = _hero().get_threat_response_state()
		response.merge(_level().call("_response_context"), true)
		var response_error: String = _scheduler().call("response_error", response)
		if not response_error.is_empty(): return "Independent native stable/body/timing preflight: " + response_error
	for segment: Dictionary in path:
		if not _geometry.finite_vector(segment.get("from")) or not _geometry.finite_vector(segment.get("to")) or not _geometry.finite_number(segment.get("start_s")) or not _geometry.finite_number(segment.get("end_s")) or float(segment.end_s) < float(segment.start_s): return "Unsupported complete response interval"
		if not bool(_scheduler().call("_floor_path_supported", segment.from, segment.to, floors)) or not bool(_scheduler().call("_capsule_path_clear", segment.from, segment.to, _hero(), floors)): return "Actual full response support/capsule predicate refused"
	for lease: Dictionary in _current_leases:
		if not lease.get("armed", false) and float(lease.active_from_s) <= float(path[-1].end_s): return "A real unarmed tracking future/due-lock is unknown; stationary/current geometry is not its certificate"
	for lease: Dictionary in _locked_history.values():
		if _geometry.timed_path_hits(lease.geometry, path, float(lease.active_from_s), float(lease.active_until_s), CinderThreatScheduler.CAPSULE_RADIUS + CinderThreatScheduler.SKIN): return "Actual current/retained locked threat intersects complete timed capsule path: " + String(lease.id)
	return _guard_error()


func _stationary_counterfactual() -> Dictionary:
	var end_s: float = float(_component.get("actual_primary_complete_s", -1.0))
	if end_s <= _begin_s or _locked_history.is_empty(): return {"accepted": false, "reason": "No actual original complete response/locked threat history"}
	var path: Array[Dictionary] = [{"from": _station, "to": _station, "start_s": _begin_s, "end_s": end_s, "kind": "counterfactual_stationary_NOT_EXECUTED"}]
	var intersections: Array[String] = []
	for lease: Dictionary in _locked_history.values():
		if _geometry.timed_path_hits(lease.geometry, path, float(lease.active_from_s), float(lease.active_until_s), CinderThreatScheduler.CAPSULE_RADIUS + CinderThreatScheduler.SKIN): intersections.append(String(lease.id))
	return {"accepted": true, "executed": false, "path": path, "actual_locked_union": _locked_history.duplicate(true), "intersects": not intersections.is_empty(), "intersecting_actual_lease_ids": intersections, "claim": "Timed stationary capsule calculation only; no observed original hold or damage, no reconstructed lock"}


func _tick() -> bool:
	if _finished or not is_instance_valid(_probe): return false
	await _probe.tick_finished
	if _finished: return false
	_ticks += 1
	if _ticks > MAX_TICKS: return _stop("inconclusive", "One fixed component reached its finite native-tick budget")
	if not _live(): return _stop("refused", "Actual native component stopped: " + _runtime_error())
	_collect_union()
	var error: String = _guard_error()
	var camera: Camera3D = _game.get("camera") as Camera3D
	var hud: GameHUD = _game.get("hud") as GameHUD
	_component.trace.append(_pure_native_data({"clock_s": _clock(), "hero": _hero().get_threat_response_state(), "response_context": _level().call("_response_context"), "sources": _state().exchanges, "native_current_leases": _current_leases.duplicate(true), "native_locked_history": _locked_history.duplicate(true), "native_pending_paths": _pending.duplicate(true), "union_error": _union_error, "guard_error": error, "framing_diagnostics": _framing_diagnostics.duplicate(true), "camera": {"transform": camera.global_transform, "projection": camera.projection, "size": camera.size, "viewport_size": camera.get_viewport().get_visible_rect().size, "hud_safe_rect": hud.combat_safe_rect()}}))
	if not error.is_empty(): return _stop("refused_or_inconclusive", "Actual native guard/support/framing failure: " + error)
	if not _incoming.is_empty() or _hero().hp != _initial_hp: return _stop("unsafe", "Actual incoming native hit/contact or HP loss; native invulnerability does not supply success")
	if not _progress_events.is_empty(): return _stop("invalid_TEST_scope", "Focus unexpectedly requested campaign progression")
	return true


func _collect_union() -> void:
	_current_leases.clear(); _pending.clear(); _union_error = ""
	var level: CinderLevel = _level()
	var owners: Array[Node3D] = []
	for key: String in ["_actors", "_mechanisms", "_clouds"]:
		for node: Node3D in (level.get(key) as Dictionary).values(): owners.append(node)
	var by_id: Dictionary = {}
	for owner: Node3D in owners:
		var control: Dictionary = _scheduler().call("source_control_state", owner)
		if control.is_empty() or not control.get("outside_transaction", false): _union_error = "Actual pure source-control custody unavailable"; return
		for lease: Dictionary in control.reservations:
			if not lease.get("id") is String or not lease.get("geometry") is Dictionary or not _geometry.error(lease.geometry).is_empty() or not _geometry.finite_number(lease.get("active_from_s")) or not _geometry.finite_number(lease.get("active_until_s")) or not _geometry.finite_number(lease.get("lock_from_s")) or not _geometry.finite_number(lease.get("recovery_until_s")): _union_error = "Actual native lease shape/timing unsupported"; return
			by_id[lease.id] = lease
			if lease.get("armed", false):
				if _locked_history.has(lease.id) and (_locked_history[lease.id].geometry != lease.geometry or _locked_history[lease.id].active_from_s != lease.active_from_s or _locked_history[lease.id].active_until_s != lease.active_until_s): _union_error = "Already locked actual geometry/timing changed"; return
				_locked_history[lease.id] = lease.duplicate(true)
	for lease: Dictionary in by_id.values(): _current_leases.append(lease)
	var exchange: Node = level.get("_exchange")
	for id: String in exchange.get("_records"):
		var record: Dictionary = (exchange.get("_records") as Dictionary)[id]
		_pending["ray:" + id] = (record.get("deferred_paths", []) as Array).duplicate(true)
	for id: String in level.get("_mechanisms"):
		var node: Node = (level.get("_mechanisms") as Dictionary)[id]
		_pending["tool:" + id] = (node.get("_pending_segments") as Dictionary).duplicate(true)
	for id: String in level.get("_clouds"):
		var node: Node = (level.get("_clouds") as Dictionary)[id]
		var stage: String = String(node.get("_pending_stage"))
		_pending["bank:" + id] = {} if stage.is_empty() else {"pending_stage": stage, "trace": (node.get("_trace") as Array).duplicate(true)}


func _guard_error() -> String:
	_framing_diagnostics.clear()
	if not _live(): return _runtime_error()
	var hero: CinderPlayer = _hero()
	var level: CinderLevel = _level()
	if hero.equipment.snapshot() != KIT: return "Fixed existing kit changed"
	if _pickup_seen or _fired.has("blast") or hero.shells < _ammo_last: return "Ammo consumption/blast/pickup occurred; ordinary opportunity cannot depend on it"
	_ammo_last = hero.shells # passive and hit-earned native reload remain legal
	var retained: Dictionary = level.get("_l4_construction")
	if not retained.is_read_only() or not (retained.positions as Dictionary).is_read_only() or retained.profile_id != "challenge" or _exact.stringify(_scheduler().call("encounter_profile")) != _exact.stringify(retained.profile): return "Frozen full Challenge construction/Scheduler profile differs"
	var custody: String = level.call("_actor_custody_error")
	if not custody.is_empty(): return custody
	for id: String in retained.positions:
		var actor: Node3D = _actor(id)
		if not is_instance_valid(actor) or actor.global_position != retained.positions[id] or actor.get("_authored_root_position") != retained.positions[id]: return "Native configured root/custody differs: " + id
	var floors: Array = level.call("floor_regions")
	var error: String = _scheduler().call("_floor_error", floors, hero.global_position)
	if not error.is_empty(): return error
	if not bool(_scheduler().call("_floor_path_supported", hero.global_position, hero.global_position, floors)) or not bool(_scheduler().call("_capsule_path_clear", hero.global_position, hero.global_position, hero, floors)): return "Current actual capsule/support predicate refused"
	var body: CollisionShape3D = hero.get_node_or_null("BodyCollision") as CollisionShape3D
	if not is_instance_valid(body) or body.disabled or not body.shape is CapsuleShape3D or body.position != Vector3(0, CinderThreatScheduler.CAPSULE_CENTER_Y, 0) or body.basis != Basis.IDENTITY or not hero.global_basis.is_equal_approx(Basis.IDENTITY) or not is_equal_approx((body.shape as CapsuleShape3D).radius, CinderThreatScheduler.CAPSULE_RADIUS) or not is_equal_approx((body.shape as CapsuleShape3D).height, CinderThreatScheduler.CAPSULE_HEIGHT): return "Actual shared capsule convention differs"
	var state: Dictionary = _state()
	for id: String in state.active_ids:
		var current: Dictionary = state.exchanges[id]
		if current.get("status") == "cancelled": return "Actual native attack cancellation: " + id + ":" + String(current.get("last_cancel_reason", ""))
		if current.get("status") != "running": continue
		if id == "putney_scout":
			if not bool(level.call("_exchange_framed", _actor(id), current)): return "Actual native Ray presentation/framing lost"
		elif id == "putney_handler":
			if not bool(level.call("_mechanism_framed", "tool_" + id, current)): return "Actual native Handler presentation/framing lost"
	# The full original wrapper still validates forecasts + published points.
	# Normal Game planning consumes that original list; this TEST never refits.
	var full_points: Array = level.camera_framing_points()
	var hero_points: Array = _game.call("player_camera_framing_points")
	_framing_diagnostics = {
		"full_original_wrapper_points": full_points.duplicate(true),
		"full_original_wrapper_error": level.last_camera_framing_error,
		"full_original_forecast_points": (level.get("_forecast_points") as Dictionary).duplicate(true),
		"normal_game_planner_state": _game.call("get_camera_framing_state"),
		"normal_game_last_camera_framing_error": _game.get("last_camera_framing_error"),
		"actual_full_hero_bounds": hero_points.duplicate(true),
		"diagnostic_scope": "Original full forecast/planner diagnostics retained; only RUNNING published points classify current containment",
	}
	if not level.last_camera_framing_error.is_empty(): return level.last_camera_framing_error
	var published_points: Array = _published_framing_points()
	_framing_diagnostics["TEST_current_published_running_points"] = published_points.duplicate(true)
	# Same cardinality and finite bounded Vector3 contract as the original
	# CinderLevel wrapper; subset classification never relaxes point validity.
	if published_points.size() > 224: return "Camera framing requires at most 224 actual world corners"
	for point: Variant in published_points:
		if not point is Vector3 or not point.is_finite() or maxf(absf(point.x), maxf(absf(point.y), absf(point.z))) > 1024.0:
			return "Camera framing corners must be finite bounded world positions"
	var camera_error: String = _game.call("camera_framing_error", published_points if not published_points.is_empty() else hero_points)
	_framing_diagnostics["TEST_current_published_containment_error"] = camera_error
	if not camera_error.is_empty(): return camera_error
	return ""


func _published_framing_points() -> Array:
	# Pure TEST projection of exact b81f host camera logic. These aliases are
	# native retained dictionaries; the three RUNNING branch bodies below are
	# byte-for-byte copied. Only two idle forecast elif lines are omitted.
	var level: CinderLevel = _level()
	var runtime_error: String = String(level.get("runtime_error"))
	var _entered: bool = bool(level.get("_entered"))
	var hero: CinderPlayer = level.hero
	var _mechanisms: Dictionary = level.get("_mechanisms")
	var _views: Dictionary = level.get("_views")
	var _exchange: Node = level.get("_exchange")
	var _clouds: Dictionary = level.get("_clouds")
	var _actors: Dictionary = level.get("_actors")
	var _bank_views: Dictionary = level.get("_bank_views")
	var LondonSequence: Script = _construction.Sequence
	var Codec: Script = _construction.Codec
	var BANK_TO_TENDER: Dictionary = {"flood_bank": "flood_tender", "villa_bank": "villa_tender"}
	if not runtime_error.is_empty(): return []
	if not _entered or not is_instance_valid(hero): return []
	var points: Array[Vector3] = []
	for id: String in _mechanisms:
		var state: Dictionary = _mechanisms[id].call("state")
		if state.status == "running":
			points.append_array(_source_points(id))
			if state.phase != "recovery": points.append_array(_lane_points(state.geometry))
			if _views.has(id):
				for key: String in ["landing", "attack_position"]: points.append_array(_landing_points(Codec.read_vector3(_views[id][key])))
	if is_instance_valid(_exchange):
		for ray_id: String in LondonSequence.RAYS:
			var scout: Dictionary = _exchange.call("state", ray_id)
			if scout.get("status") == "running":
				points.append_array(_source_points(ray_id))
				if scout.phase != "recovery":
					points.append_array(_lane_points(_state_geometry(scout)))
					var proof: Dictionary = scout.get("proof", {})
					if not proof.get("landing") is Vector3: proof = scout.get("presentation_witness", {})
					for key: String in ["landing", "attack_position"]:
						if proof.get(key) is Vector3: points.append_array(_landing_points(proof[key]))
	for id: String in _clouds:
		if _clouds[id].call("state").status == "running":
			points.append_array(_required_source_points(_actors[BANK_TO_TENDER[id]]))
			points.append_array(_clouds[id].call("get_required_camera_points"))
			if _bank_views.has(id):
				for key: String in ["landing", "attack_position"]: points.append_array(_landing_points(Codec.read_vector3(_bank_views[id][key])))
	return points

func _screen_delta(direction: Vector3) -> Vector2:
	var camera: Camera3D = _game.get("camera") as Camera3D
	var right: Vector3 = camera.global_basis.x
	var down: Vector3 = camera.global_basis.z
	right.y = 0.0; down.y = 0.0
	return Vector2(direction.dot(right.normalized()), direction.dot(down.normalized())) * 120.0


func _swipe(direction: Vector3) -> bool:
	var start: Vector2 = Vector2(0.5, 0.60) * root.get_visible_rect().size
	var end: Vector2 = start + _screen_delta(direction)
	if not _expect((_game.call("screen_to_direction", end - start) as Vector3).is_equal_approx(direction), "actual input", "real camera maps native selected direction without a side tie-break"): return false
	var press := InputEventScreenTouch.new()
	press.index = 23; press.pressed = true; press.position = start
	root.push_input(press, true)
	var drag := InputEventScreenDrag.new()
	drag.index = 23; drag.position = end; drag.relative = end - start
	root.push_input(drag, true)
	var release := InputEventScreenTouch.new()
	release.index = 23; release.position = end
	root.push_input(release, true)
	return _expect(_hero().get_committed_dash_state().get("active", false), "actual input", "real routed swipe commits the native dash")


func _primary(id: String) -> bool:
	var hero: CinderPlayer = _hero()
	var actor: Node3D = _actor(id)
	var direction: Vector3 = actor.global_position - hero.global_position
	direction.y = 0.0
	if direction.length_squared() <= EPS: return _stop("inconclusive", "Ordinary aim direction is degenerate at the fixed native opening")
	direction = direction.normalized()
	var position: Vector2 = _game.call("get_aim_anchor") + _screen_delta(direction)
	if not _expect((_game.call("aim_direction", position) as Vector3).is_equal_approx(direction), "actual input", "ordinary aim preserves last swipe final-release anchor"): return false
	var before: int = _actions.size()
	var hp_before: float = float(actor.get("hp"))
	var press := InputEventScreenTouch.new()

	root.push_input(press, true)
	var release := InputEventScreenTouch.new()
	release.index = 24; release.position = position

	var accepted: bool = _actions.size() == before + 1 and _actions.back().kind == "primary" and int(_actions.back().hits) > 0 and float(actor.get("hp")) < hp_before and not _fired.has("blast")

	if not _expect(accepted, "actual useful primary", "real first tap damages its native recovering HP30 source without blast or synthetic damage"): return _stop("refused", "Actual ordinary-primary effect unavailable")

func _close_component() -> void:
	if _component.is_empty(): return
	_component["actual_actions"] = _actions.duplicate(true)
	_component["actual_incoming_native_contacts"] = _incoming.duplicate(true)
	_component["progress_events"] = _progress_events.duplicate()
	_component["actual_fired_events"] = _fired.duplicate()
	_component["passive_or_hit_earned_reload_preserved"] = true
	_component["last_native_locked_history"] = _locked_history.duplicate(true)
	if is_instance_valid(_hero()): _component["final_player"] = {"position": _hero().global_position, "hp": _hero().hp, "shells": _hero().shells}
	_component = _pure_native_data(_component)
	if is_instance_valid(_game):
		var level: CinderLevel = _level()
		var reference: WeakRef = weakref(level) if is_instance_valid(level) else null
		var game_reference: WeakRef = weakref(_game)
		if is_instance_valid(level): level.exit_level()
		_game.free()
		_game = null
		var released: bool = (reference == null or reference.get_ref() == null) and game_reference.get_ref() == null
		_cleanup_ok = _cleanup_ok and released
		_expect(released, "cleanup", "normal attached exit/free retires owned focus subtree")
	_components.append(_component)
	_component = {}

func _pure_native_data(value: Variant) -> Variant:
	# Native geometric/timing types remain native in the binary trace. Only
	# ephemeral handles become EXPLICIT observation-time identity/binding data;
	# these records cannot reconstruct or authorize a Node/Resource/Callable.
	match typeof(value):
		TYPE_DICTIONARY:
			var result: Dictionary = {}
			for key: Variant in value: result[key] = _pure_native_data(value[key])
			return result
		TYPE_ARRAY:
			var result: Array = []
			for child: Variant in value: result.append(_pure_native_data(child))
			return result
		TYPE_OBJECT:
			_explicit_binding_refs += 1
			if not is_instance_valid(value):
				_evidence_error = "Native binding released before observation-time projection"
				return {"UNSUPPORTED_NATIVE_REFERENCE": _evidence_error}
			if not value is CinderPlayer and not value is StaticBody3D and not value is CollisionShape3D and not value is BoxShape3D and not value is CapsuleShape3D:
				_evidence_error = "Unrecognized native Object binding: " + value.get_class()
				return {"UNSUPPORTED_NATIVE_REFERENCE": _evidence_error}
			var binding: Dictionary = {"EXPLICIT_NATIVE_BINDING_REFERENCE": value.get_class(), "instance_id": value.get_instance_id(), "reconstructible": false}
			if value is Node:
				binding["path"] = String(value.get_path()) if value.is_inside_tree() else "outside-tree"
				var script: Script = value.get_script() as Script
				binding["script_path"] = script.resource_path if script != null else ""
				if value is Node3D:
					binding["actual_global_transform"] = value.global_transform
					binding["world_instance_id"] = value.get_world_3d().get_instance_id() if value.is_inside_tree() else 0
				if value is CollisionObject3D:
					binding["collision_layer"] = value.collision_layer
					binding["collision_mask"] = value.collision_mask
				if value is CollisionShape3D:
					binding["disabled"] = value.disabled
					binding["actual_shape"] = _pure_native_data(value.shape)
			elif value is Resource:
				binding["resource_path"] = value.resource_path
				if value is BoxShape3D: binding["actual_box_size"] = value.size
				elif value is CapsuleShape3D:
					binding["actual_capsule_radius"] = value.radius
					binding["actual_capsule_height"] = value.height
			return binding
		TYPE_CALLABLE, TYPE_RID, TYPE_SIGNAL:
			_evidence_error = "Unsupported native handle encountered in raw trace: type " + str(typeof(value))
			return {"UNSUPPORTED_NATIVE_HANDLE": _evidence_error}
		_: return value


func _sha256_bytes(bytes: PackedByteArray) -> String:
	var digest := HashingContext.new()
	if digest.start(HashingContext.HASH_SHA256) != OK or digest.update(bytes) != OK: return ""
	return digest.finish().hex_encode()


func _write_raw(path: String, bytes: PackedByteArray) -> Dictionary:
	var expected_sha: String = _sha256_bytes(bytes)
	var file: FileAccess = FileAccess.open(path, FileAccess.WRITE)
	if file == null: return {"accepted": false, "reason": "raw native trace open failed"}
	file.store_buffer(bytes); file.flush()
	var io_ok: bool = file.get_error() == OK
	file.close()
	var observed: PackedByteArray = FileAccess.get_file_as_bytes(path)
	var observed_sha: String = _sha256_bytes(observed)
	var roundtrip: bool = var_to_bytes(bytes_to_var(observed)) == bytes
	return {"accepted": io_ok and not expected_sha.is_empty() and observed == bytes and observed_sha == expected_sha and roundtrip, "path": path, "bytes": bytes.size(), "sha256": expected_sha, "readback_sha256": observed_sha, "native_binary_roundtrip_exact": roundtrip, "encoding": "Godot var_to_bytes pure-data binary; Vector/Basis/Transform/Rect/float/int types retained exactly; explicit binding-reference metadata, never serialized Object/Callable handles"}


func _finish() -> void:
	if _finished: return
	# Latch before cleanup; any waiting coroutine must check before use.
	_finished = true
	_close_component()
	if node_added.is_connected(_observe_node): node_added.disconnect(_observe_node)
	if is_instance_valid(_probe): _probe.free()
	_probe = null
	paused = false
	var raw_payload: Dictionary = {"api_revision": "TEST-a2-l4-native-comparison-trace-1", "source_sha256": FileAccess.get_sha256(get_script().resource_path), "embedded_host_sha256": FOCUS_HOST_SHA256, "embedded_game_sha256": FOCUS_GAME_SHA256, "components": _components, "source_observations": _source_observations, "checks_detail": _rows, "explicit_native_binding_reference_count": _explicit_binding_refs, "handle_policy": "Each ephemeral handle becomes explicit observation-time identity/physical binding data. Native geometric and scalar values remain their original Godot types. No Object/Callable/RID is serialized as a live handle. This is not a save/restore codec."}
	var raw_receipt: Dictionary = {"accepted": false, "reason": "report directory unavailable"}
	if _report_owned:
		var bytes: PackedByteArray = var_to_bytes(raw_payload)
		if bytes.size() <= 128 * 1024 * 1024:
			raw_receipt = _write_raw(_report_path.get_base_dir().path_join("native-trace.bin"), bytes)
		else:
			raw_receipt = {"accepted": false, "reason": "finite trace exceeded 128MiB persistence budget; no paths truncated and no positive result", "bytes": bytes.size()}
	var difference: bool = _original_success and _candidate_success and _counterfactual_intersects and _cleanup_ok and _evidence_error.is_empty() and _failures == 0 and raw_receipt.get("accepted", false) and _verdict == "supported_fixed_response_difference"
	var report: Dictionary = {"api_revision": "TEST-a2-l4-fixed-comparison-1", "inert_source_origin": true, "source_sha256": FileAccess.get_sha256(get_script().resource_path), "embedded_host_sha256": FOCUS_HOST_SHA256, "embedded_game_sha256": FOCUS_GAME_SHA256, "original_actual_success": _original_success, "candidate_actual_success": _candidate_success, "original_stationary_counterfactual_executed": false, "fixed_response_difference_supported": difference, "campaign_or_policy_acceptance_claimed": false, "normal_cleanup_completed": _cleanup_ok, "native_evidence_error": _evidence_error, "verdict": _verdict, "reason": _reason, "checks": _checks, "failures": _failures, "wall_ms": Time.get_ticks_msec() - _start_ms, "raw_native_trace": raw_receipt}
	var persisted: bool = false
	if _report_owned:
		var wire: String = JSON.stringify(report, "\t", true, true) + "\n"
		var file: FileAccess = FileAccess.open(_report_path, FileAccess.WRITE)
		if file != null:
			file.store_string(wire); file.flush()
			var io_ok: bool = file.get_error() == OK
			file.close()
			persisted = io_ok and FileAccess.get_file_as_string(_report_path) == wire and FileAccess.get_sha256(_report_path) == wire.sha256_text()
	if not persisted or not raw_receipt.get("accepted", false): push_error("A2-L4 fixed comparison summary/raw trace not safely persisted; no success exit")
	print("A2-L4 fixed comparison: verdict=%s original_actual=%s candidate_actual=%s fixed_difference=%s checks=%d failures=%d summary_persisted=%s raw_persisted=%s; no campaign/policy acceptance" % [_verdict, str(_original_success), str(_candidate_success), str(difference), _checks, _failures, str(persisted), str(raw_receipt.get("accepted", false))])
	quit(0 if difference and persisted else 2 if persisted and raw_receipt.get("accepted", false) and _evidence_error.is_empty() and _failures == 0 else 1)
