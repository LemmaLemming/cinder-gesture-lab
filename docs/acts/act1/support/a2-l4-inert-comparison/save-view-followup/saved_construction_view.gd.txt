func _flush_raw(label: String,cleanup_pending: bool = true) -> bool:
	if not _output_owned:

	var bytes: PackedByteArray = var_to_bytes(payload)
	if bytes.size() > 128 * 1024 * 1024:
		_persist_ok = false
		return _expect(false,"Full native raw evidence exceeds finite128MiB budget; never truncate paths or Store data")

	var summary: Dictionary = {"scope":"Raw receipt is provisional while cleanup_pending; no acceptance or reconstructed outcome","label":label,"cleanup_pending":cleanup_pending,"checks":_checks,"failures":_failures,"complete":_complete,"receipt":receipt,"evidence_error":_evidence_error}

	receipt["summary_path"] = summary_path
	receipt["summary_sha256"] = FileAccess.get_sha256(summary_path) if summary_ok else ""

	_checkpoints.append(receipt)

func _retain_disk() -> Dictionary:
	var disk: Dictionary = _disk()
	var retained: Dictionary = {}
	for name: String in FILES:
		var slot: Dictionary = disk[name]
		if not slot.exists:
			retained[name] = {"exists":false,"sha256":""}
			continue
		var sha: String = slot.sha256
		var path: String = _output.path_join("raw-Store-"+sha+".txt")
		var ok: bool = not sha.is_empty() and slot.raw.sha256_text() == sha
		if _store_artifacts.has(sha):
			ok = ok and FileAccess.get_file_as_string(path) == slot.raw and FileAccess.get_sha256(path) == sha
		else:
			ok = ok and _write_text_once(path,slot.raw)
			_store_artifacts[sha] = {"path":path,"sha256":sha,"bytes":slot.raw.to_utf8_buffer().size(),"raw_byte_readback_exact":ok,"transport":"Literal actual generated raw Store bytes, not historical archive generation"}
		var receipt: Dictionary = _store_artifacts[sha].duplicate(true)
		receipt["exists"] = true
		receipt["current_readback_exact"] = ok
		if not ok:
			receipt["unpersisted_full_raw"] = slot.raw
			_persist_ok = false
		retained[name] = receipt
	return retained

func _write_text_once(path: String,wire: String) -> bool:
	if not _output_owned or not path.begins_with(_output+"/") or FileAccess.file_exists(path): return false
	var file: FileAccess = FileAccess.open(path,FileAccess.WRITE)
	if file == null: return false
	file.store_string(wire); file.flush()
	var io_ok: bool = file.get_error() == OK
	file.close()
	return io_ok and FileAccess.get_file_as_string(path) == wire and FileAccess.get_sha256(path) == wire.sha256_text()

func _validation_event(kind: String,node: Node,payload: Dictionary) -> Dictionary:
	if _validation_spans.is_empty(): return {"allowed":false,"reason":"No synchronous validation disposal span"}
	var span: Dictionary = _validation_spans.back()
	if not span.eligible or kind != "ray-state": return {"allowed":false,"ticket":span.ticket,"reason":"Only eligible pristine Ray idle-clear state can qualify"}
	var owner: CinderCampaignShell = (span.owner as WeakRef).get_ref() as CinderCampaignShell
	var candidate: Dictionary = {}
	for key: String in span.candidate:
		candidate[key] = (span.candidate[key] as WeakRef).get_ref()
	if not is_instance_valid(owner) or owner != _shell or not _validation_map_member(owner,candidate): return {"allowed":false,"ticket":span.ticket,"reason":"Exact native validation-map provenance or same Shell lost"}
	var level: CinderLevel = candidate.level
	var hero: CinderPlayer = candidate.player
	for key: String in candidate:
		var witness: Dictionary = span.metadata.candidate_identities[key]
		var script_id: int = candidate[key].get_script().get_instance_id() if candidate[key].get_script() is Script else 0
		if not candidate[key].is_inside_tree() or candidate[key].is_queued_for_deletion() or str(candidate[key].get_instance_id()) != witness.id or script_id != int(witness.script_id): return {"allowed":false,"ticket":span.ticket,"reason":"Frozen tuple Node/Script identity changed"}
	var driver: Node = (span.exchange as WeakRef).get_ref() as Node
	var scheduler: Node = (span.scheduler as WeakRef).get_ref() as Node
	if not is_instance_valid(driver) or not is_instance_valid(scheduler) or driver != level.get("_exchange") or scheduler != level.get("_scheduler") or driver.get_parent() != level or scheduler.get_parent() != level or driver.get_script().get_instance_id() != int(span.exchange_script_id) or scheduler.get_script().get_instance_id() != int(span.scheduler_script_id): return {"allowed":false,"ticket":span.ticket,"reason":"Frozen native driver/scheduler identities or parent bindings changed"}
	var id: String = String(payload.get("id",""))
	if not is_instance_valid(node) or node != driver or not span.rays.has(id) or driver.get_script().resource_path != "res://scripts/acts/act2/ray_scout_exchange.gd": return {"allowed":false,"ticket":span.ticket,"reason":"Unknown or changed Ray emitter/member"}

func _validation_after(owner: CinderCampaignShell,ticket: int) -> void:
	if _validation_spans.is_empty():
		_expect(false,"Validation disposal returned without observation span")
		return
	var span: Dictionary = _validation_spans.pop_back()
	var exact: bool = span.ticket == ticket and (span.owner as WeakRef).get_ref() == owner
	var released: bool = true
	for ref: WeakRef in span.subtree:
		if ref.get_ref() != null: released = false
	_validation_records.append({"boundary":"after_unchanged_synchronous_super_dispose","ticket":ticket,"matching_span":exact,"eligible_before":span.eligible,"all_original_subtree_weakrefs_released":released,"captured_descendants":span.subtree.size(),"actual_qualified_ray_ids":span.seen_ray_ids.duplicate()})
	if span.eligible:
		if not _expect(exact and released,"Exact synchronous pristine validation disposal normally releases its original subtree") and _watch: _forbidden.append("validation-disposal-custody")

func _pure_native_data(value: Variant) -> Variant:
	# Native geometric/timing types remain native in the binary trace. Only
	# ephemeral handles become EXPLICIT observation-time identity/binding data;
	# these records cannot reconstruct or authorize a Node/Resource/Callable.
	match typeof(value):
		TYPE_DICTIONARY:
			var result: Dictionary = {}
			for key: Variant in value:
				if typeof(key) in [TYPE_OBJECT,TYPE_CALLABLE,TYPE_RID,TYPE_SIGNAL]:
					_evidence_error = "Unsupported native handle key in evidence"
					return {"UNSUPPORTED_NATIVE_HANDLE_KEY":_evidence_error}
				result[key] = _pure_native_data(value[key])
			return result
		TYPE_ARRAY:
			var result: Array = []
			for child: Variant in value: result.append(_pure_native_data(child))
			return result
		TYPE_OBJECT:
			_explicit_binding_refs += 1
			if not is_instance_valid(value):
				_evidence_error = "Native binding released before observation-time projection"
				return {"UNSUPPORTED_NATIVE_REFERENCE": _evidence_error}
			if not value is CinderPlayer and not value is StaticBody3D and not value is CollisionShape3D and not value is BoxShape3D and not value is CapsuleShape3D:
				_evidence_error = "Unrecognized native Object binding: " + value.get_class()
				return {"UNSUPPORTED_NATIVE_REFERENCE": _evidence_error}
			var binding: Dictionary = {"EXPLICIT_NATIVE_BINDING_REFERENCE": value.get_class(), "instance_id": value.get_instance_id(), "reconstructible": false}
			if value is Node:
				binding["path"] = String(value.get_path()) if value.is_inside_tree() else "outside-tree"
				var script: Script = value.get_script() as Script
				binding["script_path"] = script.resource_path if script != null else ""
				if value is Node3D:
					binding["actual_global_transform"] = value.global_transform
					binding["world_instance_id"] = value.get_world_3d().get_instance_id() if value.is_inside_tree() else 0
				if value is CollisionObject3D:
					binding["collision_layer"] = value.collision_layer
					binding["collision_mask"] = value.collision_mask
				if value is CollisionShape3D:
					binding["disabled"] = value.disabled
					binding["actual_shape"] = _pure_native_data(value.shape)
			elif value is Resource:
				binding["resource_path"] = value.resource_path
				if value is BoxShape3D: binding["actual_box_size"] = value.size
				elif value is CapsuleShape3D:
					binding["actual_capsule_radius"] = value.radius
					binding["actual_capsule_height"] = value.height
			return binding
		TYPE_CALLABLE, TYPE_RID, TYPE_SIGNAL:
			_evidence_error = "Unsupported native handle encountered in raw trace: type " + str(typeof(value))
			return {"UNSUPPORTED_NATIVE_HANDLE": _evidence_error}
		_: return value

func _sha256_bytes(bytes: PackedByteArray) -> String:
	var digest := HashingContext.new()
	if digest.start(HashingContext.HASH_SHA256) != OK or digest.update(bytes) != OK: return ""
	return digest.finish().hex_encode()

func _write_raw(path: String, bytes: PackedByteArray) -> Dictionary:
	if not _output_owned or not path.begins_with(_output+"/") or FileAccess.file_exists(path): return {"accepted":false,"reason":"Refuse overwrite or unowned raw path"}
	var expected_sha: String = _sha256_bytes(bytes)
	var file: FileAccess = FileAccess.open(path, FileAccess.WRITE)
	if file == null: return {"accepted": false, "reason": "raw native trace open failed"}
	file.store_buffer(bytes); file.flush()
	var io_ok: bool = file.get_error() == OK
	file.close()
	var observed: PackedByteArray = FileAccess.get_file_as_bytes(path)
	var observed_sha: String = _sha256_bytes(observed)
	var roundtrip: bool = var_to_bytes(bytes_to_var(observed)) == bytes
	return {"accepted": io_ok and not expected_sha.is_empty() and observed == bytes and observed_sha == expected_sha and roundtrip, "path": path, "bytes": bytes.size(), "sha256": expected_sha, "readback_sha256": observed_sha, "native_binary_roundtrip_exact": roundtrip, "encoding": "Godot var_to_bytes pure-data binary; Vector/Basis/Transform/Rect/float/int types retained exactly; explicit binding-reference metadata, never serialized Object/Callable handles"}
