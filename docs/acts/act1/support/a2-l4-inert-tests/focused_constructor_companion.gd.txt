func _positions_equal(actual: Dictionary, expected: Dictionary) -> bool:
	if actual.size() != expected.size(): return false
	for id: String in expected:
		if not actual.has(id) or not actual[id] is Vector3 or actual[id] != expected[id]: return false
	return true

func _restore_case(tuple: Dictionary, newer_preference: String, label: String, legacy: bool) -> bool:
	if not _new_game(newer_preference): return false
	# The initial view's separate fresh recipient is retired normally. The
	# saved candidate is a NEW level, never a relocated/reconfigured donor.
	var fresh: CinderLevel = _game.get("active_level") as CinderLevel
	var fresh_ref: WeakRef = weakref(fresh)
	fresh.exit_level()
	fresh.free()
	_game.set("active_level", null)
	_expect(fresh_ref.get_ref() == null, "native-cleanup", "fresh bootstrap recipient retires before saved candidate " + label)
	var scene: PackedScene = load(LEVEL_SCENE) as PackedScene
	var candidate: CinderLevel = scene.instantiate() as CinderLevel
	var world: Node3D = _game.get("world") as Node3D
	world.add_child(candidate)
	_game.set("active_level", candidate)
	_owned_levels.append(weakref(candidate))
	var hero: CinderPlayer = _game.get("player") as CinderPlayer
	var effects: PixelEffects = _game.get("fx") as PixelEffects
	if not await _settle(): return false
	var source_wire: String = _exact.stringify(tuple)
	_events.clear(); _watch_events = true
	_entry_observations.clear(); _observe_entries = true
	var entered: bool = candidate.enter_restore_candidate(hero, effects, _game, tuple.level, tuple.player)
	_observe_entries = false
	if not _expect(entered, "native-saved", "saved choice precedes native construction under opposite preference " + label + ": " + candidate.last_snapshot_error): return false
	var selected: Dictionary = _construction.saved(tuple.level.local)
	if not _native_receipt(candidate, selected, label): return false
	_expect(selected.legacy == legacy and String(_game.call("get_difficulty_preference")) == newer_preference, "native-saved", "saved construction identity survives newer preference " + label)
	for observation: Dictionary in _entry_observations:
		_expect(observation.frozen_before_entry and observation.unconfigured_at_entry and observation.selected_root_at_entry and observation.full_profile_already_selected, "native-order-observation", "saved freeze/root/full profile precedes actor entry " + label + "/" + String(observation.id))
	_expect(_entry_observations.size() == 9, "native-saved", "saved candidate creates nine distinct native actors " + label)
	if not _expect(hero.snapshot_error(tuple.player).is_empty() and candidate.snapshot_error_with_player(tuple.level, tuple.player).is_empty(), "native-saved", "whole native Player/local prevalidation " + label + ": " + candidate.last_snapshot_error): return false
	# Exact public Player first, then Level quiet commit; no direct field edit.
	if not _expect(hero.restore_state(tuple.player) and candidate.restore_state(tuple.level), "native-saved", "Player-first exact quiet restoration " + label + ": " + candidate.last_snapshot_error): return false
	var recaptured: Dictionary = {"player": hero.snapshot_state(), "level": candidate.snapshot_state()}
	_expect(_exact.stringify(recaptured) == source_wire and not recaptured.player.is_empty() and not recaptured.level.is_empty(), "native-saved", "exact actual tuple recapture preserves format, HP/history/clocks/leases " + label)
	_expect((recaptured.level.local.size() == 16 and not recaptured.level.local.has("arrangement")) if legacy else (recaptured.level.local.size() == 17 and recaptured.level.local.has("arrangement")), "native-saved", "literal writer format survives recapture " + label)
	_expect(_events.is_empty() and _exact.stringify(tuple) == source_wire and paused and candidate.is_restore_candidate(), "native-saved", "paused quiet restoration emits no observed action/damage/gear/progression and leaves input source unchanged " + label)
	_watch_events = false
	_cleanup_game()
	return _failures == 0


func _malformed_native_case() -> bool:

	var fresh: CinderLevel = _game.get("active_level") as CinderLevel

	_watch_events = false
	_cleanup_game()
	return _failures == 0


func _legacy_case() -> bool:
	if not _expect(FileAccess.get_sha256(LEGACY_ATTEMPTS_PATH) == LEGACY_ATTEMPTS_SHA, "legacy-input", "read exact named literal Standard Attempts artifact; never rewritten"): return false
	var raw: String = FileAccess.get_file_as_string(LEGACY_ATTEMPTS_PATH)
	var decoded: Dictionary = _exact.parse(raw)
	if not _expect(decoded.get("accepted", false) and decoded.get("value") is Dictionary, "legacy-input", "literal archived Attempts exact decode: " + String(decoded.get("reason", ""))): return false
	var state: Dictionary = decoded.value
	if not _expect(state.get("story") is Dictionary and state.story.get("checkpoint") is Dictionary, "legacy-input", "named archive contains actual earlier checkpoint"): return false
	var unit: Dictionary = state.story.checkpoint
	if not _expect(unit.get("level_id") == "A2-L4" and unit.get("player") is Dictionary and unit.get("level") is Dictionary and unit.level.get("local") is Dictionary and unit.level.local.size() == 16 and not unit.level.local.has("arrangement") and unit.level.local.profile_id == "standard" and unit.level.local.sequence.stage_index == 7 and unit.level.local.sequence.defeated_ids.size() == 7, "legacy-input", "genuine old16 Standard earlier Putney checkpoint, not invented Challenge/progress"): return false
	if not await _restore_case({"player": unit.player.duplicate(true), "level": unit.level.duplicate(true)}, "challenge", "literal archived old16 Standard checkpoint", true): return false
	var clear: Dictionary = state.story.get("snapshot", {})
	if not _expect(clear.get("level_id") == "A2-L4" and clear.get("player") is Dictionary and clear.get("level") is Dictionary and clear.level.get("local") is Dictionary and clear.level.local.size() == 16 and clear.level.local.profile_id == "standard" and clear.level.local.sequence.stage_index == 8 and clear.level.local.sequence.defeated_ids.size() == 9 and clear.level.local.scheduler.profile.is_empty() and clear.level.local.scheduler.encounter_id.is_empty(), "legacy-input", "same unchanged archive contains genuine committed quiet clear with retired Scheduler"): return false
	_legacy_complete = await _restore_case({"player": clear.player.duplicate(true), "level": clear.level.duplicate(true)}, "challenge", "literal archived old16 Standard quiet clear", true)
	_expect(FileAccess.get_file_as_string(LEGACY_ATTEMPTS_PATH) == raw and FileAccess.get_sha256(LEGACY_ATTEMPTS_PATH) == LEGACY_ATTEMPTS_SHA, "legacy-input", "exact literal archive bytes preserved")
	return _legacy_complete and _failures == 0


func _native_receipt(level: CinderLevel, selected: Dictionary, label: String) -> bool:
	var before: int = _failures
	var retained: Dictionary = level.get("_l4_construction")
	_expect(retained.is_read_only() and (retained.profile as Dictionary).is_read_only() and (retained.positions as Dictionary).is_read_only(), "native-constructor", "frozen retained selection " + label)
	_expect(_positions_equal(retained.positions, selected.positions) and _exact.stringify(retained.profile) == _exact.stringify(selected.profile) and String(level.get("_profile_id")) == selected.profile_id, "native-constructor", "actual frozen map/exact eight-field profile " + label)
	var scheduler: Node = level.get("_scheduler")
	_expect(is_instance_valid(scheduler) and _exact.stringify(scheduler.call("encounter_profile")) == _exact.stringify(selected.profile), "native-constructor", "native Scheduler joins exact selected full profile before actor configure " + label)
	var actors: Dictionary = level.get("_actors")
	_expect(actors.size() == 9 and (level.get("_mechanisms") as Dictionary).size() == 4 and (level.get("_clouds") as Dictionary).size() == 2, "native-constructor", "all original native recipients retained " + label)
	for id: String in selected.positions:
		var actor: Node3D = actors.get(id) as Node3D
		_expect(is_instance_valid(actor) and actor.get_parent() == level and actor.global_position == selected.positions[id] and actor.get("_authored_root_position") == selected.positions[id] and bool(actor.get("_configured")), "native-constructor", "native actor root and own immutable configure receipt " + label + "/" + id)
	var tools: Dictionary = level.get("_mechanisms")
	for id: String in tools:
		var tool: Node3D = tools[id]
		var target: String = id.trim_prefix("tool_")
		var configuration: Dictionary = tool.get("_configuration")
		var at: Vector3 = selected.positions[target]
		_expect(tool.global_position == at and configuration.opening_position == at and configuration.geometry == {"kind": "lane", "from": at, "to": at + Vector3.BACK * 3.8, "radius": 0.31}, "native-constructor", "tool source/opening/native lane follows same selected root " + label + "/" + id)
	var exchange: Node = level.get("_exchange")
	var ray_anchors: Dictionary = exchange.get("_anchors")
	var presentation_guard: Callable = exchange.get("_presentation_guard")
	var context_provider: Callable = exchange.get("_context_provider")
	_expect(bool(exchange.get("_guard_required")) and bool(exchange.get("_provider_required")) and presentation_guard.is_valid() and context_provider.is_valid(), "native-constructor", "original Ray context/presentation guards remain bound " + label)
	for id: String in _sequence.RAYS:
		_expect(ray_anchors.get(id) == selected.positions[id] and (exchange.get("_actors") as Dictionary).get(id) == actors[id], "native-constructor", "Ray immutable anchor and actual actor binding " + label + "/" + id)
	var banks: Dictionary = level.get("_clouds")
	var bank_to_tender: Dictionary = {"flood_bank": "flood_tender", "villa_bank": "villa_tender"}
	for id: String in bank_to_tender:
		var bank: Node3D = banks[id]
		var configuration: Dictionary = bank.get("_configuration")
		_expect(bank.global_position == _sequence.CLOUDS[id] and configuration.opening_position == selected.positions[bank_to_tender[id]] and configuration.geometry == {"kind": "circle", "origin": _sequence.CLOUDS[id], "radius": 1.05}, "native-constructor", "bank native source/radius and actual Tender opening retained " + label + "/" + id)
	return _failures == before


func _native_ids(level: CinderLevel) -> Dictionary:
	var result: Dictionary = {}
	for key: String in ["_actors", "_mechanisms", "_clouds"]:
		for id: String in level.get(key):
			var node: Node = (level.get(key) as Dictionary)[id]
			result[key + ":" + id] = node.get_instance_id()
	for key: String in ["_scheduler", "_exchange", "_bank_guard"]:
		var node: Node = level.get(key)
		result[key] = node.get_instance_id() if is_instance_valid(node) else 0
	return result


func _observe_node_added(node: Node) -> void:
	if node is CinderLevel:
		node.checkpoint_requested.connect(func(_id: String, _checkpoint: String, _kind: String) -> void: _event("checkpoint"))
		node.completion_requested.connect(func(_id: String, _completion: String) -> void: _event("completion"))
		node.contact_exit_requested.connect(func(_id: String, _exit: String) -> void: _event("exit"))
	if node is CinderPlayer:
		node.world_action_executed.connect(func(_record: Dictionary) -> void: _event("world-action"))
		node.died.connect(func() -> void: _event("player-death"))
		node.fired.connect(func(_kind: String) -> void: _event("player-fired"))
		node.equipment_changed.connect(func(_id: String) -> void: _event("gear-changed"))
		node.action_resolved.connect(func(_kind: String, _hits: int, _damage: float) -> void: _event("action-resolved"))
	var source: Script = node.get_script() as Script
	var source_path: String = source.resource_path if source != null else ""
	if source_path == "res://scripts/combat/lane_mechanism.gd":
		node.connect("hit_resolved", func(_hero_id: String, _cycle: int, _result: Dictionary) -> void: _event("lane-hit"))
	elif source_path == "res://scripts/acts/act2/ray_scout_exchange.gd":
		node.connect("hit_resolved", func(_actor_id: String, _result: Dictionary) -> void: _event("ray-hit"))
	elif source_path == "res://scripts/combat/smoke_bank.gd":
		node.connect("tick_resolved", func(_hero_id: String, _cycle: int, _result: Dictionary) -> void: _event("bank-tick"))
	if node.has_signal("defeated"): node.connect("defeated", func(_id: String) -> void: _event("target-defeat"))
	if not _observe_entries or not String(node.name).begins_with("LondonApproaches_") or not node is Node3D: return
	var parent: CinderLevel = node.get_parent() as CinderLevel
	if not is_instance_valid(parent): return
	var id: String = String(node.name).trim_prefix("LondonApproaches_")
	var receipt: Dictionary = parent.get("_l4_construction")
	var scheduler: Node = parent.get("_scheduler")
	var positions: Dictionary = receipt.get("positions", {})
	_entry_observations.append({
		"id": id,
		"frozen_before_entry": receipt.is_read_only() and (positions as Dictionary).is_read_only(),
		"unconfigured_at_entry": not bool(node.get("_configured")),
		"selected_root_at_entry": positions.has(id) and (node as Node3D).global_position == positions[id],
		"full_profile_already_selected": receipt.get("profile") is Dictionary and receipt.profile.size() == 8 and is_instance_valid(scheduler) and _exact.stringify(scheduler.call("encounter_profile")) == _exact.stringify(receipt.profile),
	})


func _event(label: String) -> void:
	if _watch_events: _events.append(label)


func _settle() -> bool:
	for frame: int in range(2):
		await process_frame
		if _finished: return false
	return true


func _cleanup_game() -> void:
	_observe_entries = false
	_watch_events = false
	if is_instance_valid(_game):
		var level: CinderLevel = _game.get("active_level") as CinderLevel
		if is_instance_valid(level): level.exit_level()
		_game.free()
	_game = null
	for reference: WeakRef in _owned_levels:
		_expect(reference.get_ref() == null, "native-cleanup", "normal attached exit/free releases each owned native recipient")
	_owned_levels.clear()


func _watchdog() -> void:
	await create_timer(TIMEOUT_S, true, false, true).timeout
	if _finished: return
	_expect(false, "watchdog", "bounded companion incomplete at wall-clock deadline; no success credit")
	_finish(false)


func _expect(condition: bool, scope: String, detail: String) -> bool:
	_checks += 1
	if not condition: _failures += 1
	_rows.append({"scope": scope, "detail": detail, "passed": condition})
	if not condition: push_error("A2-L4 focused constructor: " + scope + ": " + detail)
	return condition


func _finish(reached_end: bool) -> void:
	if _finished: return
	_finished = true
	_cleanup_game()
	if node_added.is_connected(_observe_node_added): node_added.disconnect(_observe_node_added)
	paused = false
	var complete: bool = reached_end and _format_complete and _native_complete
	var success: bool = complete and _failures == 0
	var report: Dictionary = {
		"api_revision": "inert-a2-l4-constructor-companion-1", "source_only_origin": true,
		"companion_source_sha256": FileAccess.get_sha256(get_script().resource_path),
		"checks": _checks, "failures": _failures, "reached_end": reached_end,
		"mandatory_suite_complete": complete, "bounded_mandatory_checks_pass": success,
		"pure_format_suite_complete": _format_complete,
		"native_fresh_new_restore_suite_complete": _native_complete,
		"native_literal_standard_checkpoint_and_clear_checked": _legacy_complete,
		"native_literal_challenge_restore_checked": false,
		"native_saved_canonical_challenge_checked": false,
		"overall_campaign_or_policy_acceptance_claimed": false,
		"wall_ms": Time.get_ticks_msec() - _start_ms, "limits": _limits,
		"source_observations": _source_observations, "results": _rows,
	}
	var persisted: bool = false
	if _report_owned:
		var wire: String = JSON.stringify(report, "\t") + "\n"
		var file: FileAccess = FileAccess.open(_report_path, FileAccess.WRITE)
		if file != null:
			file.store_string(wire)
			file.flush()
			var io_ok: bool = file.get_error() == OK
			file.close()
			persisted = io_ok and FileAccess.get_file_as_string(_report_path) == wire
	if not persisted: push_error("A2-L4 companion result was not safely persisted; no PASS")
	print("A2-L4 constructor companion: checks=%d failures=%d mandatory_complete=%s legacy_standard=%s persisted=%s; no combat, Shell transport or policy acceptance" % [_checks, _failures, str(complete), str(_legacy_complete), str(persisted)])
	quit(0 if success and persisted else 1)
