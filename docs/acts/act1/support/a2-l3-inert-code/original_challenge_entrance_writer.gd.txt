func _run() -> void:

func _arguments() -> bool:
	var seen: Dictionary = {}
	for argument: String in OS.get_cmdline_user_args():
		var key: String = argument.get_slice("=", 0)
		if key not in ["--case", "--output-root", "--fixture-sha256"] or seen.has(key):
			return _expect(false, "only unique declared selectors are accepted: " + argument)
		seen[key] = true
		if key == "--case" and argument != "--case=" + CASE: return _expect(false, "one original entrance case only")
		if key == "--output-root": output_root = argument.trim_prefix("--output-root=").trim_suffix("/")
		if key == "--fixture-sha256": expected_fixture_hash = argument.trim_prefix("--fixture-sha256=")
	if not _expect(seen.size() == 3 and output_root.begins_with("res://.cinder/") and not ".." in output_root and _hash_string(expected_fixture_hash), "explicit case, new ignored output root and executable-text hash required"): return false
	fixture_path = get_script().resource_path
	fixture_hash = FileAccess.get_sha256(fixture_path)
	return _expect(fixture_hash == expected_fixture_hash and not fixture_hash.is_empty(), "actual materialized writer bytes match externally supplied reviewed hash")

func _check_sources(destination: Dictionary) -> bool:

func _pristine(snapshot: Dictionary) -> bool:

	if not _expect(local.sequence.stage_index == 0 and local.sequence.defeated_ids.is_empty() and local.sequence.crossed_contacts.is_empty() and not snapshot.level.progress.completed and snapshot.level.progress.checkpoint_id.is_empty(), "untouched road entrance has no earned boss/prefix/contact/completion history"): return false
	if not _expect(float(local.scheduler.clock_s) == 0.0 and local.scheduler.profile.id == "challenge" and local.scheduler.reservations.is_empty() and float(local.scenic_clock) == 0.0 and int(local.last_action_sequence) == 0, "original actual epoch is pristine with zero clocks/leases/actions"): return false
	for packet: Dictionary in local.mechanisms.values():
		if not _expect(packet.status == "idle" and packet.cycle == 0 and packet.exchange.is_empty(), "all actual original tools remain unstarted"): return false
	for packet: Dictionary in local.banks.values():
		if not _expect(packet.status == "idle" and packet.cycle == 0 and packet.exchange.is_empty(), "all actual original smoke banks remain unstarted"): return false
	for id: String in local.targets:
		var actor: Dictionary = local.targets[id].actor if id == "handling_machine" else local.targets[id]
		if not _expect(float(actor.hp) == 30.0, "original entrance target retains native initial HP: " + id): return false
	var boss: Dictionary = local.targets.handling_machine
	return _expect(boss.boss_phase == 1 and not boss.transition_pending and boss.tool_action == "reach" and local.rays.records.apron_scout.cycle == 0 and float(local.rays.actors.apron_scout.hp) == 30.0 and int(snapshot.player.world_actions.sequence) == 0 and snapshot.player.resources.hp == snapshot.player.resources.max_hp and snapshot.player.resources.shells == snapshot.player.resources.max_shells and not snapshot.player.resources.dead, "actual initial boss/ray/Player remain untouched")

func _archive_stage(label: String, expected_payload: String, expected_snapshot: String) -> bool:
	var destination: String = output_root + "/" + label
	if not _expect(not DirAccess.dir_exists_absolute(ProjectSettings.globalize_path(destination)), "new branch directory: " + label): return false
	if not _expect(DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(destination)) == OK, "create branch directory: " + label): return false
	var store: CinderSaveStore = Store.new(user_root + "campaign.json")
	var model: CinderCampaignAttempts = Attempts.new(shell.registry)
	store.payload_validator = Callable(model, "saved_payload_error")
	var payload: Dictionary = store.read_payload()
	if not _expect(not payload.is_empty() and not store.loaded_backup and _wire(payload, label + " literal Store payload") == expected_payload, "primary Store decodes to the exact whole Attempts: " + label): return false
	var row: Dictionary = {"label": label, "generation": store.generation, "exact_attempts": expected_payload, "exact_snapshot": _wire(payload.story.snapshot, label + " current"), "exact_checkpoint": _wire(payload.story.checkpoint, label + " checkpoint"), "raw_files": []}
	if not _expect(row.exact_snapshot == expected_snapshot and row.exact_checkpoint == expected_snapshot, "current/checkpoint exactly join actual capture: " + label): return false
	for name: String in COMPANIONS:
		var source: String = user_root + name
		var item: Dictionary = {"name": name, "original_path": source, "present": FileAccess.file_exists(source)}
		if item.present:
			var bytes: PackedByteArray = FileAccess.get_file_as_bytes(source)
			if not _expect(not bytes.is_empty() and bytes.size() <= Store.MAX_FILE_BYTES, "literal companion bytes bounded: " + name): return false
			var copied: String = destination + "/" + name
			if not _write_new_bytes(copied, bytes): return false
			item["path"] = copied
			item["bytes"] = bytes.size()
			item["sha256"] = _sha256(bytes)
			if not _expect(FileAccess.get_file_as_bytes(source) == bytes, "source companion stable through exact byte copy: " + name): return false
		else:
			if not _expect(not DirAccess.dir_exists_absolute(ProjectSettings.globalize_path(source)), "absent companion is not a directory: " + name): return false
		row.raw_files.append(item)
	if not _write_new_bytes(destination + "/attempts.exact.json", expected_payload.to_utf8_buffer()): return false
	if not _write_new_bytes(destination + "/snapshot.exact.json", expected_snapshot.to_utf8_buffer()): return false
	if not _write_new_bytes(destination + "/checkpoint.exact.json", String(row.exact_checkpoint).to_utf8_buffer()): return false
	if not _write_new_bytes(destination + "/manifest.json", JSON.stringify(row, "\t").to_utf8_buffer()): return false
	stages.append(row)
	return true

func _wire(value: Variant, label: String) -> String:
	var encoded: String = Exact.stringify(value)
	if not _expect(not encoded.is_empty(), "nonempty exact wire: " + label): return ""
	var parsed: Dictionary = Exact.parse(encoded)
	if not _expect(parsed.get("accepted", false) and Exact.stringify(parsed.get("value")) == encoded, "typed exact round trip: " + label): return ""
	return encoded

func _write_new_bytes(path: String, bytes: PackedByteArray) -> bool:
	if not _expect(not FileAccess.file_exists(path), "write never overwrites evidence: " + path): return false
	var file: FileAccess = FileAccess.open(path, FileAccess.WRITE)
	if not _expect(file != null, "open new evidence: " + path): return false
	file.store_buffer(bytes)
	file.flush()
	var error: Error = file.get_error()
	file.close()
	return _expect(error == OK and FileAccess.get_file_as_bytes(path) == bytes, "literal evidence bytes verified: " + path)

func _settle() -> bool:
	var stable: int = 0
	while Time.get_ticks_msec() < deadline_ms:
		await process_frame
		if not is_instance_valid(shell): return _expect(false, "Shell remains alive during bounded operation")
		if not shell.campaign_error.is_empty(): return _expect(false, "Shell operation: " + shell.campaign_error)
		if _idle(): stable += 1
		else: stable = 0
		if stable >= 8: return _expect(paused, "operation settles while actual simulation remains paused")
	return _expect(false, "120s bounded writer operation deadline")

func _idle() -> bool:
	return is_instance_valid(shell) and not shell.get("_draining") and not shell.get("_drain_queued") and not shell.get("_registration_inflight") and not shell.get("_registration_pending") and shell.get("_operations").is_empty()

func _release_shell() -> bool:
	if not is_instance_valid(shell): return true
	var limit: int = Time.get_ticks_msec() + 2000
	while not _idle() and Time.get_ticks_msec() < limit: await process_frame
	if not _expect(_idle(), "cleanup waits for async Shell operation to close; never frees its pending coroutine"): return false
	var old: WeakRef = weakref(shell)
	if not candidate.is_empty():
		shell.call("_dispose", candidate)
		candidate.clear()
	if is_instance_valid(shell.active_level): shell.active_level.exit_level()
	shell.free()
	shell = null
	return _expect(old.get_ref() == null, "ordinary attached-level exit releases old Shell subtree")

func _archive_partial_files() -> void:
	# Preserve whatever original companions really exist after an incomplete run.
	# This is neither a completed branch nor a recovered/reconstructed Store.
	if not owns_output_root or user_root.is_empty(): return
	var destination: String = output_root + "/partial-originals"
	if DirAccess.dir_exists_absolute(ProjectSettings.globalize_path(destination)): return
	if DirAccess.make_dir_recursive_absolute(ProjectSettings.globalize_path(destination)) != OK: return
	for name: String in COMPANIONS:
		var source: String = user_root + name
		var item: Dictionary = {"name": name, "original_path": source, "present": FileAccess.file_exists(source), "scope": "partial original only"}
		if item.present:
			var bytes: PackedByteArray = FileAccess.get_file_as_bytes(source)
			item["bytes"] = bytes.size()
			item["sha256"] = _sha256(bytes)
			item["path"] = destination + "/" + name
			item["copied"] = _write_new_bytes(item.path, bytes)
		partial_files.append(item)

	var report_ok: bool = false
	if not report_path.is_empty() and DirAccess.dir_exists_absolute(ProjectSettings.globalize_path(output_root)) and not FileAccess.file_exists(report_path):
		var file: FileAccess = FileAccess.open(report_path, FileAccess.WRITE)
		if file != null:
			file.store_string(JSON.stringify(report, "\t")); file.flush()
			report_ok = file.get_error() == OK
			file.close()
	print("A2-L3 ORIGINAL ENTRANCE: %d checks, %d failures; complete=%s; report=%s" % [assertions.size(), _failure_count(), str(complete and report_ok).to_lower(), report_path])
	quit(0 if complete and report_ok else 1)

func _expect(ok: bool, label: String) -> bool:
	var row: Dictionary = {"index": assertions.size() + 1, "passed": ok, "label": label}
	assertions.append(row)
	if not ok:
		assertion_errors.append(label) # Assertion diagnostics only; external runner owns raw script/resource errors.
		if first_failure.is_empty(): first_failure = row.duplicate(true)
		push_error(label)
	return ok

func _failure_count() -> int:
	var count: int = 0
	for row: Dictionary in assertions:
		if not row.passed: count += 1
	return count

func _hash_string(value: String) -> bool:
	if value.length() != 64: return false
	for i: int in range(value.length()):
		if not value.substr(i, 1) in "0123456789abcdef": return false
	return true

func _sha256(bytes: PackedByteArray) -> String:
	var context := HashingContext.new()
	context.start(HashingContext.HASH_SHA256)
	context.update(bytes)
	return context.finish().hex_encode()
