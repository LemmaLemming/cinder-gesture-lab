func _setup_dash(direction: Vector3) -> bool:
	for _i: int in range(120):
		if finishing or not is_instance_valid(game): return false
		if paused: return _expect(false, "actual pause interrupted fixed setup; no forced resume/focus")
		if _stable() and float(_hero().get_threat_response_state().dash_cooldown_left_s) == 0.0: break
		await _step()
	if finishing or not is_instance_valid(game): return false
	var before: int = _sequence()
	if not _expect(_hero().request_dash(direction), "fixed setup ordinary controller dash accepted"): return false
	for _i: int in range(120):
		if finishing or not is_instance_valid(game): return false
		if paused: return _expect(false, "actual pause interrupted fixed setup motion; no forced resume/focus")
		await _step()
		if finishing or not is_instance_valid(game): return false
		if _sequence() > before and _stable():
			var record: Dictionary = _hero().get_world_action_records().back()
			return _expect(record.kind == "dash" and record.landing == _hero().global_position, "fixed seed uses actual completed stop/collision-shortened receipt")
	return _expect(false, "fixed setup dash did not complete; no alternate direction")

func _drive(goal: String) -> bool:
	var selected: Dictionary = {}
	var cursor: int = 0
	var dash_sequence: int = -1
	var key: String = ""
	for _i: int in range(MAX_ARM_FRAMES):
		if not _healthy(): return false
		var current: Dictionary = _state()
		if goal == "eligibility" and not component.get("notices").is_empty(): return true
		if goal == "threshold" and current.actor_hp == 15.0 and current.transition_pending: return true
		# A real legal earlier primary preempts planned waits/moves. No timing,
		# source availability, direction or proof-index preference is invented.
		if goal != "eligibility" and _primary_ready(goal): return await _primary(goal)
		var offer: Dictionary = _offer(goal)
		if failures > 0: return false
		if not offer.is_empty() and offer.key != key and _stable():
			selected = offer.proof.duplicate(true); key = offer.key; cursor = 0; dash_sequence = -1
			commands.append({"kind": "selected-first-current-legal-offer", "key": key, "clock_s": _clock(), "proof": selected.duplicate(true)})
		# Consume already-finished wait segments in this same control turn;
		# do not add a fixture-only physics delay before the next legal input.
		while not selected.is_empty() and cursor < selected.path.size():
			var segment: Dictionary = selected.path[cursor]
			if _clock() < float(segment.start_s): break
			if segment.kind in ["escape_dash", "positioning_dash"]:
				if dash_sequence < 0:
					if _clock() > float(segment.end_s) + Scheduler.EPSILON: return _expect(false, "first legal response dash window expired; no new admission")
					if _stable() and float(_hero().get_threat_response_state().dash_cooldown_left_s) == 0.0:
						dash_sequence = _sequence()
						var direction: Vector3 = (segment["to"] - segment["from"]).normalized()
						commands.append({"kind": "dash", "clock_s": _clock(), "direction": direction, "proof_key": key})
						if not _expect(_hero().request_dash(direction), "ordinary proof-selected dash accepted without forced direction"): return false
					break
				elif _sequence() > dash_sequence and _stable():
					cursor += 1; dash_sequence = -1
					continue
				break
			elif _clock() >= float(segment.end_s):
				if segment.kind == "ordinary_primary" and goal != "eligibility": return _expect(false, "first response ended without useful ordinary opening")
				cursor += 1
				continue
			break
		await _step()
	return _expect(false, "bounded first native response incomplete; no repeated convenient cycle")

func _offer(goal: String) -> Dictionary:
	var current: Dictionary = _state()
	if current.ray.cycle > (2 if goal == "paired" else 1) or current.foot.cycle > (2 if goal == "paired" else 1):
		_expect(false, "native first-cycle scope rolled over; stop instead of waiting for a convenient proof")
		return {}
	var pair: Dictionary = current.pair
	if goal == "paired" and pair.get("validated", false):
		if first_pair.is_empty():
			first_pair = pair.duplicate(true)
			paired_bait_sample = current.ray.bait_sample.duplicate(true)
		return {"key": "paired-%d-%d" % [pair.foot_cycle, pair.ray_cycle], "proof": pair.proof}
	if goal == "threshold" and current.ray.get("proof_available", false) and current.ray.proof.get("accepted", false): return {"key": "ray-%d" % current.ray.cycle, "proof": current.ray.proof}
	if goal in ["eligibility", "paired"] and current.foot.status == "running":
		for row: Dictionary in current.accepted_cycles:
			if row.kind == "foot" and row.cycle == current.foot.cycle and row.get("proof", {}).get("accepted", false): return {"key": "foot-%d" % row.cycle, "proof": row.proof}
	return {}

func _primary_ready(goal: String) -> bool:
	if not _stable(): return false
	var response: Dictionary = _hero().get_threat_response_state()
	var current: Dictionary = _state()
	if float(response.primary_cooldown_left_s) != 0.0 or current.joint_cue.state != "available" or current.ray.status != "running" or current.ray.phase != "recovery": return false
	if goal == "paired" and (not current.pair.get("validated", false) or current.foot.status != "running" or current.foot.phase != "recovery"): return false
	var upper: float = float(current.ray.exchange.recovery_until_s) - Scheduler.TIME_MARGIN
	if goal == "paired": upper = minf(upper, float(current.pair.foot_exchange.recovery_until_s) - Scheduler.TIME_MARGIN)
	return _clock() + float(response.stats.primary_cooldown) <= upper and Geometry.planar(_hero().global_position - _actor().global_position).length() <= float(response.stats.primary_range) - Scheduler.SKIN and _los().is_empty()

func _primary(goal: String) -> bool:
	var current: Dictionary = _state()
	var before: float = current.actor_hp
	var start: float = _clock()
	var finish: float = start + float(_hero().get_threat_response_state().stats.primary_cooldown)
	var upper: float = float(current.ray.exchange.recovery_until_s) - Scheduler.TIME_MARGIN
	if goal == "paired":
		if first_pair.is_empty():
			first_pair = current.pair.duplicate(true)
			paired_bait_sample = current.ray.bait_sample.duplicate(true)
		upper = minf(upper, float(current.pair.foot_exchange.recovery_until_s) - Scheduler.TIME_MARGIN)
	var receipt: Dictionary = {"clock_s": start, "complete_s": finish, "original_upper_s": upper, "actual_position": _hero().global_position, "joint_position": _actor().global_position, "range": _hero().get_threat_response_state().stats.primary_range, "los": _los(), "hp_before": before, "state_before": current.duplicate(true)}
	if not _expect(_hero().slash(_actor().global_position - _hero().global_position) == 1, "earliest legal ordinary primary physically reaches the actual low joint"): return false
	receipt["hp_after"] = _state().actor_hp
	commands.append({"kind": "ordinary_primary", "receipt": receipt})
	while _clock() < finish:
		if not _healthy(): return false
		await _step()
	if failures > 0: return false
	if not _expect(_clock() <= upper + Scheduler.EPSILON and _hero().hp == initial_hp and not _hero().dead and _stable(), "complete actual primary cooldown/hold fits original recoveries with unchanged HP"): return false
	if goal == "paired":
		outcome = receipt.duplicate(true)
		return _expect(before == 15.0 and _state().actor_hp == 0.0 and _state().stage == "cleared", "real second finite pool ordinary outcome; no host route/exit claim")
	return _expect(before == 30.0 and _state().actor_hp == 15.0, "actual ordinary first-pool cap earns15 rather than seeded HP")

func _step() -> void:
	if finishing or not is_instance_valid(probe): return
	await probe.after_tick
	arm_frames += 1
	if is_instance_valid(component) and component.get("component_ready"):
		var current: Dictionary = _state()
		var sample: Dictionary = _position_sample()
		var radius: float = _radius()
		var active: Array[Dictionary] = _active_exchanges(current)
		var clears_active: bool = true
		if not motion_previous.is_empty() and float(sample.clock_s) > float(motion_previous.clock_s):
			var interval: Array[Dictionary] = [{"start_s": motion_previous.clock_s, "end_s": sample.clock_s, "from": motion_previous.position, "to": sample.position}]
			for exchange: Dictionary in active + previous_exchanges:
				if Geometry.timed_path_hits(exchange.geometry, interval, exchange.active_from_s, exchange.active_until_s, radius): clears_active = false
		motion_previous = sample.duplicate(true)
		previous_exchanges = active.duplicate(true)
		if trace_enabled:
			if not previous.is_empty() and float(sample.clock_s) > float(previous.clock_s): trace.append({"kind": "actual_native_interval", "start_s": previous.clock_s, "end_s": sample.clock_s, "from": previous.position, "to": sample.position})
			previous = sample.duplicate(true)
			if pair_start.is_empty() and not current.pair.is_empty(): pair_start = {"foot_start_s": current.pair.foot_exchange.start_s, "sample": sample.duplicate(true), "bait": component.get("_bait").call("state")}
		var geometry_points: Array = component.call("_sentry_required_points")
		var held: bool = current.ray.status == "running" or current.foot.status == "running"
		var framing: bool = not geometry_points.is_empty() and bool(component.call("_sentry_points_framed", geometry_points))
		var supported: bool = false
		for region: Dictionary in component.call("floor_bindings").values():
			if region.safe_rect.grow(-radius).has_point(Geometry.planar(sample.position)): supported = true
		var contacts: Array[Dictionary] = []
		for i: int in range(_hero().get_slide_collision_count()):
			var c: KinematicCollision3D = _hero().get_slide_collision(i)
			contacts.append({"position": c.get_position(), "normal": c.get_normal(), "collider_decimal": str(c.get_collider_id())})
		var compact: Dictionary = {}
		for field: String in ["stage", "clock_s", "actor_hp", "boss_phase", "transition_pending", "joint_cue"]: compact[field] = current[field]
		compact["held_exchanges"] = active
		for source: String in ["ray", "foot"]:
			compact[source] = {}
			for field: String in ["status", "phase", "cycle", "direction", "armed", "geometry", "exchange", "reservation_id", "last_cancel_reason"]:
				if current[source].has(field): compact[source][field] = current[source][field]
		var row: Dictionary = {"sample": sample, "state": compact, "capsule": _capsule_values(), "supported_floor_footprint": supported, "on_floor": _hero().is_on_floor(), "contacts": contacts, "held_union_required": held, "complete_union_framed": framing, "actual_interval_clears_active": clears_active, "required_points": geometry_points, "actor_basis": _actor().global_basis, "actor_position": _actor().global_position}
		if tick_log != null:
			tick_log.store_line(JSON.stringify(_safe(row), "", true, true)); tick_rows += 1
		_expect(supported and clears_active and (not held or framing) and _hero().hp == initial_hp and not _hero().dead and _hero().equipment.snapshot() == Equipment.STARTER and Codec.same_values(_hero().stats, neutral_stats), "each native interval retains supported complete capsule/current armed-held union, neutral stats and immunity-free unchanged HP")

func _active_exchanges(current: Dictionary) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	if current.ray.status == "running" and current.ray.get("armed", false) and not current.ray.exchange.is_empty(): result.append(current.ray.exchange.duplicate(true))
	if current.foot.status == "running":
		for row: Dictionary in current.accepted_cycles:
			if row.kind == "foot" and row.cycle == current.foot.cycle: result.append(row.admitted_exchange.duplicate(true)); break
	return result

func _healthy() -> bool:
	if finishing or failures > 0 or not is_instance_valid(game) or not is_instance_valid(component): return false
	if paused: return _expect(false, "actual pause interrupted the fixed comparison; no forced resume/focus")
	if not component.get("errors").is_empty(): return _expect(false, "first actual B05/bait refusal: " + str(component.get("errors")))
	if Time.get_ticks_msec() - started_ms > WALL_LIMIT_MS or arm_frames > MAX_ARM_FRAMES or _clock() > MAX_ARM_NATIVE_S: return _expect(false, "fixed bounded comparison budget; no impossibility inference")
	return not _hero().dead

func _position_sample() -> Dictionary:
	return {"clock_s": _clock(), "position": _hero().global_position, "velocity": _hero().velocity, "hp": _hero().hp, "response": _hero().get_threat_response_state(), "equipment": _hero().equipment.snapshot(), "capsule": _capsule_values()}

func _capsule_values() -> Dictionary:
	var body: Dictionary = BodySweep.source_description(_hero())
	if body.has("error"): return body
	return {"footprint_half": body.footprint_half, "height": body.height, "foot_offset": body.foot_offset, "collision_instance_decimal": str(body.collision.get_instance_id())}

func _radius() -> float:
	var description: Dictionary = BodySweep.source_description(_hero())
	if description.has("error"): _expect(false, String(description.error)); return Scheduler.CAPSULE_RADIUS + Scheduler.SKIN
	return maxf(Scheduler.CAPSULE_RADIUS, maxf(description.footprint_half.x, description.footprint_half.y)) + Scheduler.SKIN

func _los() -> Dictionary:
	return _hero().get_world_3d().direct_space_state.intersect_ray(PhysicsRayQueryParameters3D.create(_hero().global_position + Vector3.UP * 0.7, _actor().global_position + Vector3.UP * 0.7, 1))

func _stable() -> bool:
	if not is_instance_valid(game) or not is_instance_valid(_hero()): return false
	var r: Dictionary = _hero().get_threat_response_state()
	return r.stable and r.motion.grounded and not _hero().action_in_progress()

func _sequence() -> int:
	var records: Array[Dictionary] = _hero().get_world_action_records()
	return 0 if records.is_empty() else int(records.back().sequence)
func _hero() -> CinderPlayer: return game.get("player") as CinderPlayer
func _parent() -> Node3D: return component.get("_sentry") as Node3D
func _actor() -> Node3D: return _parent().call("get_actor") as Node3D
func _state() -> Dictionary: return _parent().call("state")
func _clock() -> float: return component.get("_scheduler").get_clock() if is_instance_valid(component) and is_instance_valid(component.get("_scheduler")) else 0.0

func _persist(label: String, required: bool) -> bool:
	# Retain raw diagnostics BEFORE strict capture or Store validation can refuse.
	var was_paused: bool = paused
	paused = true
	await process_frame
	await process_frame
	var stem: String = arm + "-" + label
	if tick_log != null: tick_log.flush()
	if event_log != null: event_log.flush()
	if not _write_exact(stem + ".diagnostic.native.json", _diagnostic()): paused = was_paused; return false
	if not is_instance_valid(component) or not component.get("component_ready"):
		paused = was_paused
		return not required or _expect(false, "component not configured; partial setup diagnostics already retained")
	var bindings: Dictionary = component.call("scheduler_bindings")
	var player: Dictionary = _hero().snapshot_state()
	var scheduler: Dictionary = component.get("_scheduler").snapshot_state(bindings)
	var sentry: Dictionary = _parent().call("snapshot_state", bindings)
	var payload: Dictionary = {"kind": "TEST:A2-L5:two-owner-component-1", "construction": component.get("_construction_descriptor").duplicate(true), "player": player, "scheduler": scheduler, "sentry": sentry}
	# This exact partial or complete payload is written before validator/Store.
	if not _write_exact(stem + ".payload.native.json", payload): paused = was_paused; return false
	var problem: String = _component_payload_error(payload)
	var record: Dictionary = {"label": label, "capture_error": problem, "player_error": _hero().last_snapshot_error, "scheduler_error": component.get("_scheduler").last_snapshot_error, "parent_error": _parent().get("last_snapshot_error"), "scope": "actual two-owner component; not L5 whole local27/Attempts or historical save"}
	if problem.is_empty():
		var store: CinderSaveStore = Store.new(output + stem + ".component-store.json")
		store.payload_validator = _component_payload_error
		var stored: bool = store.write_payload(payload)
		record["store_write"] = stored
		record["store_error"] = store.last_error
		for suffix: String in ["", ".bak"]:
			var path: String = output + stem + ".component-store.json" + suffix
			if FileAccess.file_exists(path): record["raw_store" + suffix] = {"path": path, "sha256": FileAccess.get_sha256(path), "bytes": FileAccess.get_file_as_bytes(path).size()}
		if not stored: problem = store.last_error
	snapshots.append(record)
	_write_exact(stem + ".capture-result.native.json", record)
	paused = was_paused
	return problem.is_empty() or not required or _expect(false, "actual component capture/Store refusal, with payload/diagnostics already retained: " + problem)

func _component_payload_error(value: Dictionary) -> String:
	var error: String = Codec.value_error(value)
	if error.is_empty(): error = Codec.keys_error(value, ["kind", "construction", "player", "scheduler", "sentry"])
	if not error.is_empty(): return error
	if value.kind != "TEST:A2-L5:two-owner-component-1" or not value.player is Dictionary or not value.scheduler is Dictionary or not value.sentry is Dictionary: return "actual closed component packet required"
	error = Construction.descriptor_error(value.construction)
	if error.is_empty() and not _same(value.construction, component.get("_construction_descriptor")): error = "component descriptor differs from configured native definition"
	if error.is_empty() and (value.player.is_empty() or value.scheduler.is_empty() or value.sentry.is_empty()): error = "actual component writer returned incomplete state"
	if error.is_empty(): error = _hero().snapshot_error(value.player)
	if error.is_empty(): error = String(_parent().call("snapshot_error", value.sentry, component.call("scheduler_bindings"), value.player, value.scheduler))
	return error

func _diagnostic() -> Dictionary:
	var row: Dictionary = {"arm": arm, "wall_elapsed_ms": Time.get_ticks_msec() - started_ms, "paused": paused, "first_failure": first_failure, "fixed_station": STATION, "tick_log": {"path": output + arm + "-ticks.jsonl", "rows": tick_rows}, "commands": commands, "trace": trace, "snapshots": snapshots, "source_gates": source_before}
	if is_instance_valid(game):
		row["scene_load_error"] = game.get("level_load_error")
		if is_instance_valid(game.get("camera")): row["camera"] = {"position": game.get("camera").global_position, "basis": game.get("camera").global_basis, "size": game.get("camera").size}
		if is_instance_valid(game.get("hud")): row["hud_safe_rect"] = game.get("hud").combat_safe_rect()
		if is_instance_valid(_hero()):
			row["hero"] = {"position": _hero().global_position, "velocity": _hero().velocity, "hp": _hero().hp, "shells": _hero().shells, "equipment": _hero().equipment.snapshot(), "response": _hero().get_threat_response_state(), "actions": _hero().get_world_action_records(), "body": _capsule_values()}
	if is_instance_valid(component):
		row["errors"] = component.get("errors")
		row["eligibility"] = component.get("notices")
		row["publications_actions_hits"] = {"path": output + arm + "-events.jsonl", "rows": event_rows}
		if is_instance_valid(component.get("_bait")): row["bait"] = component.get("_bait").call("state")
		if is_instance_valid(component.get("_sentry")):
			row["parent"] = _state()
			row["parent_runtime_error"] = _parent().get("runtime_error")
			if is_instance_valid(_parent().call("get_ray")): row["ray_last_error"] = _parent().call("get_ray").get("last_error")
			if is_instance_valid(_parent().call("get_foot")): row["foot_last_error"] = _parent().call("get_foot").get("last_error")
	return row

func _compare() -> void:
	var old: Dictionary = results.v1
	var fresh: Dictionary = results.v2
	var a: Dictionary = old.pair
	var b: Dictionary = fresh.pair
	var matched: bool = Geometry.planar(old.start.sample.position - fresh.start.sample.position).length() <= Scheduler.EPSILON and _same(old.start.sample.equipment, fresh.start.sample.equipment) and _same(old.start.sample.response.stats, fresh.start.sample.response.stats) and old.start.sample.hp == fresh.start.sample.hp and _same(old.start.bait.get("initial_floor"), fresh.start.bait.get("initial_floor")) and _same(a.ray_exchange.geometry, b.ray_exchange.geometry) and _same(old.ray_bait_sample.get("point"), fresh.ray_bait_sample.get("point")) and old.ray_bait_sample.get("boundary") == fresh.ray_bait_sample.get("boundary") and absf(float(old.radius) - float(fresh.radius)) <= Scheduler.EPSILON and _same(old.start.sample.capsule.footprint_half, fresh.start.sample.capsule.footprint_half) and _same(old.start.sample.capsule.height, fresh.start.sample.capsule.height) and _same(old.start.sample.capsule.foot_offset, fresh.start.sample.capsule.foot_offset) and _same(old.start.sample.velocity, fresh.start.sample.velocity) and old.start.sample.response.dash_cooldown_left_s == fresh.start.sample.response.dash_cooldown_left_s and old.start.sample.response.primary_cooldown_left_s == fresh.start.sample.response.primary_cooldown_left_s
	# No source or timestamp is changed: normalize only the copied diagnostic
	# trace relative to the actual first-Foot start in each arm.
	for kind: String in ["foot_exchange", "ray_exchange"]:
		for field: String in ["start_s", "lock_from_s", "active_from_s", "active_until_s", "recovery_until_s"]:
			if absf((float(a[kind][field]) - float(a.foot_exchange.start_s)) - (float(b[kind][field]) - float(b.foot_exchange.start_s))) > Scheduler.EPSILON: matched = false
	if not matched:
		note = "INCONCLUSIVE_UNMATCHED_ACTUAL_PAIR_CONTEXT"
		_write_exact("comparison.native.json", {"classification": note, "old": old, "candidate": fresh, "scope": "no counterfactual or novelty claim when actual entry/bait/lane/timing context differs"})
		return
	var translated: Array[Dictionary] = old.trace.duplicate(true)
	var delta: float = float(b.foot_exchange.start_s) - float(a.foot_exchange.start_s)
	for row: Dictionary in translated:
		row.start_s = float(row.start_s) + delta
		row.end_s = float(row.end_s) + delta
	var conflict: bool = Geometry.timed_path_hits(b.foot_exchange.geometry, translated, b.foot_exchange.active_from_s, b.foot_exchange.active_until_s, float(fresh.radius))
	var ray_conflict: bool = Geometry.timed_path_hits(b.ray_exchange.geometry, translated, b.ray_exchange.active_from_s, b.ray_exchange.active_until_s, float(fresh.radius))
	var spatially_different: bool = _different_strategy(old, fresh)
	note = "DISTINCT_COMPONENT_RESPONSE_NATIVE_COUNTERFACTUAL" if conflict and not ray_conflict and spatially_different else "EQUIVALENT_OR_UNPROVED_SPATIAL_STRATEGY"
	_write_exact("comparison.native.json", {"classification": note, "matched_actual_context": matched, "old_measured_trace_against_new_foot": conflict, "old_trace_against_matched_ray": ray_conflict, "different_executed_spatial_response": spatially_different, "normalized_old_trace": translated, "old": old, "candidate": fresh, "scope": "actual success in each native world plus copied old-trace counterfactual against matched actual new union; old inputs were NOT executed in the candidate world; no full-host save/GUI/route/art/balance/policy credit"})

func _different_strategy(old: Dictionary, fresh: Dictionary) -> bool:
	var a: Array[Vector3] = []
	var b: Array[Vector3] = []
	for row: Dictionary in old.commands:
		if row.kind == "dash": a.append(row.direction)
	for row: Dictionary in fresh.commands:
		if row.kind == "dash": b.append(row.direction)
	if a.size() != b.size(): return true
	for i: int in range(a.size()):
		if Geometry.planar(a[i] - b[i]).length() > Scheduler.EPSILON: return true
	return Geometry.planar(old.outcome.actual_position - fresh.outcome.actual_position).length() > Scheduler.EPSILON

func _cleanup_case() -> void:
	if not is_instance_valid(game): return
	var refs: Array[WeakRef] = []
	_collect_refs(game, refs)
	# Public cleanup while all native descendants still have their actual tree.
	if is_instance_valid(component): component.call("exit_level")
	if is_instance_valid(game.get("active_level")): game.get("active_level").call("exit_level")
	game.free(); game = null; component = null
	if tick_log != null: tick_log.flush(); tick_log.close(); tick_log = null
	if event_log != null: event_log.flush(); event_log.close(); event_log = null
	await process_frame
	await process_frame
	for ref: WeakRef in refs: _expect(ref.get_ref() == null, "normal partial/whole component, Hero, world, HUD and source cleanup")

func _event(value: Dictionary) -> void:
	if event_log != null:
		event_log.store_line(JSON.stringify(_safe(value), "", true, true)); event_rows += 1

func _collect_refs(node: Node, refs: Array[WeakRef]) -> void:
	refs.append(weakref(node))
	for child: Node in node.get_children(): _collect_refs(child, refs)

func _expect(ok: bool, label: String) -> bool:
	checks += 1
	if not ok:
		failures += 1
		if first_failure.is_empty(): first_failure = {"label": label, "arm": arm, "clock_s": _clock(), "wall_elapsed_ms": Time.get_ticks_msec() - started_ms}
		push_error(label)
	return ok

func _finish() -> void:
	if finishing: return
	finishing = true
	if is_instance_valid(game): await _persist("abort-before-cleanup", false)
	await _cleanup_case()
	if tick_log != null: tick_log.flush(); tick_log.close(); tick_log = null
	if event_log != null: event_log.flush(); event_log.close(); event_log = null
	if is_instance_valid(probe): probe.free()
	for path: String in source_before: _expect(FileAccess.get_sha256(path) == source_before[path], "source gate remains exact after execution: " + path)
	var complete: bool = failures == 0 and note == "DISTINCT_COMPONENT_RESPONSE_NATIVE_COUNTERFACTUAL"
	var report: Dictionary = {"checks": checks, "failures": failures, "scope_complete": complete, "classification": note, "first_failure": first_failure, "results_present": results.keys(), "snapshots": snapshots, "output": output, "scope": "one fixed station/setup, two native component definitions/actual ordinary responses; conditional matched native old-trace counterfactual only. No Retry/restore, literal historical Challenge packet, strict L5 local27/Attempts, route, full-playthrough, portrait, balance or difficulty acceptance claim."}
	if output_created:
		var file: FileAccess = FileAccess.open(output + "report.json", FileAccess.WRITE)
		if file != null: file.store_string(JSON.stringify(_safe(report), "\t", true, true)); file.close()
		else: failures += 1; complete = false; push_error("final report could not be retained")
	if process_frame.is_connected(_wall_guard): process_frame.disconnect(_wall_guard)
	AudioServer.set_bus_layout(old_audio); Engine.max_fps = old_fps
	paused = false
	print("A2-L5 fixed response comparison: %d checks, %d failures; scope_complete=%s; classification=%s; output=%s" % [checks, failures, str(complete), note, output])
	quit(1 if failures > 0 else (0 if complete else 2))

func _wall_guard() -> void:
	if not finishing and Time.get_ticks_msec() - started_ms > WALL_LIMIT_MS:
		_expect(false, "fixed wall watchdog; actual partial evidence retained, no route/impossibility conclusion")
		_finish.call_deferred()

func _write_exact(name: String, value: Variant) -> bool:
	if not output_created: return _expect(false, "output was not freshly created by this comparison")
	var wire: String = Exact.stringify(_safe(value))
	if wire.is_empty(): return _expect(false, "lossless diagnostic encoding refused before file write: " + name)
	var file: FileAccess = FileAccess.open(output + name, FileAccess.WRITE)
	if file == null: return _expect(false, "diagnostic output could not be opened: " + name)
	file.store_string(wire); file.close()
	return true

func _same(left: Variant, right: Variant) -> bool:
	var a: String = Exact.stringify(_safe(left))
	return not a.is_empty() and a == Exact.stringify(_safe(right))

func _safe(value: Variant) -> Variant:
	if value is StringName: return String(value)
	if value is Vector3: return [value.x, value.y, value.z]
	if value is Vector2: return [value.x, value.y]
	if value is Color: return [value.r, value.g, value.b, value.a]
	if value is AABB: return {"position": _safe(value.position), "size": _safe(value.size)}
	if value is Vector2i: return [value.x, value.y]
	if value is Vector3i: return [value.x, value.y, value.z]
	if value is Rect2: return {"position": _safe(value.position), "size": _safe(value.size)}
	if value is Basis: return {"x": _safe(value.x), "y": _safe(value.y), "z": _safe(value.z)}
	if value is Transform3D: return {"basis": _safe(value.basis), "origin": _safe(value.origin)}
	if value is int and (value > 9007199254740991 or value < -9007199254740991): return {"native_integer_decimal": str(value)}
	if value is Object: return {"native_class": value.get_class(), "native_instance_decimal": str(value.get_instance_id())} if is_instance_valid(value) else {"freed_native_object": true}
	if value is RID: return {"native_rid_decimal": str(value.get_id())}
	if value is Dictionary:
		var result: Dictionary = {}
		for key: Variant in value: result[String(key)] = _safe(value[key])
		return result
	if value is PackedVector3Array or value is PackedVector2Array or value is PackedFloat32Array or value is PackedFloat64Array or value is PackedInt32Array or value is PackedInt64Array or value is PackedByteArray or value is PackedStringArray: return _safe(Array(value))
	if value is Array:
		var result: Array = []
		for child: Variant in value: result.append(_safe(child))
		return result
	return value
