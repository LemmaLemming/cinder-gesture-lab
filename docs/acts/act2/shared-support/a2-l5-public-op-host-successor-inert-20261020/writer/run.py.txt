def require(ok, reason):

def digest(data):
    return hashlib.sha256(data).hexdigest()


def strict(data):
    def pairs(items):
        result = {}
        for key, value in items:
            require(key not in result, "Duplicate JSON key: " + key)
            result[key] = value
        return result
    def finite(token):
        value = float(token)
        require(math.isfinite(value), "Nonfinite JSON exponent")
        return value
    return json.loads(data.decode("utf-8", errors="strict"), object_pairs_hook=pairs,
                      parse_float=finite,
                      parse_constant=lambda token: (_ for _ in ()).throw(ValueError(token)))


def stable(path):
    path = Path(path)
    require(path.is_absolute() and path.resolve() == path, "Literal nonsymlink absolute path required")
    before = path.lstat()
    require(stat.S_ISREG(before.st_mode), "Regular file required")
    fd = os.open(path, os.O_RDONLY | getattr(os, "O_NOFOLLOW", 0))
    with os.fdopen(fd, "rb") as stream:
        opened = os.fstat(stream.fileno())
        require((before.st_dev, before.st_ino) == (opened.st_dev, opened.st_ino), "Changed before open")
        data = stream.read()
        closed_fd = os.fstat(stream.fileno())
    second = path.read_bytes()
    after = path.lstat()
    key = lambda row: (row.st_dev, row.st_ino, row.st_size, row.st_mode, row.st_mtime_ns)
    require(key(before) == key(opened) == key(closed_fd) == key(after) and data == second,
            "Unstable literal two-read observation")
    return {"path": str(path), "sha256": digest(data), "bytes": len(data),
            "mode": format(stat.S_IMODE(after.st_mode), "04o"), "stable_two_reads": True}, data

def fsync_directory(path):
    fd = os.open(path, os.O_RDONLY)
    try: os.fsync(fd)
    finally: os.close(fd)


def save(path, value):
    data = (json.dumps(value, indent=2, allow_nan=False) + "\n").encode()
    temporary = path.with_name(path.name + ".owned-" + uuid.uuid4().hex + ".tmp")
    with temporary.open("xb") as stream:
        stream.write(data); stream.flush(); os.fsync(stream.fileno())
    os.replace(temporary, path); fsync_directory(path.parent)


def record(path):
    path = Path(path)
    require(path.is_absolute() and path.parent.resolve() == path.parent, "No symlink parent allowed")
    st = path.lstat() if os.path.lexists(path) else None
    regular = bool(st and stat.S_ISREG(st.st_mode))
    data = stable(path)[1] if regular else None
    return {"exists": st is not None, "regular": regular, "symlink": bool(st and stat.S_ISLNK(st.st_mode)),
            "sha256": digest(data) if data is not None else None, "bytes": len(data) if data is not None else None,
            "mode": format(stat.S_IMODE(st.st_mode), "04o") if st else None}, data


def write_literal(path, data):
    with path.open("xb") as stream:
        stream.write(data); stream.flush(); os.fsync(stream.fileno())
    fsync_directory(path.parent)

def replace_subject(path, data, mode, expected_prior, on_created=None, expected_identity=None):
    """O2 final-prior-check replacement, plus exclusive link for genuine absence."""
    require(expected_prior == ABSENT or expected_prior["regular"] and not expected_prior["symlink"], "Known regular or absent prior required")
    temporary = path.with_name(path.name + ".l5-fixed-owned-" + uuid.uuid4().hex + ".tmp")
    try:
        write_literal(temporary, data); os.chmod(temporary, int(mode, 8))
        require(record(path)[0] == expected_prior, "Final prior guard changed; unknown left in situ")
        if expected_prior != ABSENT:
            prior_stat = path.lstat()
            require(expected_identity is not None and expected_identity == {"device": prior_stat.st_dev, "inode": prior_stat.st_ino},
                    "Final prior inode changed; identical foreign file left in situ")
        created_stat = temporary.lstat()
        identity = {"device": created_stat.st_dev, "inode": created_stat.st_ino}
        if expected_prior == ABSENT:
            os.link(temporary, path)  # exclusive no-clobber even for identical foreign arrivals
        else:
            os.replace(temporary, path)
        if on_created is not None: on_created(identity)
        fsync_directory(path.parent)
    finally:
        if os.path.lexists(temporary): temporary.unlink()


def retain_unknown(path, out, label):

    save(out / (label + ".json"), meta)
    return row


def restore_subject(path, initial, initial_data, candidate, out, label, owned_closed, created_by_this_run, creation_identity=None):
    receipt = {"path": str(path), "owned_children_closed": owned_closed, "created_by_this_run": created_by_this_run,
               "restored_exact_initial": False, "write_attempted": False, "under_canonical_lock": True,
               "unknown_overwrite_authorized": False}
    try:
        current = record(path)[0]
        if not owned_closed:
            receipt["status"] = "OWNED_CHILD_NOT_CLOSED_NO_WRITE"
        elif current == initial:
            receipt.update(status="ALREADY_EXACT_INITIAL_NO_WRITE", restored_exact_initial=True)
        elif current == candidate and created_by_this_run:
            require(type(creation_identity) is dict and set(creation_identity) == {"device", "inode"}
                    and all(type(value) is int for value in creation_identity.values()), "Actual own inode identity required")
            require(record(path)[0] == candidate, "Final candidate bytes/mode/type guard changed")
            current_stat = path.lstat()
            require(creation_identity == {"device": current_stat.st_dev, "inode": current_stat.st_ino},
                    "Identical foreign replacement inode left in situ")
            receipt["write_attempted"] = True
            if initial == ABSENT:
                path.unlink(); fsync_directory(path.parent)
                receipt.update(status="REMOVED_ONLY_EXACT_OWN_CREATED_CANDIDATE", restored_exact_initial=record(path)[0] == ABSENT)
            else:
                replace_subject(path, initial_data, initial["mode"], candidate, expected_identity=creation_identity)
                receipt.update(status="RESTORED_EXACT_INITIAL_BYTES_MODE", restored_exact_initial=record(path)[0] == initial)
        else:
            receipt["status"] = "UNKNOWN_OR_UNOWNED_CANDIDATE_LEFT_IN_SITU"
        if not receipt["restored_exact_initial"]:
            receipt["observed"] = retain_unknown(path, out, label + "-unexpected")
    except Exception as error:
        receipt.update(status="GUARDED_RESTORE_FAILED_NO_UNKNOWN_WRITE", error=str(error))
        receipt["observed"] = retain_unknown(path, out, label + "-guard-conflict")
    save(out / (label + "-restore.json"), receipt)
    return receipt

def copy_native(out):
    """Finite partial artifact retention before any JSON/outcome interpretation."""
    source = out / "native"; retained = out / "native-retained"
    require(source.is_dir() and source.resolve() == source, "Own native output root required")
    retained.mkdir(exist_ok=False)
    rows = {}; issues = []
    for name in NATIVE_NAMES:
        path = source / name
        if not os.path.lexists(path):
            rows[name] = {"present": False}; continue
        try:
            st = path.lstat()
            require(stat.S_ISREG(st.st_mode) and st.st_size <= 64 * 1024 * 1024,
                    "Finite regular native artifact over byte limit or unknown type")
            meta, literal = stable(path)
            target = retained / name; write_literal(target, literal); os.chmod(target, int(meta["mode"], 8))
            copy_meta, copied = stable(target)
            require(copied == literal and all(copy_meta[key] == meta[key] for key in ("sha256", "bytes", "mode")),
                    "Literal partial artifact retention differs")
            rows[name] = dict(meta, present=True, retained_copy=str(target.relative_to(out)), copied_verbatim=True)
        except Exception as error:
            issues.append({"name": name, "path": str(path), "error": str(error), "left_in_situ": True})
            rows[name] = {"present": True, "retention_error": str(error)}

        require(report.get("source_before") == report.get("source_after") == unit.get("source_before") == spec["native_common"],
                "Native exact named common-source controls differ")

                        and row.get("created_by_invocation") is True and row.get("cleanup", {}).get("remaining") == []
                        and row["cleanup"].get("unknown_retained") is False, "Exact native own-directory cleanup control differs")
                actual = Path(spec["user_data_root"]) / path.removeprefix("user://").removesuffix("/")
                require(not os.path.lexists(actual), "Native reported removed own directory is still present")
            if spec["operation"] == "writer":

                                and archive.get("path") == meta["path"] and archive.get("sha256") == slot.get("sha256") == meta["sha256"]
                                and archive.get("bytes") == slot.get("bytes") == meta["bytes"], "Literal original writer raw export receipt join differs")
                    else:
                        require(meta.get("present") is False, "Absent writer slot cannot have an invented raw export")
        expected_exit = 0 if positive else 2
        result["expected_native_exit"] = expected_exit
        result["functional_pass"] = (positive and type(api_return) is int and api_return == 0
            and type(child_return) is int and child_return == 0 and not custody.get("issues") and not result["errors"])
    except Exception as error:

def seal_input(spec, out, legacy_report):
    path = ROOT / spec["run_input"]
    require(record(path)[0] == ABSENT, "Existing native RUN_INPUT left untouched")
    native = out / "native"; native.mkdir(exist_ok=False)
    value = {"operation": spec["operation"], "output_dir": str(native),
             "source_pins_sha256": spec["source_pins_sha256"], "legacy_report": legacy_report}
    data = (json.dumps(value, indent=2, allow_nan=False) + "\n").encode()
    creation = {}
    def created(identity):
        creation.update(identity)
        save(out / "input-creation.json", {"path": str(path), "identity": identity, "sha256": digest(data), "closed4": value})
    replace_subject(path, data, "0644", ABSENT, created)
    row = record(path)[0]
    require(row["sha256"] == digest(data) and creation, "Own exact closed4 input creation failed")
    receipt = {"path": str(path), "record": row, "creation_identity": creation, "closed4": value}
    save(out / "sealed-native-input.json", receipt)
    return receipt


def run_native(dev, context, lock_fd, spec, out, result):
    user_base = Path(spec["user_data_root"])
    require(user_base == Path.home() / "Library/Application Support/Godot/app_userdata/Cinder"
            and user_base.resolve() == user_base, "Exact normal C7 user context required")
    argv = [spec["engine_expected"]["path"], "--path", str(ROOT), "--headless", "--log-file", str(out / "engine.log"),
            "--script", "res://" + spec["fixture"]]  # native leaf requires NO user args
    save(out / "command.json", {"argv": argv, "canonical_api": "dev.godot_lock/run_validation", "one_owned_child": True,
        "timeout_after_lock_seconds": spec["timeout_after_lock_seconds"], "cwd": str(ROOT), "user_data_root": str(user_base),
        "native_input": str(ROOT / spec["run_input"]), "native_output": str(out / "native"),
        "engine_environment": {key: context.environ.get(key) for key in ("CINDER_GODOT", "CINDER_CANONICAL_ROOT", "PYTHONDONTWRITEBYTECODE")}})
    owned = []; original_popen = dev.subprocess.Popen
    def tracked(command, *pos, **kw):
        require(command == argv and not owned and kw.get("cwd") == ROOT and kw.get("env") == context.environ
                and kw.get("pass_fds") == (lock_fd,), "Exactly one planned owned canonical child/context/lock required")
        child = original_popen(command, *pos, **kw); owned.append(child)
        result.update(native_child_launch_state="SPAWNED", owned_children=[{"pid": child.pid, "closed": False, "popen_return": None}])
        save(out / "result.json", result); return child
    result.update(engine_launch_attempted=True, native_child_launch_state="ADMISSION_PENDING")
    save(out / "result.json", result)
    try:
        dev.subprocess.Popen = tracked
        result["canonical_api_return"] = dev.run_validation(argv, context, lock_fd,
            spec["timeout_after_lock_seconds"], detect_godot_errors=True)
    finally:
        dev.subprocess.Popen = original_popen
        for child in owned:
            if child.poll() is None: dev.stop_validation(child)  # only this invocation's actual owned handle
            child.wait(timeout=5)
        result["owned_children"] = [{"pid": child.pid, "closed": child.poll() is not None, "popen_return": child.poll()} for child in owned]
        result["native_child_launch_state"] = "CLOSED" if len(owned) == 1 and owned[0].poll() is not None else "NO_CLOSED_CHILD_RECEIPT"
        save(out / "result.json", result); sys.stdout.flush(); sys.stderr.flush()
        if result["native_child_launch_state"] == "CLOSED":
            # Retain every finite partial even when report/unit is absent or outcome is negative.
            try:
                custody = copy_native(out)
            except Exception as error:

                result["native_artifact_custody_error"] = str(error)
            observation = observe(out, spec, custody, (out / "raw.log").read_text(),
                                  result["canonical_api_return"], owned[0].poll())
            result["native_observation"] = observation
            result["native_functional_pass"] = observation["functional_pass"] and not custody.get("issues")
            save(out / "native-observation.json", observation); save(out / "result.json", result)
    require(len(owned) == 1 and owned[0].poll() is not None, "One actual owned child closure required")
    require(result.get("native_functional_pass") is True, "Native negative/failure/partial/custody refusal; no automatic repeat")


def restore_all(spec, out, before, state, owned_closed, beforeimage_base=None):
    receipts = []
    beforeimage_base = out if beforeimage_base is None else beforeimage_base
    for index, (name, variants) in enumerate(reversed(list(spec["subjects"].items()))):
        row = before[name]
        require(row["record"] == variants["original"], "Beforeimage must be one exact known initial variant")
        require(type(state[name].get("created_by_this_run")) is bool, "Literal own-creation Boolean required")
        data = (beforeimage_base / row["blob"]).read_bytes() if row["record"]["regular"] else None
        if data is not None: require(digest(data) == row["record"]["sha256"], "Durable beforeimage changed")
        receipts.append(restore_subject(ROOT / name, row["record"], data, variants["candidate"], out,
                                        "subject-" + str(index), owned_closed, state[name]["created_by_this_run"],
                                        state[name].get("creation_identity")))
    save(out / "restore-receipts.json", receipts)
    return all(row["restored_exact_initial"] for row in receipts)
