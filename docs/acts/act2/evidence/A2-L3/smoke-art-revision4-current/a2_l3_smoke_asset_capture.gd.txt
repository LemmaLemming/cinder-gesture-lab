func _vector(value: Vector3) -> Array:
	return [value.x, value.y, value.z]


func _cue_json(state: Dictionary) -> Dictionary:
	var shape: Dictionary = state["geometry"]
	return {"api_revision": state["api_revision"], "phase": state["phase"], "geometry": {"kind": shape["kind"], "origin": _vector(shape["origin"]), "radius": shape["radius"]}, "source_position": _vector(state["source_position"]), "required": state["required"], "source_visible": state["source_visible"], "footprint_visible": state["footprint_visible"], "active_fill_visible": state["active_fill_visible"], "no_reservation_or_damage_authority": true}


func _source_vertices(cue: Node3D) -> Array[Vector3]:
	var result: Array[Vector3] = []
	var marker := cue.get_node_or_null("RequiredSourceMarker") as MeshInstance3D
	if marker == null or marker.mesh == null or not marker.is_visible_in_tree(): return result
	for surface: int in range(marker.mesh.get_surface_count()):
		var arrays: Array = marker.mesh.surface_get_arrays(surface)
		var vertices: PackedVector3Array = arrays[Mesh.ARRAY_VERTEX]
		for vertex: Vector3 in vertices: result.append(marker.global_transform * vertex)
	return result


func _circle_boundary(cue: Node3D) -> Array[Vector3]:
	var result: Array[Vector3] = []
	var state: Dictionary = cue.call("state")
	var outline := cue.get_node_or_null("RequiredFootprintOutline") as MeshInstance3D
	if outline == null or state.get("geometry", {}).get("kind") != "circle" or outline.transform != Transform3D(Basis.IDENTITY, Vector3.UP * ThreatCue.FLOOR_OFFSET): return result
	# The shared boundary is ordered local centerline geometry, not native
	# outline triangle vertices. Use the actual cue transform and outline Y.
	for point: Vector3 in CueMesh.boundary(state.geometry):
		result.append(cue.global_transform * (point + Vector3.UP * ThreatCue.FLOOR_OFFSET))
	return result


func _native_masks_valid(smoke: Node3D, camera: Camera3D, hero_points: Array[Vector3], source_points: Array[Vector3], boundary_points: Array[Vector3]) -> bool:
	var state: Dictionary = smoke.call("readability_state")
	if not state.masks_ready or not state.camera_current: return false
	var geometry: Dictionary = state.geometry
	if not geometry.has_all(["hero_rect_pixels", "source_rect_pixels", "boundary_pixels", "rim_padding_pixels", "boundary_count"]) or not geometry.hero_rect_pixels is Rect2 or not geometry.source_rect_pixels is Rect2 or not geometry.boundary_pixels is PackedVector2Array or geometry.boundary_count != 64 or geometry.boundary_pixels.size() != 64 or float(geometry.rim_padding_pixels) < SmokeVisual.MASK_PIXEL_MARGIN: return false
	if not geometry.hero_rect_pixels.has_area() or not geometry.source_rect_pixels.has_area(): return false
	for point: Vector3 in hero_points:
		if not geometry.hero_rect_pixels.has_point(camera.unproject_position(point)): return false
	for point: Vector3 in source_points:
		if not geometry.source_rect_pixels.has_point(camera.unproject_position(point)): return false
	for index: int in range(64):
		if geometry.boundary_pixels[index] != camera.unproject_position(boundary_points[index]): return false
	return true


func _shader_parameters(material: ShaderMaterial) -> Dictionary:
	var result: Dictionary = {}
	for key: String in MASK_PARAMETERS: result[key] = material.get_shader_parameter(key)
	return result


func _parcel_pose(smoke: Node3D) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	for child: Node in smoke.get_children():
		var part := child as MeshInstance3D
		if part == null: continue
		var material: Material = part.material_override
		var parcel: Dictionary = {"name": str(part.name), "transform": part.transform, "visible": part.visible, "mesh_instance_id": part.mesh.get_instance_id(), "material_instance_id": material.get_instance_id()}
		if material is StandardMaterial3D:
			parcel["kind"] = "base"
			parcel["tint"] = (material as StandardMaterial3D).albedo_color
		elif material is ShaderMaterial:
			parcel["kind"] = "wisp"
			var shader_material := material as ShaderMaterial
			parcel["shader_instance_id"] = shader_material.shader.get_instance_id()
			parcel["parameters"] = _shader_parameters(shader_material)
		result.append(parcel)
	return result


func _base_parcel_pose(smoke: Node3D) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	for parcel: Dictionary in _parcel_pose(smoke):
		if parcel.kind == "base": result.append(parcel)
	return result


func _material_policy(smoke: Node3D) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	for child: Node in smoke.get_children():
		var part := child as MeshInstance3D
		if part == null or not part.material_override is StandardMaterial3D: continue
		var material := part.material_override as StandardMaterial3D
		result.append({"parcel": str(part.name), "alpha": material.albedo_color.a, "priority": material.render_priority, "no_depth_test": material.no_depth_test, "depth_draw_mode": material.depth_draw_mode, "transparency": material.transparency})
	return result


func _wisp_material_policy(smoke: Node3D) -> Array[Dictionary]:
	var result: Array[Dictionary] = []
	for child: Node in smoke.get_children():
		var part := child as MeshInstance3D
		if part == null or not part.material_override is ShaderMaterial: continue
		var material := part.material_override as ShaderMaterial
		var code: String = material.shader.code
		result.append({"parcel": str(part.name), "visible": part.visible, "material_instance_id": material.get_instance_id(), "shader": material.shader.resource_path, "shader_matches": material.shader == SmokeVisual.WispShader, "alpha": float(material.get_shader_parameter("wisp_alpha")), "priority": material.render_priority, "masks_ready": material.get_shader_parameter("masks_ready") == true, "depth_tested": not code.contains("depth_test_disabled"), "no_depth_writes": code.contains("depth_draw_never"), "unshaded_mix_back_culled": code.contains("render_mode unshaded, blend_mix, depth_draw_never, cull_back;"), "mask_parameters": _shader_parameters(material)})
	return result


func _visible_wisps(smoke: Node3D) -> int:
	var result: int = 0
	for child: Node in smoke.get_children():
		if child is MeshInstance3D and child.material_override is ShaderMaterial and child.visible: result += 1
	return result


func _camera_signature(camera: Camera3D) -> Dictionary:
	var projection: Projection = camera.get_camera_projection()
	return {"transform": camera.get_camera_transform(), "projection": [projection.x, projection.y, projection.z, projection.w], "near": camera.near, "far": camera.far, "viewport": camera.get_viewport().get_visible_rect().size}


func _native_json(value: Variant) -> Variant:
	# Metadata is informational; native mask equality above remains typed and
	# exact. Encode native pixel/world values as numeric arrays, not strings.
	if value is Vector2: return [value.x, value.y]
	if value is Vector3: return _vector(value)
	if value is Vector4: return [value.x, value.y, value.z, value.w]
	if value is Rect2: return {"position": [value.position.x, value.position.y], "size": [value.size.x, value.size.y]}
	if value is PackedVector2Array or value is Array:
		var result: Array = []
		for item: Variant in value: result.append(_native_json(item))
		return result
	if value is Dictionary:
		var result: Dictionary = {}
		for key: Variant in value: result[key] = _native_json(value[key])
		return result
	return value


func _visible_parcels(smoke: Node3D) -> int:
	var result: int = 0
	for child: Node in smoke.get_children():
		if child is MeshInstance3D and child.visible:
			result += 1
	return result


func _settle(count: int) -> void:
	for ignored: int in range(count):
		await physics_frame
	await process_frame


func _expect(ok: bool, label: String) -> void:
	_checks += 1
	if not ok:
		_failures += 1
		push_error("FAIL: " + label)
